  function ensureStandaloneJobSystemTab(){
    const nav=document.querySelector('.ops-tabs');
    if(!nav||nav.querySelector('[data-tool="jobs"]'))return;
    const button=document.createElement('button');
    button.className='ops-tab';
    button.dataset.tool='jobs';
    button.type='button';
    button.textContent='Standalone Jobs & Events';
    const worldButton=nav.querySelector('[data-tool="world"]');
    nav.insertBefore(button,worldButton||null);
    button.addEventListener('click',()=>setTool('jobs'));
  }

  const activeToolBeforeStandaloneJobSystem=activeTool;
  activeTool=function activeToolWithStandaloneJobSystem(){
    const raw=window.location.hash.replace(/^#/,'');
    if(raw==='jobs'||raw==='job-system'||raw==='standalone-jobs')return 'jobs';
    return activeToolBeforeStandaloneJobSystem();
  };

  const renderBeforeStandaloneJobSystem=render;

  };
