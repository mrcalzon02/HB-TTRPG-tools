
  const ACTIVE_TRANSIT_TURN_VERSION = '1.0.0';

  function dashboardGraphIndex(map = state.world?.map) {
    if (!map?.nodes?.length) return { nodeById: {}, edgesByNode: {} };
    const cacheKey = `${map.worldId || map.seed || ''}|${map.generatedAt || ''}|${map.nodes.length}|${map.edges.length}`;
    if (globalThis.__BAROTRAUMA_DASHBOARD_GRAPH_CACHE?.key === cacheKey) return globalThis.__BAROTRAUMA_DASHBOARD_GRAPH_CACHE.value;
    const nodeById = Object.fromEntries(map.nodes.map(node => [node.id, node]));
    const edgesByNode = Object.fromEntries(map.nodes.map(node => [node.id, []]));
    for (const edge of map.edges || []) {
      if (edgesByNode[edge.a]) edgesByNode[edge.a].push({ edge, nextId: edge.b });
      if (edgesByNode[edge.b]) edgesByNode[edge.b].push({ edge, nextId: edge.a });
    }
    const value = { nodeById, edgesByNode };
    globalThis.__BAROTRAUMA_DASHBOARD_GRAPH_CACHE = { key: cacheKey, value };
    return value;
  }

  function dashboardAdjacentStationRoutes(originId, map = state.world?.map) {
    const { nodeById, edgesByNode } = dashboardGraphIndex(map);
    const origin = nodeById[originId];
    if (!origin || origin.type !== 'station') return [];
    const best = new Map([[originId, 0]]);
    const queue = [{ nodeId: originId, distance: 0, pathNodeIds: [originId], edgeIds: [], danger: false }];
    const routes = new Map();
    while (queue.length) {
      queue.sort((left, right) => left.distance - right.distance);
      const current = queue.shift();
      if (current.distance !== best.get(current.nodeId)) continue;
      for (const step of edgesByNode[current.nodeId] || []) {
        const next = nodeById[step.nextId];
        if (!next || current.pathNodeIds.includes(next.id)) continue;
        const distance = current.distance + Math.max(1, number(step.edge.distance, 1));
        const pathNodeIds = [...current.pathNodeIds, next.id];
        const edgeIds = [...current.edgeIds, step.edge.id];
        const danger = current.danger || Boolean(step.edge.danger);
        if (next.type === 'station' && next.id !== originId) {
          const existing = routes.get(next.id);
          if (!existing || distance < existing.distance) routes.set(next.id, {
            stationId: next.id,
            station: next,
            distance,
            danger,
            pathNodeIds,
            edgeIds
          });
          continue;
        }
        if (distance < (best.get(next.id) ?? Infinity)) {
          best.set(next.id, distance);
          queue.push({ nodeId: next.id, distance, pathNodeIds, edgeIds, danger });
        }
      }
    }
    return [...routes.values()].sort((left, right) => left.distance - right.distance || left.station.name.localeCompare(right.station.name));
  }

  function dashboardStationCorridors(map = state.world?.map) {
    if (!map?.nodes?.length) return [];
    const cacheKey = `${map.worldId || map.seed || ''}|${map.generatedAt || ''}|${map.nodes.length}|${map.edges.length}|station-corridors`;
    if (globalThis.__BAROTRAUMA_STATION_CORRIDOR_CACHE?.key === cacheKey) return globalThis.__BAROTRAUMA_STATION_CORRIDOR_CACHE.value;
    const corridors = new Map();
    for (const station of map.nodes.filter(node => node.type === 'station')) {
      for (const route of dashboardAdjacentStationRoutes(station.id, map)) {
        const key = [station.id, route.stationId].sort().join('|');
        const existing = corridors.get(key);
        if (!existing || route.distance < existing.distance) corridors.set(key, {
          id: key,
          a: station.id,
          b: route.stationId,
          distance: route.distance,
          danger: route.danger,
          pathNodeIds: route.pathNodeIds,
          edgeIds: route.edgeIds
        });
      }
    }
    const value = [...corridors.values()];
    globalThis.__BAROTRAUMA_STATION_CORRIDOR_CACHE = { key: cacheKey, value };
    return value;
  }

  function defaultDashboardTurnOrder() {
    return {
      version: ACTIVE_TRANSIT_TURN_VERSION,
      mode: 'rotation',
      order: [],
      activeIndex: 0,
      round: 1,
      turn: 1,
      encounterId: '',
      rolls: {},
      rolledAt: '',
      history: []
    };
  }

  function dashboardTurnActors() {
    const settings = normalizePlayerSubmissionSettings();
    const actors = [];
    if ((state.character.status || 'Onboard') !== 'Dead') actors.push({
      ref: 'local',
      crewId: settings.characterId,
      name: state.character.name || 'Current Character',
      status: state.character.status || 'Onboard',
      current: state.character.current,
      agility: number(state.character.current?.Agility),
      type: 'local'
    });
    const management = normalizeCrewManagement();
    for (const record of management.roster) {
      if (!record.active || record.approval !== 'Accepted' || !['Onboard','Injured'].includes(record.status)) continue;
      if (record.sourceCharacterId && record.sourceCharacterId === settings.characterId) continue;
      actors.push({
        ref: record.rosterId,
        crewId: activeDashboardCrewId(record),
        name: record.characterName || 'Crew Member',
        status: record.status,
        current: record.current,
        agility: number(record.current?.Agility),
        type: 'crew'
      });
    }
    return actors;
  }

  function normalizeDashboardTurnOrder() {
    const dashboard = state.operationsDashboard || (state.operationsDashboard = defaultActiveSubmarineDashboard());
    dashboard.operationalState = normalizeRouteCrossing().status === 'underway' ? 'in-route' : 'at-station';
    dashboard.worldViewerOpen = Boolean(dashboard.worldViewerOpen);
    dashboard.turnOrder = dashboard.turnOrder && typeof dashboard.turnOrder === 'object' ? dashboard.turnOrder : defaultDashboardTurnOrder();
    const turnOrder = dashboard.turnOrder;
    const defaults = defaultDashboardTurnOrder();
    for (const [key, value] of Object.entries(defaults)) if (!(key in turnOrder)) turnOrder[key] = Array.isArray(value) ? [] : value && typeof value === 'object' ? { ...value } : value;
    turnOrder.version = ACTIVE_TRANSIT_TURN_VERSION;
    turnOrder.mode = turnOrder.mode === 'initiative' ? 'initiative' : 'rotation';
    turnOrder.order = Array.isArray(turnOrder.order) ? turnOrder.order : [];
    turnOrder.rolls = turnOrder.rolls && typeof turnOrder.rolls === 'object' ? turnOrder.rolls : {};
    turnOrder.history = Array.isArray(turnOrder.history) ? turnOrder.history.slice(0, 100) : [];
    const validRefs = new Set(dashboardTurnActors().map(actor => actor.ref));
    turnOrder.order = turnOrder.order.filter(ref => validRefs.has(ref));
    turnOrder.activeIndex = clamp(turnOrder.activeIndex, 0, Math.max(0, turnOrder.order.length - 1));
    turnOrder.round = Math.max(1, number(turnOrder.round, 1));
    turnOrder.turn = Math.max(1, number(turnOrder.turn, 1));
    return turnOrder;
  }

  function buildDashboardTurnOrder(forceInitiativeRoll = false) {
    const turnOrder = normalizeDashboardTurnOrder();
    const actors = dashboardTurnActors();
    if (turnOrder.mode === 'initiative') {
      const rolls = {};
      for (const actor of actors) {
        const dice = [d6(), d6()];
        const total = dice[0] + dice[1] + number(actor.agility);
        rolls[actor.ref] = { dice, agility: number(actor.agility), total, name: actor.name };
      }
      turnOrder.rolls = rolls;
      turnOrder.order = actors.sort((left, right) => {
        const leftRoll = rolls[left.ref];
        const rightRoll = rolls[right.ref];
        return rightRoll.total - leftRoll.total || rightRoll.agility - leftRoll.agility || left.name.localeCompare(right.name);
      }).map(actor => actor.ref);
      turnOrder.rolledAt = new Date().toISOString();
    } else {
      turnOrder.order = actors.map(actor => actor.ref);
      if (forceInitiativeRoll) turnOrder.rolls = {};
      turnOrder.rolledAt = '';
    }
    turnOrder.activeIndex = 0;
    turnOrder.round = 1;
    turnOrder.turn = 1;
    turnOrder.encounterId = '';
    stampDashboardEncounterTurn(true);
    save();
    return turnOrder;
  }

  function activeDashboardTurnActor() {
    const turnOrder = normalizeDashboardTurnOrder();
    const actors = dashboardTurnActors();
    const actorByRef = Object.fromEntries(actors.map(actor => [actor.ref, actor]));
    if (!turnOrder.order.length) buildDashboardTurnOrder(false);
    for (let offset = 0; offset < Math.max(1, turnOrder.order.length); offset += 1) {
      const index = (turnOrder.activeIndex + offset) % Math.max(1, turnOrder.order.length);
      const actor = actorByRef[turnOrder.order[index]];
      if (actor) {
        turnOrder.activeIndex = index;
        return actor;
      }
    }
    return actors[0] || null;
  }

  function stampDashboardEncounterTurn(force = false) {
    const slot = typeof nextPendingCrossingSlot === 'function' ? nextPendingCrossingSlot() : null;
    const actor = activeDashboardTurnActor();
    const turnOrder = normalizeDashboardTurnOrder();
    if (!slot || !actor) return null;
    if (force || !slot.assignedCrewRef) {
      slot.assignedCrewRef = actor.ref;
      slot.assignedCrewId = actor.crewId;
      slot.assignedCrewName = actor.name;
      slot.assignedRound = turnOrder.round;
      slot.assignedTurn = turnOrder.turn;
      turnOrder.encounterId = slot.id;
    }
    return slot;
  }

  function setDashboardActiveActor(ref) {
    const turnOrder = normalizeDashboardTurnOrder();
    const index = turnOrder.order.indexOf(ref);
    if (index < 0) return false;
    turnOrder.activeIndex = index;
    turnOrder.encounterId = '';
    stampDashboardEncounterTurn(true);
    save();
    return true;
  }

  function advanceDashboardTurn(reason = '') {
    const turnOrder = normalizeDashboardTurnOrder();
    const livingRefs = dashboardTurnActors().map(actor => actor.ref);
    turnOrder.order = turnOrder.order.filter(ref => livingRefs.includes(ref));
    if (!turnOrder.order.length) {
      turnOrder.activeIndex = 0;
      turnOrder.encounterId = '';
      save();
      return null;
    }
    const previousIndex = turnOrder.activeIndex;
    turnOrder.activeIndex = (turnOrder.activeIndex + 1) % turnOrder.order.length;
    if (turnOrder.activeIndex <= previousIndex) turnOrder.round += 1;
    turnOrder.turn += 1;
    turnOrder.encounterId = '';
    if (reason) turnOrder.history.unshift({ at: new Date().toISOString(), round: turnOrder.round, turn: turnOrder.turn, reason });
    turnOrder.history = turnOrder.history.slice(0, 100);
    stampDashboardEncounterTurn(true);
    save();
    return activeDashboardTurnActor();
  }

  function dashboardSetActiveOperator(actor = activeDashboardTurnActor()) {
    if (!actor) return false;
    const management = normalizeCrewManagement();
    management.activeCrewId = actor.ref;
    return true;
  }

  const generalEncounterRollOperatorBeforeTurns = generalEncounterRollOperator;
  generalEncounterRollOperator = function generalEncounterRollOperatorByTransitTurn() {
    if (normalizeDashboardTurnOrder() && state.operationsDashboard.operationalState === 'in-route') {
      const actor = activeDashboardTurnActor();
      if (actor) {
        dashboardSetActiveOperator(actor);
        if (actor.ref === 'local') return localGeneralEncounterTarget();
        const record = normalizeCrewManagement().roster.find(item => item.rosterId === actor.ref);
        if (record) return { id: record.rosterId, type: 'crew', name: record.characterName || 'Crew Member', current: record.current, status: record.status, record };
      }
    }
    return generalEncounterRollOperatorBeforeTurns();
  };

  const generateValidCrossingBeforeStationAdjacency = generateValidCrossing;
  generateValidCrossing = function generateAdjacentStationCrossing() {
    const context = normalizeActiveSubmarineDashboard(false);
    normalizeDashboardTurnOrder();
    if (state.operationsDashboard.operationalState !== 'at-station') throw new Error('A new route may only be filed while At Station.');
    const map = state.world?.map;
    const origin = crossingNode(context.group.currentNodeId, map);
    if (!origin || origin.type !== 'station') throw new Error('The active submarine must be positioned at a station before filing a route.');
    state.route.startNodeId = origin.id;
    state.route.start = origin.name;
    const corridor = dashboardAdjacentStationRoutes(origin.id, map).find(route => route.stationId === state.route.endNodeId);
    if (!corridor) throw new Error('Destination must be an immediately adjacent station route from the current station.');
    const fullMap = state.world.map;
    const nodeIds = new Set(corridor.pathNodeIds);
    const edgeIds = new Set(corridor.edgeIds);
    state.world.map = { ...fullMap, nodes: fullMap.nodes.filter(node => nodeIds.has(node.id)), edges: fullMap.edges.filter(edge => edgeIds.has(edge.id)), startId: origin.id };
    let crossing;
    try {
      crossing = generateValidCrossingBeforeStationAdjacency();
    } finally {
      state.world.map = fullMap;
    }
    crossing.stationCorridor = { originId: origin.id, destinationId: corridor.stationId, pathNodeIds: [...corridor.pathNodeIds], edgeIds: [...corridor.edgeIds], distance: corridor.distance, danger: corridor.danger };
    state.operationsDashboard.operationalState = 'at-station';
    buildDashboardTurnOrder(false);
    normalizeActiveSubmarineDashboard(true);
    return crossing;
  };

  const startCrossingBeforeTransitTurns = startCrossing;
  startCrossing = function startCrossingWithTransitTurns() {
    const result = startCrossingBeforeTransitTurns();
    if (!result) return result;
    state.operationsDashboard.operationalState = 'in-route';
    buildDashboardTurnOrder(state.operationsDashboard.turnOrder?.mode === 'initiative');
    stampDashboardEncounterTurn(true);
    normalizeActiveSubmarineDashboard(true);
    return result;
  };

  const resolveCrossingEventBeforeTransitTurns = resolveCrossingEvent;
  resolveCrossingEvent = function resolveCrossingEventWithTransitTurns(slotId, status, roll = '') {
    normalizeDashboardTurnOrder();
    if (state.operationsDashboard.operationalState !== 'in-route') {
      state.route.crossingMessage = 'Encounter cards may only be resolved while In Route.';
      save();
      return false;
    }
    const slot = state.route.plan.find(item => item.id === slotId);
    const wasPending = slot?.status === 'pending';
    const actor = activeDashboardTurnActor();
    if (!actor) {
      state.route.crossingMessage = 'No living crew member remains in the turn order.';
      save();
      return false;
    }
    dashboardSetActiveOperator(actor);
    stampDashboardEncounterTurn(false);
    const result = resolveCrossingEventBeforeTransitTurns(slotId, status, roll);
    if (result && wasPending) {
      slot.resolvedByCrewRef = actor.ref;
      slot.resolvedByCrewId = actor.crewId;
      slot.resolvedByCrewName = actor.name;
      slot.resolvedRound = number(state.operationsDashboard.turnOrder.round, 1);
      slot.resolvedTurn = number(state.operationsDashboard.turnOrder.turn, 1);
      advanceDashboardTurn(`${actor.name} resolved ${crossingSelectedEvent(slot)?.name || slot.label} as ${status}.`);
      normalizeActiveSubmarineDashboard(true);
    }
    return result;
  };

  const rollCrossingEventBeforeTransitTurns = rollCrossingEvent;
  rollCrossingEvent = function rollCrossingEventWithTransitTurns(slotId) {
    if (state.operationsDashboard.operationalState !== 'in-route') {
      state.route.crossingMessage = 'Encounter rolls are only available while In Route.';
      save();
      return false;
    }
    const actor = activeDashboardTurnActor();
    if (!actor) return false;
    dashboardSetActiveOperator(actor);
    return rollCrossingEventBeforeTransitTurns(slotId);
  };

  const redrawCrossingEventBeforeTransitTurns = redrawCrossingEvent;
  redrawCrossingEvent = function redrawCrossingEventWithTransitTurns(slotId) {
    if (state.operationsDashboard.operationalState !== 'in-route') {
      state.route.crossingMessage = 'Route event cards may only be redrawn while In Route.';
      save();
      return false;
    }
    return redrawCrossingEventBeforeTransitTurns(slotId);
  };

  const completeCrossingLegBeforeTransitTurns = completeCrossingLeg;
  completeCrossingLeg = function completeCrossingLegWithOperationalState() {
    const result = completeCrossingLegBeforeTransitTurns();
    if (!result) return result;
    state.operationsDashboard.operationalState = normalizeRouteCrossing().status === 'arrived' ? 'at-station' : 'in-route';
    if (state.operationsDashboard.operationalState === 'at-station') {
      state.operationsDashboard.turnOrder.encounterId = '';
      state.operationsDashboard.worldViewerOpen = false;
    }
    normalizeActiveSubmarineDashboard(true);
    return result;
  };

  const abortCrossingBeforeTransitTurns = abortCrossing;
  abortCrossing = function abortCrossingToStationMode() {
    const result = abortCrossingBeforeTransitTurns();
    if (!result) return result;
    state.operationsDashboard.operationalState = 'at-station';
    state.operationsDashboard.turnOrder.encounterId = '';
    normalizeActiveSubmarineDashboard(true);
    return result;
  };

  const clearRouteCrossingBeforeTransitTurns = clearRouteCrossing;
  clearRouteCrossing = function clearRouteCrossingToStationMode() {
    const result = clearRouteCrossingBeforeTransitTurns();
    state.operationsDashboard.operationalState = 'at-station';
    state.operationsDashboard.turnOrder = defaultDashboardTurnOrder();
    normalizeActiveSubmarineDashboard(true);
    return result;
  };
