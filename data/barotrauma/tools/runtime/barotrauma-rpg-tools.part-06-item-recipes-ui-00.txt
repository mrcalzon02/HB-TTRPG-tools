  const itemToolBehaviorBeforeRecipeAudit=itemToolBehavior;
  itemToolBehavior=function itemToolBehaviorWithUsageAudit(itemOrId){
    const id=typeof itemOrId==='string'?itemOrId:itemOrId?.id;
    const base=itemToolBehaviorBeforeRecipeAudit(itemOrId);
    if(REUSABLE_CONDITION_TOOL_IDS.has(id))return {...(base||{}),tracker:'condition',wearPerUse:5,action:base?.action||'Reusable hand tool operation.'};
    return base;
  };
