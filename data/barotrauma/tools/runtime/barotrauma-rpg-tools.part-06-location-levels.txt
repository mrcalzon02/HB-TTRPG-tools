
  const LOCATION_LEVEL_REGISTRY = globalThis.BAROTRAUMA_LOCATION_LEVEL_REGISTRY || { schemaVersion: '0', levels: [], locations: [] };
  const LOCATION_LEVEL_VERSION = LOCATION_LEVEL_REGISTRY.schemaVersion || '1.0.0';
  const LOCATION_LEVEL_MAP = Object.fromEntries((LOCATION_LEVEL_REGISTRY.levels || []).map(item => [number(item.level), item]));
  const LOCATION_TYPE_MAP = Object.fromEntries((LOCATION_LEVEL_REGISTRY.locations || []).map(item => [item.id, item]));
  const LOCATION_LEVEL_COLORS = { 1:'#9fbf8f', 2:'#8eb5a0', 3:'#7caeb5', 4:'#7594bd', 5:'#a98a70', 6:'#8b7f89', 7:'#806b99', 8:'#bd7658', 9:'#7960a8', 10:'#d6aa54' };

  function locationLevelForRing(ring, rings) {
    const safeRing = Math.max(0, number(ring));
    const safeRings = Math.max(1, number(rings, 48));
    if (safeRing === 0) return 10;
    return clamp(1 + Math.floor((safeRings - safeRing) * 9 / safeRings), 1, 9);
  }

  function locationLevelProfile(level) {
    return LOCATION_LEVEL_MAP[clamp(level, 1, 10)] || { level: clamp(level, 1, 10), name: `Level ${level}`, subtitle: '', summary: '', availability: '', shopBias: [], danger: level };
  }

  function locationProfilesForLevel(level, mode = 'all') {
    return (LOCATION_LEVEL_REGISTRY.locations || []).filter(item => number(item.level) === number(level) && (mode === 'station' ? item.stationEligible : mode === 'encounter' ? item.encounterEligible : true));
  }

  function seededLocationChoice(values, rng) {
    if (!values.length) return null;
    const total = values.reduce((sum, item) => sum + Math.max(0.1, number(item.rarity, 1)), 0);
    let roll = rng() * total;
    for (const item of values) {
      roll -= Math.max(0.1, number(item.rarity, 1));
      if (roll <= 0) return item;
    }
    return values[values.length - 1];
  }

  function locationMapNodeType(profile, originalType) {
    if (originalType === 'station' || originalType === 'anomaly') return originalType;
    const id = String(profile?.id || '');
    if (/outpost|vault|reactor|biodome/.test(id)) return 'wreck';
    if (/monolith|resonance|maze/.test(id)) return 'ruins';
    if (/void|depths|expanse|ice-field|trench|cavern|fissure|vent|formation|graveyard/.test(id)) return 'hazard';
    if (/husk|feeding|breeding|lair/.test(id)) return 'hazard';
    return originalType === 'outpost' ? 'outpost' : 'hazard';
  }

  function locationDetailText(node, profile, levelProfile) {
    const stationText = node.type === 'station' ? `${node.parentFaction || node.faction || 'Independent'} — ${node.faction || 'Unassigned'}. ` : '';
    const publicNames = (node.visibleInfluences || []).filter(item => item.visibility === 'public').map(item => item.name);
    const influenceText = publicNames.length ? ` Public influences: ${publicNames.join(', ')}.` : '';
    return `${stationText}Level ${profile.level}: ${levelProfile.name} — ${profile.name}. ${profile.lore} Availability: ${profile.availability}${node.locals ? ` Important locals: ${node.locals}.` : ''}${influenceText}`;
  }

  function applyLocationProfileToNode(node, profile, levelProfile, map) {
    if (!node || !profile) return;
    const originalType = node.type;
    node.locationLevel = number(profile.level);
    node.locationLevelName = levelProfile.name;
    node.locationLevelSubtitle = levelProfile.subtitle || '';
    node.locationTypeId = profile.id;
    node.locationTypeName = profile.name;
    node.locationLore = profile.lore;
    node.locationAvailability = profile.availability;
    node.locationShops = cloneWorldData(profile.shops || []);
    node.locationServices = cloneWorldData(profile.services || []);
    node.locationHazards = cloneWorldData(profile.hazards || []);
    node.locationFactions = cloneWorldData(profile.factions || []);
    node.locationDanger = number(levelProfile.danger, profile.level);
    node.locationLevelColor = LOCATION_LEVEL_COLORS[node.locationLevel] || '#888';
    node.encounterLocation = Boolean(profile.encounterEligible);
    node.stationLocation = Boolean(profile.stationEligible);
    node.availabilityTier = node.locationLevel;
    if (originalType !== 'station' && originalType !== 'anomaly') node.type = locationMapNodeType(profile, originalType);
    if (profile.unique) {
      node.name = profile.name;
      node.type = 'anomaly';
    } else if (originalType !== 'station') {
      const prefix = MAP_PREFIX[Math.abs(seededHash(`${map.seed}|${node.id}|location-name`)) % MAP_PREFIX.length];
      node.name = `${prefix} ${profile.name}`;
    }
    node.details = locationDetailText(node, profile, levelProfile);
  }

  function worldLocationLevelsComplete(map = state.world?.map) {
    return Boolean(map?.nodes?.length && map.locationLevelVersion === LOCATION_LEVEL_VERSION && map.nodes.every(node => node.locationLevel && node.locationTypeId));
  }

  function assignWorldLocationLevels(map = state.world?.map, force = false) {
    if (!map?.nodes) return null;
    if (!force && worldLocationLevelsComplete(map)) return map;
    const salt = number(state.world.locationLevelSeedSalt, 0);
    const rng = seededRandom(`${map.seed}|locations|${LOCATION_LEVEL_VERSION}|${salt}`);
    for (const node of map.nodes) {
      const level = locationLevelForRing(node.ring, map.rings);
      const levelProfile = locationLevelProfile(level);
      let pool;
      if (level === 10) pool = locationProfilesForLevel(10, 'encounter').filter(item => item.unique);
      else if (node.type === 'station') pool = locationProfilesForLevel(level, 'station');
      else {
        const nonStation = locationProfilesForLevel(level, 'encounter').filter(item => !item.stationEligible);
        pool = nonStation.length ? nonStation : locationProfilesForLevel(level, 'encounter');
      }
      const profile = seededLocationChoice(pool, rng) || locationProfilesForLevel(level)[0];
      applyLocationProfileToNode(node, profile, levelProfile, map);
    }
    map.locationLevelVersion = LOCATION_LEVEL_VERSION;
    map.locationLevelSeedSalt = salt;
    state.world.locationLevelVersion = LOCATION_LEVEL_VERSION;
    return map;
  }

  function stationLocationProfile(station) {
    if (!station) return null;
    return LOCATION_TYPE_MAP[station.locationTypeId] || LOCATION_TYPE_MAP[station.derived?.locationTypeId] || locationProfilesForLevel(station.locationLevel || station.derived?.locationLevel || 1, 'station')[0] || null;
  }

  function decorateStandaloneStation(station, entry = null) {
    if (!station) return station;
    const desiredLevel = clamp(entry?.locationLevel ?? station.locationLevel ?? 1, 1, 9);
    const eligible = locationProfilesForLevel(desiredLevel, 'station');
    const profile = eligible.find(item => item.id === (entry?.locationTypeId || station.locationTypeId)) || eligible[0] || locationProfilesForLevel(1, 'station')[0];
    const levelProfile = locationLevelProfile(profile?.level || desiredLevel);
    if (!profile) return station;
    station.locationLevel = profile.level;
    station.locationLevelName = levelProfile.name;
    station.locationLevelSubtitle = levelProfile.subtitle;
    station.locationTypeId = profile.id;
    station.locationTypeName = profile.name;
    station.locationLore = profile.lore;
    station.locationAvailability = profile.availability;
    station.locationShops = cloneWorldData(profile.shops || []);
    station.locationServices = cloneWorldData(profile.services || []);
    station.locationHazards = cloneWorldData(profile.hazards || []);
    station.availabilityTier = profile.level;
    station.specialty = [station.specialty, ...(profile.shops || []), ...(profile.services || [])].filter(Boolean).join(', ');
    return station;
  }

  function selectedWorldLocationNode() {
    const map = state.world?.map;
    if (!map?.nodes) return null;
    const routeNode = map.nodes.find(node => node.name === state.route.end);
    if (routeNode?.locationTypeId) return routeNode;
    const selectedIds = Array.isArray(state.world.selected) ? [...state.world.selected].reverse() : [];
    return selectedIds.map(id => map.nodes.find(node => node.id === id)).find(node => node?.locationTypeId) || null;
  }

  function filteredLocationEncounterProfiles() {
    const planner = state.encounters;
    const selected = planner.useSelectedWorldLocation ? selectedWorldLocationNode() : null;
    if (selected?.locationTypeId && LOCATION_TYPE_MAP[selected.locationTypeId]) return [{ profile: LOCATION_TYPE_MAP[selected.locationTypeId], node: selected }];
    let profiles = (LOCATION_LEVEL_REGISTRY.locations || []).filter(item => item.encounterEligible);
    if (planner.locationLevel !== 'All') profiles = profiles.filter(item => number(item.level) === number(planner.locationLevel));
    if (planner.locationTypeId !== 'All') profiles = profiles.filter(item => item.id === planner.locationTypeId);
    return profiles.map(profile => ({ profile, node: null }));
  }

  function levelRingRange(level, rings) {
    if (number(level) === 10) return 'Core only';
    const matching = Array.from({ length: rings }, (_, index) => index + 1).filter(ring => locationLevelForRing(ring, rings) === number(level));
    if (!matching.length) return 'None';
    return matching.length === 1 ? `Ring ${matching[0]}` : `Rings ${Math.min(...matching)}–${Math.max(...matching)}`;
  }

    const factionPanel = root.querySelector('.faction-world-panel');
    if (factionPanel?.parentElement) factionPanel.parentElement.insertBefore(panel, factionPanel);
    else root.querySelector('.tool-grid')?.appendChild(panel);
    panel.querySelector('#reseed-location-types')?.addEventListener('click', () => {
      state.world.locationLevelSeedSalt = number(state.world.locationLevelSeedSalt) + 1;
      assignWorldLocationLevels(state.world.map, true);
      seedWorldFactions(state.world.map, false);
      syncWorldMapStations();
      normalizeStationCommerce();
      save(); renderWorld();
    });
    for (const element of root.querySelectorAll('.map-node')) {
      const node = map.nodes.find(item => item.id === element.dataset.nodeId);
      if (!node) continue;
      element.dataset.locationLevel = node.locationLevel;
      element.dataset.locationType = node.locationTypeId;
      const title = element.querySelector('title') || document.createElementNS('http://www.w3.org/2000/svg', 'title');
      title.textContent = `Level ${node.locationLevel}: ${node.locationLevelName} — ${node.locationTypeName}. ${node.name}. Availability: ${node.locationAvailability || 'None'}`;
      if (!title.parentElement) element.prepend(title);
    }
  }
