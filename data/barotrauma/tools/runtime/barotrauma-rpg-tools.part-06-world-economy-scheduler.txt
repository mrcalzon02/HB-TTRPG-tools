
  function scheduleWorldEconomyProgressiveSimulation(now=new Date(),reason='background catch-up') {
    if (globalThis.__BAROTRAUMA_WORLD_ECONOMY_SIMULATION_RUNNING) return false;
    const economy=normalizeWorldEconomyState();
    syncAllWorldPlayerVessels();
    seedWorldEconomyNpcFleet();
    const stationList=Object.values(economy.stationEconomies).filter(station=>!station.archived);
    const last=Date.parse(economy.simulation.lastSimulatedAt||now.toISOString());
    const baseTime=Number.isFinite(last)?last:now.getTime();
    const elapsed=Math.max(0,now.getTime()-baseTime);
    const cycles=Math.min(WORLD_ECONOMY_DEFAULTS.catchupCycleLimit,Math.floor(elapsed/(WORLD_ECONOMY_DEFAULTS.stationCycleMinutes*60000)));
    const vessels=Object.values(economy.vessels).filter(vessel=>vessel.kind==='npc'&&vessel.active!==false);
    let cycleIndex=0;
    let stationIndex=0;
    let vesselIndex=0;
    let processedStations=0;
    let departedTrades=0;
    let completedTrades=0;
    globalThis.__BAROTRAUMA_WORLD_ECONOMY_SIMULATION_RUNNING=true;

    const finish=()=>{
      economy.simulation.lastSimulatedAt=now.toISOString();
      economy.simulation.tickSequence=Math.max(0,number(economy.simulation.tickSequence))+1;
      economy.simulation.lastSummary=`${reason}: processed ${cycles} station cycle${cycles===1?'':'s'}, ${departedTrades} freight departures, and ${completedTrades} completed trades.`;
      syncAllWorldPlayerVessels();
      publishWorldEconomyDiagnostics();
      normalizeWorldHub().updatedAt=now.toISOString();
      save();
      globalThis.__BAROTRAUMA_WORLD_ECONOMY_SIMULATION_RUNNING=false;
      globalThis.dispatchEvent?.(new CustomEvent('barotrauma-world-economy-updated',{detail:{cycles,processedStations,departedTrades,completedTrades}}));
    };

    const processVessels=()=>{
      const batch=vessels.slice(vesselIndex,vesselIndex+12);
      for (const vessel of batch) {
        const due=Date.parse(vessel.nextActionAt||0);
        if (!Number.isFinite(due)||due>now.getTime()) continue;
        if (vessel.status==='in-transit') {
          if (arriveWorldEconomyFreighter(vessel,new Date(due))) completedTrades+=1;
        } else if (vessel.status==='docked') {
          if (departWorldEconomyFreighter(vessel,new Date(Math.max(due,Date.parse(vessel.lastActionAt||0)||now.getTime())))) departedTrades+=1;
        }
      }
      vesselIndex+=12;
      if (vesselIndex<vessels.length) setTimeout(processVessels,0);
      else finish();
    };

    const processStations=()=>{
      if (!cycles||!stationList.length) return processVessels();
      let processed=0;
      while (cycleIndex<cycles&&processed<40) {
        const station=stationList[stationIndex];
        const cycleAt=new Date(baseTime+(cycleIndex+1)*WORLD_ECONOMY_DEFAULTS.stationCycleMinutes*60000).toISOString();
        processWorldStationEconomyCycle(station,cycleAt);
        stationIndex+=1;
        processed+=1;
        processedStations+=1;
        if (stationIndex>=stationList.length) { stationIndex=0; cycleIndex+=1; }
      }
      if (cycleIndex<cycles) setTimeout(processStations,0);
      else processVessels();
    };

    setTimeout(processStations,0);
    return true;
  }

  function ensureWorldEconomyBackgroundTimer() {
    if (!globalThis.__BAROTRAUMA_WORLD_ECONOMY_TIMER) {
      globalThis.__BAROTRAUMA_WORLD_ECONOMY_TIMER=setInterval(()=>scheduleWorldEconomyProgressiveSimulation(new Date(),'scheduled minute check'),60000);
    }
    setTimeout(()=>scheduleWorldEconomyProgressiveSimulation(new Date(),'initial world-economy catch-up'),0);
  }

  ensureWorldEconomyBackgroundTimer();
