  function masterWorldSiteNameHash(value) {
    let hash = 2166136261;
    const text = String(value ?? '');
    for (let index = 0; index < text.length; index += 1) {
      hash ^= text.charCodeAt(index);
      hash = Math.imul(hash, 16777619);
    }
    return hash >>> 0;
  }

  function masterWorldSiteNameCandidate(seed, attempt = 0) {
    const nautical = MASTER_WORLD_SITE_NAUTICAL_WORDS[
      masterWorldSiteNameHash(`${seed}|nautical|${attempt}`) % MASTER_WORLD_SITE_NAUTICAL_WORDS.length
    ];
    const designation = MASTER_WORLD_SITE_DESIGNATIONS[
      masterWorldSiteNameHash(`${seed}|designation|${attempt}`) % MASTER_WORLD_SITE_DESIGNATIONS.length
    ];
    const city = MASTER_WORLD_SITE_CITIES[
      masterWorldSiteNameHash(`${seed}|city|${attempt}`) % MASTER_WORLD_SITE_CITIES.length
    ];
    return `${nautical} ${designation} - ${city}`;
  }

  function masterWorldSiteNameSeed(node, index) {
    if (node?.id) return String(node.id);
    const ring = Number.isFinite(Number(node?.ring)) ? Number(node.ring) : 0;
    const x = Number.isFinite(Number(node?.x)) ? Number(node.x).toFixed(3) : '0';
    const y = Number.isFinite(Number(node?.y)) ? Number(node.y).toFixed(3) : '0';
    return `ring-${ring}|x-${x}|y-${y}|index-${index}`;
  }

  function masterWorldReplaceSiteReferences(value, previousName, nextName) {
    if (typeof value === 'string') return value.includes(previousName) ? value.split(previousName).join(nextName) : value;
    if (Array.isArray(value)) {
      for (let index = 0; index < value.length; index += 1) {
        value[index] = masterWorldReplaceSiteReferences(value[index], previousName, nextName);
      }
      return value;
    }
    if (value && typeof value === 'object') {
      Object.keys(value).forEach(key => {
        value[key] = masterWorldReplaceSiteReferences(value[key], previousName, nextName);
      });
    }
    return value;
  }

  function masterWorldResolveSiteMap(target) {
    if (Array.isArray(target?.nodes)) return target;
    if (Array.isArray(target?.map?.nodes)) return target.map;
    if (Array.isArray(target?.world?.map?.nodes)) return target.world.map;
    if (Array.isArray(target?.state?.world?.map?.nodes)) return target.state.world.map;
    return null;
  }

  function masterWorldNormalizeGenericSiteNames(target) {
    const map = masterWorldResolveSiteMap(target);
    const nodes = Array.isArray(map?.nodes) ? map.nodes : [];
    if (!nodes.length) return 0;

    const usedNames = new Set(
      nodes
        .map(node => String(node?.name || '').trim())
        .filter(name => name && !MASTER_WORLD_GENERIC_SITE_PATTERN.test(name))
    );
    let renamedCount = 0;

    nodes.forEach((node, index) => {
      const previousName = String(node?.name || '').trim();
      if (!MASTER_WORLD_GENERIC_SITE_PATTERN.test(previousName)) return;
      const seed = masterWorldSiteNameSeed(node, index);
      let attempt = 0;
      let nextName = masterWorldSiteNameCandidate(seed, attempt);
      while (usedNames.has(nextName) && attempt < 10000) {
        attempt += 1;
        nextName = masterWorldSiteNameCandidate(seed, attempt);
      }
      if (usedNames.has(nextName)) throw new Error(`Unable to allocate a unique Master World site name for ${previousName}.`);
      masterWorldReplaceSiteReferences(node, previousName, nextName);
      usedNames.add(nextName);
      renamedCount += 1;
    });

    return renamedCount;
  }

  function masterWorldBootstrapCurrentReport() {
    const map = state.world?.map;
    const nodeCount = Array.isArray(map?.nodes) ? map.nodes.length : 0;
    const ringCount = Math.max(0, number(map?.rings, Math.max(0, ...(map?.nodes || []).map(node => number(node.ring, 0)))));
    const validDepth = typeof worldDepthValidateMap === 'function' ? worldDepthValidateMap(map) : { valid: false };
    return { map, nodeCount, ringCount, valid: Boolean(validDepth.valid) };
  }

  function masterWorldBootstrapShouldImport() {
    const report = masterWorldBootstrapCurrentReport();
    return !report.valid || report.ringCount < MASTER_WORLD_BOOTSTRAP_MINIMUM_RINGS || report.nodeCount < MASTER_WORLD_BOOTSTRAP_MINIMUM_LOCATIONS;
  }

  function masterWorldBootstrapImport() {
    const payload = globalThis.BAROTRAUMA_MASTER_WORLD_SAVE;
    if (!payload || payload.schema !== WORLD_SAVE_SCHEMA) return false;

    const normalizedPayload = cloneWorldData(payload);
    const payloadRenamedCount = masterWorldNormalizeGenericSiteNames(normalizedPayload);
    globalThis.BAROTRAUMA_MASTER_WORLD_SAVE = normalizedPayload;

    if (!masterWorldBootstrapShouldImport()) {
      const liveRenamedCount = masterWorldNormalizeGenericSiteNames(state.world?.map);
      if (liveRenamedCount > 0) save();
      globalThis.BAROTRAUMA_MASTER_WORLD_SITE_NAMING = {
        version: MASTER_WORLD_SITE_NAMING_VERSION,
        combinations: MASTER_WORLD_SITE_NAUTICAL_WORDS.length * MASTER_WORLD_SITE_DESIGNATIONS.length * MASTER_WORLD_SITE_CITIES.length,
        payloadRenamedCount,
        liveRenamedCount
      };
      return false;
    }

    const previousSavedStates = cloneWorldData(normalizeWorldHub().savedStates || []);
    const previousMasterWorldId = state.world?.masterWorldId || state.worldHub?.masterWorldId || '';
    importWorldSave(normalizedPayload);
    if (previousSavedStates.length) {
      state.worldHub.savedStates = [...previousSavedStates, ...(state.worldHub.savedStates || [])].slice(0, 30);
    }
    if (previousMasterWorldId && !state.world.masterWorldId) state.world.masterWorldId = previousMasterWorldId;
    if (typeof synchronizeMasterWorldId === 'function') synchronizeMasterWorldId(false);
    const liveRenamedCount = masterWorldNormalizeGenericSiteNames(state.world?.map);
    if (state.world?.map) {
      if (typeof reconcileWorldStationLevels === 'function') reconcileWorldStationLevels(state.world.map);
      if (typeof seedWorldFactions === 'function') seedWorldFactions(state.world.map, false);
      if (typeof assignWorldLocationLevels === 'function') assignWorldLocationLevels(state.world.map, false);
      if (typeof revalidateLocationGroupStarts === 'function') revalidateLocationGroupStarts();
      if (typeof syncWorldMapStations === 'function') syncWorldMapStations();
    }
    if (typeof normalizeWorldEconomyState === 'function') normalizeWorldEconomyState();
    if (typeof syncAllWorldPlayerVessels === 'function') syncAllWorldPlayerVessels();
    if (typeof seedWorldEconomyNpcFleet === 'function') seedWorldEconomyNpcFleet();
    if (typeof normalizeWorldEconomySimulationClocks === 'function') normalizeWorldEconomySimulationClocks();
    if (typeof normalizeActiveSubmarineDashboard === 'function') normalizeActiveSubmarineDashboard(false);
    if (typeof normalizeDashboardTurnOrder === 'function') normalizeDashboardTurnOrder();
    if (typeof normalizeDashboardStartingLocationState === 'function') normalizeDashboardStartingLocationState();
    globalThis.__BAROTRAUMA_DASHBOARD_GRAPH_CACHE = null;
    globalThis.__BAROTRAUMA_STATION_CORRIDOR_CACHE = null;
    globalThis.__BAROTRAUMA_LOCAL_STATION_TOPOLOGY_CACHE = null;
    globalThis.__BAROTRAUMA_LOCAL_WORLD_TOPOLOGY_CACHE = null;
