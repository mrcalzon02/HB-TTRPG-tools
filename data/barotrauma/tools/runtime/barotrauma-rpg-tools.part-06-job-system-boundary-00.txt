  function persistentJobSystemBoundary(){
    const dashboard=typeof activeSubmarineDashboardContext==='function'&&activeSubmarineDashboardContext();
    const authorized=!dashboard?false:(typeof dashboardManagementAuthorized!=='function'||dashboardManagementAuthorized());
    let context=null;
    let actors=[];
    if(dashboard&&authorized){
      try{
        context=normalizeActiveSubmarineDashboard(false);
        actors=typeof dashboardTurnActors==='function'?dashboardTurnActors().filter(actor=>actor&&actor.status!=='Dead'):[];
      }catch(error){
        console.error('Job System dashboard boundary evaluation failed.',error);
      }
    }
    const expeditionActive=state.submarine?.managedExpedition?.status!=='lost';
    const allowed=Boolean(dashboard&&authorized&&context?.submarineId&&actors.length&&expeditionActive);
    const message=!dashboard
      ?'Persistent Jobs, Events, and Stations are available only inside the Active Submarine Dashboard.'
      :!authorized
        ?'Log in to the Active Submarine Dashboard before accessing persistent Jobs, Events, and Stations.'
        :!context?.submarineId
          ?'The dashboard has no active submarine attached to the Masterworld.'
          :!actors.length
            ?'At least one living active crew member is required before persistent Jobs or Events can be used.'
            :!expeditionActive
              ?'The active expedition is lost or inactive and cannot accept or resolve persistent Jobs.'
              :'Persistent Job System ready.';
    return {allowed,dashboard,authorized,context,actors,message};
  }

  function requirePersistentJobSystem(){
    const boundary=persistentJobSystemBoundary();
    return boundary.allowed?{ok:true,boundary}:{ok:false,boundary,message:boundary.message};
  }
