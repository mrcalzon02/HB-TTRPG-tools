  function calculateSubmarine() {
    const submarine = state.submarine;
    const base = SUBMARINES[submarine.model] || SUBMARINES.Barsuk;
    const totals = {
      speed: base.speed + number(submarine.manual.speed),
      maneuverability: base.maneuverability + number(submarine.manual.maneuverability),
      defense: base.defense + number(submarine.manual.defense),
      capacity: base.capacity + number(submarine.manual.capacity),
      hull: base.hull + number(submarine.manual.hull)
    };
    for (const id of submarine.upgrades) {
      const upgrade = UPGRADES.find(item => item.id === id);
      if (!upgrade) continue;
      for (const [stat, modifier] of Object.entries(upgrade.effects)) totals[stat] = number(totals[stat]) + number(modifier);
    }
    const weaponCapacity = submarine.weapons.reduce((sum, mount) => sum + (WEAPONS.find(item => item.name === mount.name)?.storage || 0), 0);
    const usedCapacity = number(submarine.resources.crew) + number(submarine.resources.cargo) + Math.ceil(Math.max(0, number(submarine.resources.supplies)) / 10) + weaponCapacity;
    return { base, totals, weaponCapacity, usedCapacity, remainingCapacity: totals.capacity - usedCapacity };
  }
