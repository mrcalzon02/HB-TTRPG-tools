
  const normalizeActiveSubmarineDashboardBeforeAuthenticatedGroupBinding=normalizeActiveSubmarineDashboard;
  normalizeActiveSubmarineDashboard=function normalizeActiveSubmarineDashboardWithAuthenticatedGroupBinding(persist=false){
    if(dashboardManagementRuntime.status==='authenticated'&&dashboardManagementRuntime.vesselId){
      const hub=normalizeWorldHub();
      const group=dashboardManagementGroupForVessel(dashboardManagementRuntime.vesselId);
      if(!group)throw new Error('The authenticated submarine is no longer assigned to an expedition group.');
      hub.activeGroupId=group.groupId;
      hub.activeSubmarineId=dashboardManagementRuntime.vesselId;
      state.submarine.worldSubmarineId=dashboardManagementRuntime.vesselId;
      state.world.activeSubmarineId=dashboardManagementRuntime.vesselId;
      state.route.activeSubmarineId=dashboardManagementRuntime.vesselId;
      state.route.activeGroupId=group.groupId;
      state.operationsDashboard=state.operationsDashboard&&typeof state.operationsDashboard==='object'?state.operationsDashboard:defaultActiveSubmarineDashboard();
      state.operationsDashboard.activeSubmarineId=dashboardManagementRuntime.vesselId;
      state.operationsDashboard.activeGroupId=group.groupId;
    }
    return normalizeActiveSubmarineDashboardBeforeAuthenticatedGroupBinding(persist);
  };

  const DASHBOARD_MANAGEMENT_RECOVERY_PARTS=[
    'data/barotrauma/tools/runtime/barotrauma-rpg-tools.part-06-dashboard-management-auth-recovery-core.txt',
    'data/barotrauma/tools/runtime/barotrauma-rpg-tools.part-06-dashboard-management-auth-recovery-stability.txt',
    'data/barotrauma/tools/runtime/barotrauma-rpg-tools.part-06-dashboard-management-auth-recovery-ui-00.txt',
    'data/barotrauma/tools/runtime/barotrauma-rpg-tools.part-06-dashboard-management-auth-recovery-ui-01.txt',
    'data/barotrauma/tools/runtime/barotrauma-rpg-tools.part-06-dashboard-management-auth-recovery-ui-02.txt'
  ];

  async function loadDashboardManagementRecoveryModules(){
    try{
      const responses=await Promise.all(DASHBOARD_MANAGEMENT_RECOVERY_PARTS.map(path=>fetch(path,{cache:'no-store'})));
      const failed=responses.find(response=>!response.ok);
      if(failed)throw new Error(`${failed.url} returned ${failed.status}`);
      const source=(await Promise.all(responses.map(response=>response.text()))).join('');
      eval(source);
      globalThis.BAROTRAUMA_DASHBOARD_MANAGEMENT_RECOVERY_READY=true;
      if(activeSubmarineDashboardContext())render();
      return true;
    }catch(error){
      globalThis.BAROTRAUMA_DASHBOARD_MANAGEMENT_RECOVERY_READY=false;
      globalThis.BAROTRAUMA_DASHBOARD_MANAGEMENT_RECOVERY_ERROR=error.message||String(error);
      console.error('Dashboard management recovery modules failed to load.',error);
      return false;
    }
  }

  globalThis.BAROTRAUMA_DASHBOARD_MANAGEMENT_RECOVERY_READY=false;
  globalThis.BAROTRAUMA_DASHBOARD_MANAGEMENT_RECOVERY_PROMISE=loadDashboardManagementRecoveryModules();
