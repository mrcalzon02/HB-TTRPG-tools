
  const normalizeActiveSubmarineDashboardBeforeAuthenticatedGroupBinding=normalizeActiveSubmarineDashboard;
  normalizeActiveSubmarineDashboard=function normalizeActiveSubmarineDashboardWithAuthenticatedGroupBinding(persist=false){
    if(dashboardManagementRuntime.status==='authenticated'&&dashboardManagementRuntime.vesselId){
      const hub=normalizeWorldHub();
      const group=dashboardManagementGroupForVessel(dashboardManagementRuntime.vesselId);
      if(!group)throw new Error('The authenticated submarine is no longer assigned to an expedition group.');
      hub.activeGroupId=group.groupId;
      hub.activeSubmarineId=dashboardManagementRuntime.vesselId;
      state.submarine.worldSubmarineId=dashboardManagementRuntime.vesselId;
      state.world.activeSubmarineId=dashboardManagementRuntime.vesselId;
      state.route.activeSubmarineId=dashboardManagementRuntime.vesselId;
      state.route.activeGroupId=group.groupId;
      state.operationsDashboard=state.operationsDashboard&&typeof state.operationsDashboard==='object'?state.operationsDashboard:defaultActiveSubmarineDashboard();
      state.operationsDashboard.activeSubmarineId=dashboardManagementRuntime.vesselId;
      state.operationsDashboard.activeGroupId=group.groupId;
    }
    return normalizeActiveSubmarineDashboardBeforeAuthenticatedGroupBinding(persist);
  };

  function dashboardManagementHydrateSnapshotBeforeAnyLogin(vesselId){
    const authority=normalizeDashboardManagementAuthority();
    const snapshot=authority.vesselSnapshots?.[vesselId];
    if(!snapshot||typeof snapshot!=='object')return null;
    const defaults=defaultState();
    snapshot.character=deepMerge(defaults.character,cloneWorldData(snapshot.character||{}));
    snapshot.submarine=deepMerge(defaults.submarine,cloneWorldData(snapshot.submarine||{}));
    snapshot.route=deepMerge(defaults.route,cloneWorldData(snapshot.route||{}));
    snapshot.operationsDashboard=deepMerge(defaultActiveSubmarineDashboard(),cloneWorldData(snapshot.operationsDashboard||{}));

    snapshot.character.rollLog=Array.isArray(snapshot.character.rollLog)?snapshot.character.rollLog:[];
    snapshot.character.catalogInventory=Array.isArray(snapshot.character.catalogInventory)?snapshot.character.catalogInventory:[];
    snapshot.character.consumptionLog=Array.isArray(snapshot.character.consumptionLog)?snapshot.character.consumptionLog:[];
    snapshot.character.inventory=String(snapshot.character.inventory||'');

    snapshot.submarine.upgrades=Array.isArray(snapshot.submarine.upgrades)?snapshot.submarine.upgrades:[];
    snapshot.submarine.weapons=Array.isArray(snapshot.submarine.weapons)?snapshot.submarine.weapons:[];
    snapshot.submarine.wikiWeapons=Array.isArray(snapshot.submarine.wikiWeapons)?snapshot.submarine.wikiWeapons:[];

    snapshot.submarine.crewManagement=snapshot.submarine.crewManagement&&typeof snapshot.submarine.crewManagement==='object'?snapshot.submarine.crewManagement:{roster:[],activeCrewId:'local'};
    snapshot.submarine.crewManagement.roster=Array.isArray(snapshot.submarine.crewManagement.roster)?snapshot.submarine.crewManagement.roster:[];

    snapshot.route.jobs=Array.isArray(snapshot.route.jobs)?snapshot.route.jobs:[];

    snapshot.route.crossing.log=Array.isArray(snapshot.route.crossing.log)?snapshot.route.crossing.log:[];

    snapshot.operationsDashboard.activeCrewIds=Array.isArray(snapshot.operationsDashboard.activeCrewIds)?snapshot.operationsDashboard.activeCrewIds:[];
    snapshot.operationsDashboard.commissioning=snapshot.operationsDashboard.commissioning&&typeof snapshot.operationsDashboard.commissioning==='object'?snapshot.operationsDashboard.commissioning:{assignedCrewIds:[]};
    snapshot.operationsDashboard.commissioning.assignedCrewIds=Array.isArray(snapshot.operationsDashboard.commissioning.assignedCrewIds)?snapshot.operationsDashboard.commissioning.assignedCrewIds:[];
    snapshot.operationsDashboard.turnOrder=snapshot.operationsDashboard.turnOrder&&typeof snapshot.operationsDashboard.turnOrder==='object'?snapshot.operationsDashboard.turnOrder:{order:[],rolls:{},history:[]};

  async function loadDashboardManagementRecoveryModules(){
    try{
      const responses=await Promise.all(DASHBOARD_MANAGEMENT_RECOVERY_PARTS.map(path=>fetch(path,{cache:'no-store'})));
      const failed=responses.find(response=>!response.ok);
      if(failed)throw new Error(`${failed.url} returned ${failed.status}`);
      const source=(await Promise.all(responses.map(response=>response.text()))).join('');
      eval(source);
      globalThis.BAROTRAUMA_DASHBOARD_MANAGEMENT_RECOVERY_READY=true;
      if(activeSubmarineDashboardContext())render();
      return true;
    }catch(error){
      globalThis.BAROTRAUMA_DASHBOARD_MANAGEMENT_RECOVERY_READY=false;
      globalThis.BAROTRAUMA_DASHBOARD_MANAGEMENT_RECOVERY_ERROR=error.message||String(error);
      console.error('Dashboard management recovery modules failed to load.',error);
      return false;
    }
  }

  globalThis.BAROTRAUMA_DASHBOARD_MANAGEMENT_RECOVERY_READY=false;
  globalThis.BAROTRAUMA_DASHBOARD_MANAGEMENT_RECOVERY_PROMISE=loadDashboardManagementRecoveryModules();
