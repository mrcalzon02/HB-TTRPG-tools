    const group = currentManagedGroup();
    const submission = currentWorldSubmissionBeforeManagedOwnershipStability(kind);
    if (!group) return submission;
    submission.groupId = group.groupId;
    if (submission.payload && typeof submission.payload === 'object') submission.payload.groupId = group.groupId;
    if (kind === 'location') {
      submission.payload.nodeId = group.currentNodeId;
      submission.payload.node = crossingNode(group.currentNodeId) || null;
    }
    if (kind === 'submarine') submission.sourceId = managedSubmarineIdentity();

  };
