    const previousCrewIds = Array.isArray(beforeGroup?.crewIds) ? [...beforeGroup.crewIds] : [];
    const previousSubmarineIds = Array.isArray(beforeGroup?.submarineIds) ? [...beforeGroup.submarineIds] : [];
    const expedition = normalizeManagedExpeditionBeforeRegistryMerge();
    const group = expedition ? hub.publicState.groups[expedition.groupId] : null;
