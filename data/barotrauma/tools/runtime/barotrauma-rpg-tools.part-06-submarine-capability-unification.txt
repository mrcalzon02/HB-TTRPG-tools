  function submarineWeaponSlotType(systemId){
    const weapon=catalogueWeapon(systemId);
    if(!weapon)return 'unknown';
    if(weapon.mount==='Turret')return 'small';
    if(weapon.mount==='Large Turret')return 'large';
    if(systemId==='depth-charge-tube')return 'depth-charge';
    if(systemId==='electrical-discharge-coil')return 'discharge-coil';
    return 'utility';
  }

  function submarineWeaponSlotLabel(type){
    return type==='small'?'Small turret':type==='large'?'Large turret':type==='depth-charge'?'Depth-charge tube':type==='discharge-coil'?'Electrical discharge coil':type==='drone'?'Auxiliary drone weapon':'Utility installation';
  }

  function submarineWeaponSlotCompatible(slot,systemId){
    if(!slot||!systemId)return false;
    const type=submarineWeaponSlotType(systemId);
    if(slot.slotType==='small')return type==='small';
    if(slot.slotType==='large')return type==='large';
    if(slot.slotType==='depth-charge')return systemId==='depth-charge-tube';
    if(slot.slotType==='discharge-coil')return systemId==='electrical-discharge-coil';
    if(slot.slotType==='drone')return systemId==='coilgun';
    return type==='utility';
  }

  function submarineWeaponSlotBlueprint(profile=submarineCapabilityProfile()){
    const slots=[];
    const counters={small:0,large:0,'depth-charge':0,'discharge-coil':0,utility:0,drone:0};
    for(const systemId of genericSystemsForProfile(profile)){
      const slotType=submarineWeaponSlotType(systemId);
      counters[slotType]=(counters[slotType]||0)+1;
      slots.push({
        slotId:`${slotType}-fixed-${counters[slotType]}`,
        slotType,
        hardpoint:false,
        locked:['depth-charge','discharge-coil','utility'].includes(slotType),
        installedId:systemId,
        source:'official-installed'
      });
    }
    for(let index=0;index<Math.max(0,number(profile.smallHardpoints));index+=1){
      counters.small+=1;
      slots.push({slotId:`small-hardpoint-${index+1}`,slotType:'small',hardpoint:true,locked:false,installedId:'',source:'official-hardpoint'});
    }
    for(let index=0;index<Math.max(0,number(profile.largeHardpoints));index+=1){
      counters.large+=1;
      slots.push({slotId:`large-hardpoint-${index+1}`,slotType:'large',hardpoint:true,locked:false,installedId:'',source:'official-hardpoint'});
    }
    if(profile.drone){
      slots.push({slotId:'drone-fixed-1',slotType:'drone',hardpoint:false,locked:true,installedId:'coilgun',source:'official-drone'});
    }
    return slots;
  }

  function submarineWeaponSlotSignature(profile=submarineCapabilityProfile()){
    return JSON.stringify({model:state.submarine.model,defaults:genericSystemsForProfile(profile),smallHardpoints:number(profile.smallHardpoints),largeHardpoints:number(profile.largeHardpoints),drone:Boolean(profile.drone)});
  }

  function syncWikiWeaponsFromWeaponSlots(){
    const previous=new Map((state.submarine.wikiWeapons||[]).map(entry=>[entry.instanceId,entry]));
    state.submarine.wikiWeapons=(state.submarine.weaponSlots||[]).filter(slot=>slot.installedId).map(slot=>{
      const weapon=catalogueWeapon(slot.installedId);
      const prior=previous.get(slot.slotId)||{};
      const ammoNames=(weapon?.ammo||[]).map(item=>item.name);
      const ammoName=ammoNames.includes(prior.ammoName)?prior.ammoName:(weapon?.ammo?.[0]?.name||'');
      const ammo=weapon?.ammo?.find(item=>item.name===ammoName);
      return {instanceId:slot.slotId,id:slot.installedId,ammoName,loaded:Math.max(0,number(prior.loaded,ammo?.magazineSize||0)),slotType:slot.slotType,slotSource:slot.source};
    });
    state.submarine.weapons=[];
    return state.submarine.wikiWeapons;
  }

  function normalizeSubmarineWeaponSlots(force=false){
    const profile=submarineCapabilityProfile();
    const signature=submarineWeaponSlotSignature(profile);
    if(force||!Array.isArray(state.submarine.weaponSlots)||state.submarine.weaponSlotSignature!==signature){
      state.submarine.weaponSlots=submarineWeaponSlotBlueprint(profile);
      state.submarine.weaponSlotSignature=signature;
      state.submarine.weaponInstallMessage='Weapon installations reset to the selected submarine’s official wiki layout.';
    }
    const blueprint=submarineWeaponSlotBlueprint(profile);
    const blueprintById=new Map(blueprint.map(slot=>[slot.slotId,slot]));
    state.submarine.weaponSlots=blueprint.map(expected=>{
      const existing=(state.submarine.weaponSlots||[]).find(slot=>slot.slotId===expected.slotId)||{};
      const installedId=existing.installedId&&submarineWeaponSlotCompatible(expected,existing.installedId)?existing.installedId:expected.installedId;
      return {...expected,installedId};
    });
    state.submarine.weaponSlots=state.submarine.weaponSlots.filter(slot=>blueprintById.has(slot.slotId));
    syncWikiWeaponsFromWeaponSlots();
    return state.submarine.weaponSlots;
  }

  function installSubmarineWeaponInSlot(slotId,systemId){
    const slots=normalizeSubmarineWeaponSlots();
    const slot=slots.find(item=>item.slotId===slotId);
    if(!slot||slot.locked)return {ok:false,message:'That installation is fixed to the official submarine layout.'};
    if(systemId&&!submarineWeaponSlotCompatible(slot,systemId))return {ok:false,message:`${catalogueWeapon(systemId)?.name||systemId} is not compatible with a ${submarineWeaponSlotLabel(slot.slotType).toLowerCase()} mount.`};
    slot.installedId=systemId||'';
    syncWikiWeaponsFromWeaponSlots();
    state.submarine.weaponInstallMessage=slot.installedId?`${catalogueWeapon(slot.installedId)?.name||slot.installedId} installed in ${submarineWeaponSlotLabel(slot.slotType).toLowerCase()} ${slot.slotId}.`:`${submarineWeaponSlotLabel(slot.slotType)} ${slot.slotId} left empty.`;
    save();
    return {ok:true,message:state.submarine.weaponInstallMessage};
  }

    if(typeof designDeployable==='function'){
      const custom=typeof customEntry==='function'?customEntry(id):null;
      if(custom&&!designDeployable('custom-content',id))return false;
    }
    const weapon=catalogueWeapon(id);
    if(!weapon)return false;
    const slots=normalizeSubmarineWeaponSlots();
    const slot=slots.find(item=>!item.locked&&!item.installedId&&submarineWeaponSlotCompatible(item,id));
    if(!slot){
      state.submarine.weaponInstallMessage=`No empty compatible ${weapon.mount||'weapon'} hardpoint is available on ${state.submarine.model}.`;
      return false;
    }
    return installSubmarineWeaponInSlot(slot.slotId,id).ok;
  };

  const normalizeCargoManagementBeforeCapabilityUnification=normalizeCargoManagement;
  normalizeCargoManagement=function normalizeUnifiedCargoManagement(){
    const cargo=normalizeCargoManagementBeforeCapabilityUnification();
    cargo.entries=(cargo.entries||[]).map(normalizeCargoEntry);
    cargo.autoSync=true;
    cargo.lastMessage=String(cargo.lastMessage||'');
    let migrationCrate=1;
    for(const entry of cargo.entries){
      if(entry.storageMode==='economic-crate'&&!entry.crateId){entry.crateId=`crate-${migrationCrate}`;migrationCrate+=1;}
    }
    const report=unifiedCargoCapacityReport(cargo);
    state.submarine.resources.cargo=report.economicCratesUsed;
    return cargo;
  };

  function unifiedCargoCrates(cargo=state.submarine.cargoManagement){
    const groups={};
    for(const entry of cargo?.entries||[]){
      if(entry.storageMode!=='economic-crate'||!entry.crateId)continue;
      groups[entry.crateId]=groups[entry.crateId]||[];
      groups[entry.crateId].push(entry);
    }
    return groups;
  }

  function unifiedCargoCapacityReport(cargo=state.submarine.cargoManagement){
    cargo=cargo&&Array.isArray(cargo.entries)?cargo:{entries:[]};
    const crates=unifiedCargoCrates(cargo);
    const crateLoads=Object.fromEntries(Object.entries(crates).map(([crateId,entries])=>[crateId,entries.reduce((sum,entry)=>sum+Math.max(0,number(entry.quantity)),0)]));
    const economicItems=Object.values(crateLoads).reduce((sum,value)=>sum+value,0);
    const looseItemsUsed=cargo.entries.filter(entry=>entry.storageMode!=='economic-crate').reduce((sum,entry)=>sum+Math.max(0,number(entry.quantity)),0);
    const ratedCrates=submarineCargoCrateCapacity();
    return {
      ratedCrates,
      economicCratesUsed:Object.keys(crateLoads).length,
      economicItems,
      economicItemCapacity:ratedCrates*SUBMARINE_STANDARD_CRATE_ITEM_LIMIT,
      looseItemsUsed,
      looseItemCapacity:SUBMARINE_LOOSE_ITEM_LIMIT,
      crateLoads,
      overloadedCrates:Object.entries(crateLoads).filter(([,load])=>load>SUBMARINE_STANDARD_CRATE_ITEM_LIMIT).map(([id])=>id),
      crateLimitExceeded:Object.keys(crateLoads).length>ratedCrates,
      looseLimitExceeded:looseItemsUsed>SUBMARINE_LOOSE_ITEM_LIMIT
    };
  }

  function unifiedCargoNextCrateId(cargo=normalizeCargoManagement()){
    let index=1;
    const used=new Set(Object.keys(unifiedCargoCrates(cargo)));
    while(used.has(`crate-${index}`))index+=1;
    return `crate-${index}`;
  }

  function unifiedCargoAppendEntry(cargo,raw){
    const match=cargo.entries.find(entry=>entry.id===(raw.id||'')&&entry.customName===(raw.customName||catalogueItem(raw.id)?.name||'Unnamed Cargo')&&entry.owner===(raw.owner||'Submarine')&&entry.location===(raw.location||'Cargo Bay')&&entry.storageMode===raw.storageMode&&entry.crateId===(raw.crateId||''));
    if(match)match.quantity+=Math.max(1,number(raw.quantity,1));
    else cargo.entries.push(normalizeCargoEntry(raw));
  }

  function addUnifiedCargoItem(id,quantity=1,options={}){
    const cargo=normalizeCargoManagement();
    const item=id?catalogueItem(id):null;
    const amount=Math.max(1,Math.floor(number(quantity,1)));
    const storageMode=options.storageMode==='economic-crate'||options.economic===true?'economic-crate':'loose';
    const name=options.customName||item?.name||'Unnamed Cargo';
    const common={id:id||'',customName:name,unitSize:options.unitSize??(item?.category==='Submarine Ammunition'?1:item?.carryUnits||1),owner:options.owner||'Submarine',location:options.location||'Cargo Bay',notes:options.notes||'',acquiredFrom:options.acquiredFrom||'',storageMode};
    const report=unifiedCargoCapacityReport(cargo);
    if(storageMode==='loose'){
      if(report.looseItemsUsed+amount>SUBMARINE_LOOSE_ITEM_LIMIT){cargo.lastMessage=`Loose storage is full: ${report.looseItemsUsed}/${SUBMARINE_LOOSE_ITEM_LIMIT} items already aboard.`;return false;}
      unifiedCargoAppendEntry(cargo,{...common,quantity:amount,crateId:''});
    }else{
      if(report.economicItems+amount>report.economicItemCapacity){cargo.lastMessage=`Economic cargo exceeds ${report.ratedCrates} rated crates (${report.economicItemCapacity} standard items).`;return false;}
      let remaining=amount;
      const crateLoads={...report.crateLoads};
      const availableCrates=Object.keys(crateLoads).sort((left,right)=>left.localeCompare(right,undefined,{numeric:true}));
      for(const crateId of availableCrates){
        if(remaining<=0)break;
        const free=Math.max(0,SUBMARINE_STANDARD_CRATE_ITEM_LIMIT-number(crateLoads[crateId]));
        if(!free)continue;
        const packed=Math.min(free,remaining);
        unifiedCargoAppendEntry(cargo,{...common,quantity:packed,crateId});
        crateLoads[crateId]=number(crateLoads[crateId])+packed;
        remaining-=packed;
      }
      while(remaining>0){
        if(Object.keys(crateLoads).length>=report.ratedCrates){cargo.lastMessage=`No rated economic cargo crate remains available on ${state.submarine.model}.`;return false;}
        const crateId=unifiedCargoNextCrateId({entries:cargo.entries});
        const packed=Math.min(SUBMARINE_STANDARD_CRATE_ITEM_LIMIT,remaining);
        unifiedCargoAppendEntry(cargo,{...common,quantity:packed,crateId});
        crateLoads[crateId]=packed;
        remaining-=packed;
      }
    }
    cargo.lastMessage=`Added ${amount} ${name} to ${storageMode==='economic-crate'?'economic crate storage':'loose onboard storage'}.`;
    state.submarine.resources.cargo=unifiedCargoCapacityReport(cargo).economicCratesUsed;
    save();
    return true;
  }

  function repackUnifiedEconomicCargo(){
    const cargo=normalizeCargoManagement();
    const economic=cargo.entries.filter(entry=>entry.storageMode==='economic-crate').map(entry=>({...entry}));
    cargo.entries=cargo.entries.filter(entry=>entry.storageMode!=='economic-crate');
    let crateIndex=1;
    let remainingInCrate=SUBMARINE_STANDARD_CRATE_ITEM_LIMIT;
    for(const entry of economic){
      let remaining=Math.max(0,number(entry.quantity));
      while(remaining>0){
        if(remainingInCrate<=0){crateIndex+=1;remainingInCrate=SUBMARINE_STANDARD_CRATE_ITEM_LIMIT;}
        const packed=Math.min(remainingInCrate,remaining);
        cargo.entries.push(normalizeCargoEntry({...entry,cargoId:uid('cargo'),quantity:packed,storageMode:'economic-crate',crateId:`crate-${crateIndex}`}));
        remaining-=packed;
        remainingInCrate-=packed;
      }
    }
    cargo.lastMessage=`Economic cargo repacked into ${unifiedCargoCapacityReport(cargo).economicCratesUsed} crates.`;
    syncCargoResourceIfEnabled();
    save();
  }

  function fabricateOnboardItem(itemId,medical=false){
    const facilities=submarineFacilityProfile();

    if(number(state.submarine.resources.supplies)<supplyCost)return {ok:false,message:`Fabrication requires ${supplyCost} submarine supplies.`};

  function worldEconomyWildlifeRoll(key){
    return (seededHash(key)%1000000)/1000000;
  }

  function worldEconomyDamageStationForWildlife(station,severity,at){
    const keys=WORLD_ECONOMY_COMMODITIES.filter(key=>key!=='credits'&&number(station.inventory?.[key])>0);
    const losses={};
    for(const key of keys){
      const fraction=.03+severity*.12;
      const loss=Math.min(number(station.inventory[key]),Math.max(0,Math.floor(number(station.inventory[key])*fraction)));
      if(loss>0){station.inventory[key]-=loss;losses[key]=loss;}
    }
    station.construction.progress=Math.max(0,number(station.construction?.progress)-Math.round(5+severity*20));
    station.production.cyclesBlocked=Math.max(0,number(station.production?.cyclesBlocked))+1;
    station.production.lastProcessedAt=at;
    station.production.lastCycleSummary='Wildlife attack disrupted production and damaged stored resources.';
    station.updatedAt=at;
    worldEconomyJournal(`${station.name} suffered a wildlife attack; production was interrupted and stores were damaged.`,'wildlife-setback',{stationNodeId:station.stationNodeId,severity,losses});
  }

  processWorldStationEconomyCycle=function processWorldStationEconomyCycleWithWildlifeSetbacks(station,cycleAt=new Date().toISOString()){
    const node=state.world?.map?.nodes?.find(item=>item.id===station?.stationNodeId);
    const depthRatio=1-(Math.max(0,number(node?.ring))/Math.max(1,number(state.world?.map?.rings,48)));

    return processWorldStationEconomyCycleBeforeWildlifeSetbacks(station,cycleAt);
  };

    vessel.lastWildlifeEncounterSequence=Math.max(-1,number(raw.lastWildlifeEncounterSequence,-1));
    vessel.wildlifeDamageTotal=Math.max(0,number(raw.wildlifeDamageTotal));
    return vessel;
  };

  seedWorldEconomyNpcFleet=function seedWorldEconomyNpcFleetSlowly(){
    const map=state.world?.map;
    if(!map?.nodes?.length)return [];
    const economy=normalizeWorldEconomyState();
    const registry=worldVesselRegistry();
    let active=Object.values(registry).filter(vessel=>vessel.kind==='npc'&&vessel.active!==false);
    const target=worldEconomyNpcFleetTargetCount();
    economy.simulation.npcExpansion=economy.simulation.npcExpansion&&typeof economy.simulation.npcExpansion==='object'?economy.simulation.npcExpansion:{lastAttemptAt:'',attempts:0,successful:0,repelled:0};
    const expansion=economy.simulation.npcExpansion;
    const now=new Date();
    const last=Date.parse(expansion.lastAttemptAt||0);
    const bootstrap=active.length===0;
    if(!bootstrap&&Number.isFinite(last)&&now.getTime()-last<SUBMARINE_ECONOMY_EXPANSION_INTERVAL_HOURS*3600000)return active;
    if(active.length>=target)return active;
    const attempts=bootstrap?Math.min(2,target-active.length):1;
    const stations=map.nodes.filter(node=>node.type==='station');
    for(let index=0;index<attempts;index+=1){
      expansion.attempts+=1;
      const station=stations[Math.abs(seededHash(`${map.worldId||map.seed}|slow-expansion-station|${expansion.attempts}`))%Math.max(1,stations.length)];
      if(!station)break;
      const stationEconomy=worldStationEconomy(station.id);
      const depthRatio=1-number(station.ring)/Math.max(1,number(map.rings,48));
      const attackChance=.22+depthRatio*.18;
      const roll=worldEconomyWildlifeRoll(`${map.worldId||map.seed}|npc-expansion|${expansion.attempts}`);
      if(roll<attackChance){
        expansion.repelled+=1;
        if(stationEconomy)worldEconomyDamageStationForWildlife(stationEconomy,clamp(roll/attackChance,0,1),now.toISOString());
        worldEconomyJournal(`Wildlife repelled an NPC economic expansion attempt from ${station.name}.`,'npc-expansion-repelled',{stationNodeId:station.id,attempt:expansion.attempts});
        continue;
      }
      const available=Object.values(registry).find(vessel=>vessel.status==='available'&&vessel.availableForCommission);
      if(available){
        available.kind='npc';available.role='trade-freighter';available.status='docked';available.availableForCommission=false;available.ownerGroupId=station.parentFactionId||station.factionId||'npc-freight-guild';available.commissionedAt=now.toISOString();available.updatedAt=now.toISOString();available.nextActionAt=new Date(now.getTime()+(120+seededHash(`${available.vesselId}|slow-commission-dwell`)%361)*60000).toISOString();available.location={mode:'station',stationNodeId:station.id,currentNodeId:station.id,arrivedAt:now.toISOString(),decisionState:'slow-npc-commissioning'};
      }else worldEconomyCreateSlowNpcVessel(station,expansion.attempts,now);
      expansion.successful+=1;
    }
    expansion.lastAttemptAt=now.toISOString();
    active=Object.values(registry).filter(vessel=>vessel.kind==='npc'&&vessel.active!==false);
    return active;
  };

  const arriveWorldEconomyFreighterBeforeWildlifeDamage=arriveWorldEconomyFreighter;
  arriveWorldEconomyFreighter=function arriveWorldEconomyFreighterThroughWildlife(vessel,arriveAt=new Date()){
    if(!vessel||vessel.kind!=='npc'||vessel.status!=='in-transit')return false;
    Object.assign(vessel,normalizeWorldVesselRecord(vessel));
    const sequence=Math.max(0,number(vessel.transitSequence));
    if(vessel.lastWildlifeEncounterSequence!==sequence){
      const roll=worldEconomyWildlifeRoll(`${vessel.vesselId}|${sequence}|transit-wildlife`);
      vessel.lastWildlifeEncounterSequence=sequence;
      if(roll<.06){
        const manifest=worldEconomyClone(vessel.cargo?.commodities||{});
        vessel.hullCurrent=0;vessel.wildlifeDamageTotal+=vessel.hullMaximum;vessel.cargo.commodities={};vessel.status='inactive';vessel.active=false;vessel.nextActionAt='';vessel.updatedAt=arriveAt.toISOString();
        worldEconomyTradeJournal(`${vessel.name} was lost to native wildlife before completing its freight run.`,{vesselId:vessel.vesselId,sequence,cargoLost:manifest});
        worldEconomyJournal(`${vessel.name} was destroyed during NPC transit; Europa’s wildlife checked economic expansion along this route.`,'wildlife-vessel-loss',{vesselId:vessel.vesselId,sequence});
        return true;
      }
      if(roll<.30){
        const severity=clamp((.30-roll)/.24,0,1);
        const damage=Math.max(1,Math.round(1+severity*4));
        vessel.hullCurrent=Math.max(1,vessel.hullCurrent-damage);vessel.wildlifeDamageTotal+=damage;
        const lost={};
        for(const [key,value] of Object.entries(vessel.cargo?.commodities||{})){
          const amount=Math.min(number(value),Math.max(1,Math.floor(number(value)*(.15+severity*.35))));
          vessel.cargo.commodities[key]=Math.max(0,number(value)-amount);lost[key]=amount;
        }
        worldEconomyTradeJournal(`${vessel.name} survived a wildlife attack with hull damage and cargo losses.`,{vesselId:vessel.vesselId,sequence,damage,lost});
      }
    }
    const result=arriveWorldEconomyFreighterBeforeWildlifeDamage(vessel,arriveAt);
    if(result&&vessel.active!==false){
      const repairDelay=120+(seededHash(`${vessel.vesselId}|${sequence}|post-transit-dwell`)%361);
      vessel.nextActionAt=new Date(arriveAt.getTime()+repairDelay*60000).toISOString();
      vessel.updatedAt=arriveAt.toISOString();
    }
    return result;
  };
