  function dashboardManagementRefreshStaticLoginCopy(){
    const subtitle=document.querySelector('.ops-header .subtitle');
    if(subtitle)subtitle.textContent='Operate one persistent master world where submitted player submarines, NPC freight vessels, station production, trade, and shipyard construction remain active across saves.   function dashboardManagementStaticAuxiliaryHtml(selected){
    if(typeof dashboardManagementCreationAndRecoveryHtml!=='function')return '';
    return dashboardManagementCreationAndRecoveryHtml(selected)
      .replace(/<details class="tool-card"/g,'<details class="tool-card dashboard-management-secondary-card"')
      .replace(/\sopen><summary>/g,'><summary>');
  }
