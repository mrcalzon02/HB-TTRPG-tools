  function cloneRouteResolutionData(value) { return value === undefined ? undefined : JSON.parse(JSON.stringify(value)); }

  function mergeRouteResolutionEffects(target = {}, source = {}) {
    const result = cloneRouteResolutionData(target) || {};
    for (const [key, value] of Object.entries(source || {})) {
      if (value && typeof value === 'object' && !Array.isArray(value)) {
        result[key] = result[key] && typeof result[key] === 'object' && !Array.isArray(result[key]) ? result[key] : {};
        for (const [nestedKey, nestedValue] of Object.entries(value)) result[key][nestedKey] = number(result[key][nestedKey]) + number(nestedValue);
      } else result[key] = number(result[key]) + number(value);
    }
    return result;
  }

  function scaleRouteResolutionEffects(effects = {}, scale = 1) {
    const result = {};
    for (const [key, value] of Object.entries(effects || {})) {
      if (value && typeof value === 'object' && !Array.isArray(value)) {
        result[key] = {};
        for (const [nestedKey, nestedValue] of Object.entries(value)) {
          const amount = number(nestedValue);
          result[key][nestedKey] = amount === 0 ? 0 : Math.sign(amount) * Math.max(1, Math.round(Math.abs(amount) * scale));
        }
      } else {
        const amount = number(value);
        result[key] = amount === 0 ? 0 : Math.sign(amount) * Math.max(1, Math.round(Math.abs(amount) * scale));
      }
    }
    return result;
  }

  function defaultRouteEventResolutionState() { return { version:ROUTE_EVENT_RESOLUTION_VERSION, sessions:{}, contracts:[], history:[], lastMessage:'' }; }

  function normalizeRouteEventResolutionState() {
    state.route = state.route && typeof state.route === 'object' ? state.route : {};
    state.route.eventResolution = state.route.eventResolution && typeof state.route.eventResolution === 'object' ? state.route.eventResolution : defaultRouteEventResolutionState();
    const resolution = state.route.eventResolution;
    resolution.version = ROUTE_EVENT_RESOLUTION_VERSION;
    resolution.sessions = resolution.sessions && typeof resolution.sessions === 'object' && !Array.isArray(resolution.sessions) ? resolution.sessions : {};
    resolution.contracts = Array.isArray(resolution.contracts) ? resolution.contracts : [];
    resolution.history = Array.isArray(resolution.history) ? resolution.history.slice(0,200) : [];
    resolution.lastMessage = String(resolution.lastMessage || '');
    return resolution;
  }

  function routeResolutionProfile(event) { return ROUTE_RESOLUTION_FAMILY_PROFILES[event?.category] || ROUTE_RESOLUTION_FAMILY_PROFILES.Custom; }
  function routeResolutionTemplate(event) { return GENERAL_ENCOUNTER_TEMPLATES.find(template => template.id === event?.encounterTemplateId) || null; }

  function routeResolutionDutyForStat(stat, profile, index = 0) {
    const roles = [...(ROUTE_RESOLUTION_STAT_ROLES[stat] || []), ...(profile?.leadRoles || [])];
    return roles[index % Math.max(1, roles.length)] || 'Command';
  }

  function routeResolutionSolutions(event) {
    const template = routeResolutionTemplate(event);
    const profile = routeResolutionProfile(event);
    const approaches = Array.isArray(event?.encounterApproaches) && event.encounterApproaches.length ? event.encounterApproaches
      : Array.isArray(template?.approaches) && template.approaches.length ? template.approaches.map((approach,index) => ({...approach,id:`${template.id}-approach-${index}`,dc:number(event?.dc,template.baseDC||8)+number(approach.dcMod)}))
      : [{id:`${event?.id||'event'}-approach-0`,stat:event?.stat||'Intelligence',label:'Resolve the immediate threat',dc:number(event?.dc,8)}];
    return approaches.map((approach,index) => {
      const stat = approach.stat || event?.stat || 'Intelligence';
      const leadRole = routeResolutionDutyForStat(stat, profile, index);
      const supportRoles = [...new Set([...(profile.supportRoles||[]),...(ROUTE_RESOLUTION_STAT_ROLES[stat]||[])])].filter(role => role !== leadRole).slice(0,4);
      const setup = template?.setup || event?.description || 'The situation develops faster than expected.';
      const crisis = template?.stages?.crisis || event?.encounterComplication || 'The crew must commit before conditions deteriorate.';
      return {
        id:approach.id || `${event?.id||'event'}-solution-${index}`, label:approach.label || `Use ${stat}`, stat,
        dc:clamp(number(approach.dc,event?.dc||8),5,20), leadRole, supportRoles,
        description:`${setup} ${crisis} This plan places ${leadRole} personnel in the lead and uses ${stat} to ${String(approach.label||'resolve the incident').toLowerCase()}.`,
        successText:`The ${leadRole} plan controls the event. ${event?.reward||template?.reward||'The expedition secures an advantage.'}`,
        mixedText:`The plan works at a cost. ${event?.encounterPersistent||template?.persistent||'A consequence follows the expedition.'}`,
        failureText:`The plan loses control. ${event?.failure||template?.failure||'The submarine and crew absorb the consequences.'}`
      };
    });
  }
