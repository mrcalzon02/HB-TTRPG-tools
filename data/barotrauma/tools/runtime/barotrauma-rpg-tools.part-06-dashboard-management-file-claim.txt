
  dashboardManagementRuntime.claimFileSetup=dashboardManagementRuntime.claimFileSetup&&typeof dashboardManagementRuntime.claimFileSetup==='object'?dashboardManagementRuntime.claimFileSetup:{characterMode:'random',randomCharacter:null,startRevision:0,startOptionNodeIds:[],selectedStartNodeId:'',importText:''};

  function dashboardClaimRandomCharacter(force=false){
    const claimSetup=dashboardManagementRuntime.claimFileSetup;
    if(claimSetup.randomCharacter&&!force)return cloneWorldData(claimSetup.randomCharacter);
    const previous=dashboardManagementRuntime.newFileSetup.randomCharacter;
    const generated=dashboardRandomCharacter(true);
    dashboardManagementRuntime.newFileSetup.randomCharacter=previous;
    claimSetup.randomCharacter=cloneWorldData(generated);
    return generated;
  }

  function dashboardClaimStartingStations(refresh=false,count=12){
    const claimSetup=dashboardManagementRuntime.claimFileSetup;
    const map=state.world?.map;
    if(!map?.nodes?.length)return [];
    const outerRing=number(map.rings,Math.max(...map.nodes.map(node=>number(node.ring))));
    const outer=map.nodes.filter(node=>node.type==='station'&&number(node.ring)===outerRing);
    const stations=outer.length?outer:map.nodes.filter(node=>node.type==='station');
    const valid=new Set(stations.map(node=>node.id));
    claimSetup.startOptionNodeIds=(claimSetup.startOptionNodeIds||[]).filter(id=>valid.has(id));
    if(refresh||!claimSetup.startOptionNodeIds.length){
      claimSetup.startRevision=Math.max(0,number(claimSetup.startRevision))+1;
      const rng=dashboardSetupRandom(`claim-new-group-start|${claimSetup.startRevision}`);
      claimSetup.startOptionNodeIds=[...stations].map(node=>({node,value:rng()})).sort((left,right)=>left.value-right.value).slice(0,Math.min(count,stations.length)).map(entry=>entry.node.id);
      claimSetup.selectedStartNodeId=claimSetup.startOptionNodeIds.includes(claimSetup.selectedStartNodeId)?claimSetup.selectedStartNodeId:claimSetup.startOptionNodeIds[0]||'';
    }
    return claimSetup.startOptionNodeIds.map(id=>map.nodes.find(node=>node.id===id)).filter(Boolean);
  }

  claimExistingDashboardSubmarine=async function claimExistingDashboardSubmarineWithFileInitialization(vesselId,password,managerName,options={}){
    const record=dashboardManagementSubmarineRecord(vesselId);
    if(!record)throw new Error('The selected public-world submarine record no longer exists.');
    if(dashboardManagementCredential(vesselId)?.verifier)throw new Error('This submarine is already claimed.');
    if(!state.world?.map?.nodes?.length)throw new Error('Load the saved Masterworld before claiming a submarine.');
    const previousCharacter=cloneWorldData(state.character||defaultState().character);
    state.character=dashboardCharacterForNewManagementFile(options.characterSetup||{});
    normalizePlayerSubmissionSettings();
    if(typeof normalizeCharacterInventory==='function')normalizeCharacterInventory();
    const hub=normalizeWorldHub();
    const authority=normalizeDashboardManagementAuthority();
    let group=null;
    let createdGroup=false;
    try{
      if(options.groupId&&hub.publicState.groups?.[options.groupId])group=hub.publicState.groups[options.groupId];
      else{
        const cleanGroupName=String(options.newGroupName||'').trim();
        if(!cleanGroupName)throw new Error('Enter a name for the new expedition group.');
        group=createWorldGroup(cleanGroupName);
        createdGroup=true;
        dashboardApplyNewGroupStart(group,String(options.startNodeId||dashboardManagementRuntime.claimFileSetup.selectedStartNodeId||''));
      }
      dashboardManagementPairVesselToGroup(vesselId,group);
      const currentNodeId=String(group.currentNodeId||group.startingNodeId||'');
      if(!authority.vesselSnapshots[vesselId]){
        const rawModel=record.submarine?.model||(typeof record.model==='string'?record.model:record.model?.name||record.model?.id)||'Barsuk';
        const modelName=SUBMARINES[rawModel]?rawModel:'Barsuk';
        const submarine=dashboardManagementBuildSubmarineState(vesselId,record.name||record.submarine?.name||modelName,modelName,group);
        submarine.managedExpedition.status='inactive';
        submarine.managedExpedition.currentNodeId=currentNodeId;
        const route=deepMerge(defaultState().route,{});
        route.activeSubmarineId=vesselId;
        route.activeGroupId=group.groupId;
        const operationsDashboard=defaultActiveSubmarineDashboard();
        operationsDashboard.activeSubmarineId=vesselId;
        operationsDashboard.activeGroupId=group.groupId;
        operationsDashboard.activeNodeId=currentNodeId;
        operationsDashboard.startingLocation={...defaultDashboardStartingLocationState(),worldKey:dashboardWorldIdentityKey(),optionNodeIds:currentNodeId?[currentNodeId]:[],selectedNodeId:currentNodeId,confirmedNodeId:currentNodeId,locked:true,everDeparted:false,updatedAt:new Date().toISOString(),creationOnly:true};
        operationsDashboard.managementFile={version:DASHBOARD_MANAGEMENT_FILE_INITIALIZATION_VERSION,createdAt:new Date().toISOString(),characterMode:String(options.characterSetup?.mode||'random'),newGroup:createdGroup,startingNodeId:currentNodeId,claimedExistingVessel:true};
        authority.vesselSnapshots[vesselId]={version:DASHBOARD_MANAGEMENT_AUTH_VERSION,worldId:hub.worldId,vesselId,groupId:group.groupId,submarine,route,operationsDashboard,character:cloneWorldData(state.character),capturedAt:new Date().toISOString(),claimedFromExistingPublicRecord:true,managementFileVersion:DASHBOARD_MANAGEMENT_FILE_INITIALIZATION_VERSION};
      }
      Object.assign(record,{groupId:group.groupId,status:'inactive',activeForGroup:false,currentNodeId,storedAtNodeId:currentNodeId,managed:true,updatedAt:new Date().toISOString()});
      dashboardManagementAudit('existing-submarine-management-file-created',vesselId,{groupId:group.groupId,newGroup:createdGroup,startingNodeId:currentNodeId,characterMode:options.characterSetup?.mode||'random'});
      save();
      const result=await createDashboardManagementCredential(vesselId,password,managerName,Boolean(options.remember));
      dashboardManagementRuntime.claimFileSetup.randomCharacter=null;
      dashboardManagementRuntime.claimFileSetup.importText='';
      return result;
    }catch(error){
      state.character=previousCharacter;
      if(createdGroup&&group){delete hub.publicState.groups[group.groupId];delete hub.publicState.locations[group.groupId];}
      save();
      throw error;
    }
  };
