  const dashboardLocalStationTopologyBeforeCentralAnomaly=dashboardLocalStationTopology;
  dashboardLocalStationTopology=function dashboardLocalStationTopologyWithCentralAnomaly(map=state.world?.map){
    const topology=dashboardLocalStationTopologyBeforeCentralAnomaly(map);
    if(topology.centralAnomalyRouteReady)return topology;
    const anomaly=topology.worldTopology?.nodeById?.anomaly||map?.nodes?.find(node=>node.id==='anomaly'||node.type==='anomaly');
    topology.destinationById={...(topology.stationById||{})};
    if(!anomaly){topology.centralAnomalyRouteReady=true;return topology;}
    topology.destinationById[anomaly.id]=anomaly;
    topology.corridorsByStation=topology.corridorsByStation||{};
    topology.corridorsByStation[anomaly.id]=topology.corridorsByStation[anomaly.id]||[];
    topology.corridorById=topology.corridorById||{};
    for(const station of Object.values(topology.stationById||{}).filter(node=>number(node.ring)===1)){
      const edge=dashboardLocalEdgeForPair(topology.worldTopology,station.id,anomaly.id);
      if(!edge)continue;
      const id=`station-path-${station.id}-${anomaly.id}`;
      if(topology.corridorById[id])continue;
      const corridor={
        id,
        a:station.id,
        b:anomaly.id,
        relation:'central-anomaly',
        distance:Math.max(1,number(edge.distance,1)),
        danger:Boolean(edge.danger),
        voyages:1,
        pathNodeIds:[station.id,anomaly.id],
        edgeIds:[edge.id],
        edges:[edge],
        localNodePath:true,
        finalCenterMission:true
      };
      topology.corridors.push(corridor);
      topology.corridorById[id]=corridor;
      topology.corridorsByStation[station.id]=topology.corridorsByStation[station.id]||[];
      topology.corridorsByStation[station.id].push(corridor);
      topology.corridorsByStation[anomaly.id].push(corridor);
    }
    topology.centralAnomalyRouteReady=true;
    return topology;
  };

  dashboardAdjacentStationRoutes=function dashboardAdjacentStationRoutesIncludingCentralAnomaly(originId,map=state.world?.map){
    const topology=dashboardLocalStationTopology(map);
    const origin=topology.destinationById?.[originId]||topology.stationById?.[originId];
    if(!origin)return [];
    return (topology.corridorsByStation?.[originId]||[]).map(corridor=>{
      const destinationId=corridor.a===originId?corridor.b:corridor.a;
      const destination=topology.destinationById?.[destinationId]||topology.stationById?.[destinationId];
      const ringDelta=number(destination?.ring)-number(origin.ring);
      const forward=corridor.a===originId;
      return {
        stationId:destinationId,
        station:destination,
        destinationId,
        destination,
        distance:corridor.distance,
        danger:corridor.danger,
        relation:destination?.type==='anomaly'?'central-anomaly':ringDelta<0?'one-ring-inward':ringDelta>0?'one-ring-outward':'same-ring-adjacent',
        pathNodeIds:forward?[...corridor.pathNodeIds]:[...corridor.pathNodeIds].reverse(),
        edgeIds:forward?[...corridor.edgeIds]:[...corridor.edgeIds].reverse(),
        edges:(forward?[...corridor.edgeIds]:[...corridor.edgeIds].reverse()).map(id=>topology.worldTopology.edgeById[id]).filter(Boolean),
        localNodePath:true,
        finalCenterMission:Boolean(corridor.finalCenterMission)
      };
    }).filter(route=>route.destination&&Math.abs(number(route.destination.ring)-number(origin.ring))<=1&&route.pathNodeIds.length>=2);
  };

  function dashboardIncrementalLocalCorridorsForRing(packageInfo,map=state.world?.map){
    if(typeof dashboardLocalWorldTopology!=='function')return [];
    const topology=dashboardLocalWorldTopology(map);
    return topology.edges.filter(edge=>{
      const left=topology.nodeById[edge.a];
      const right=topology.nodeById[edge.b];
      if(!left||!right)return false;
      const leftRing=Math.round(number(left.ring));
      const rightRing=Math.round(number(right.ring));
      const difference=Math.abs(leftRing-rightRing);
      if(difference>1)return false;
      const displayRing=difference===0?leftRing:Math.min(leftRing,rightRing);
      return displayRing===packageInfo.ring;
    });
  }

  function dashboardMasterWorldFitDimensions(map,viewport){
    const mapWidth=Math.max(1,number(map?.width,1000));
    const mapHeight=Math.max(1,number(map?.height,1000));
    const availableWidth=Math.max(320,(viewport?.clientWidth||1000)-18);
    const availableHeight=Math.max(320,(viewport?.clientHeight||760)-18);
    const fitScale=Math.min(availableWidth/mapWidth,availableHeight/mapHeight);
    return {
      width:Math.max(320,Math.round(mapWidth*fitScale)),
      height:Math.max(320,Math.round(mapHeight*fitScale))
    };
  }

  function dashboardMasterWorldApplyZoom(nextZoom,anchorClientX=null,anchorClientY=null){
    const {viewport,svg,zoomLabel}=dashboardMasterWorldViewportElements();
    if(!viewport||!svg)return false;
    const previousWidth=Math.max(1,svg.getBoundingClientRect().width||dashboardMasterWorldViewportRuntime.baseWidth||1);
    const previousHeight=Math.max(1,svg.getBoundingClientRect().height||dashboardMasterWorldViewportRuntime.baseHeight||1);
    const viewportRect=viewport.getBoundingClientRect();
    const localAnchorX=anchorClientX===null?viewport.clientWidth/2:anchorClientX-viewportRect.left;
    const localAnchorY=anchorClientY===null?viewport.clientHeight/2:anchorClientY-viewportRect.top;
    const mapRatioX=(viewport.scrollLeft+localAnchorX)/previousWidth;
    const mapRatioY=(viewport.scrollTop+localAnchorY)/previousHeight;
    const zoom=dashboardMasterWorldClampZoom(nextZoom);
    const renderedWidth=Math.max(1,Math.round(dashboardMasterWorldViewportRuntime.baseWidth*zoom));
    const renderedHeight=Math.max(1,Math.round(dashboardMasterWorldViewportRuntime.baseHeight*zoom));
    dashboardMasterWorldViewportRuntime.zoom=zoom;
    svg.style.width=`${renderedWidth}px`;
    svg.style.height=`${renderedHeight}px`;
    svg.style.minWidth='0';
    svg.style.maxWidth='none';
    viewport.scrollLeft=Math.max(0,mapRatioX*renderedWidth-localAnchorX);
    viewport.scrollTop=Math.max(0,mapRatioY*renderedHeight-localAnchorY);
    if(zoomLabel)zoomLabel.textContent=`${Math.round(zoom*100)}%`;
    const zoomOut=root.querySelector('#dashboard-world-zoom-out');
    const zoomIn=root.querySelector('#dashboard-world-zoom-in');
    if(zoomOut)zoomOut.disabled=zoom<=DASHBOARD_MASTERWORLD_ZOOM_MIN;
    if(zoomIn)zoomIn.disabled=zoom>=DASHBOARD_MASTERWORLD_ZOOM_MAX;
    return true;
  }

      line.setAttribute('aria-label',`Route ${left?.name||edge?.a||''} to ${right?.name||edge?.b||''}`);
      if(dashboardMasterWorldViewportRuntime.selectedType==='route'&&dashboardMasterWorldViewportRuntime.selectedId===line.dataset.localCorridorId)line.classList.add('viewer-selected');
      const hitArea=line.cloneNode(false);
      hitArea.removeAttribute('tabindex');
      hitArea.removeAttribute('role');
      hitArea.removeAttribute('aria-label');
      hitArea.dataset.viewerHitArea='true';
      hitArea.classList.remove('dashboard-incremental-corridor','viewer-selected');
      hitArea.classList.add('dashboard-incremental-corridor-hit');
      line.after(hitArea);
    }
    dashboardMasterWorldApplySelectionClasses();
    return added;
  };

  dashboardIncrementalUpdateDetail=function dashboardIncrementalUpdateSelectableNodeDetail(node){
    dashboardIncrementalUpdateDetailBeforeViewportSelection(node);
    dashboardMasterWorldSelectNode(node,state.world?.map);
  };

  function dashboardMasterWorldBindViewportInteractions(map=state.world?.map){
    const {viewport,svg}=dashboardMasterWorldViewportElements();
    if(!viewport||!svg||viewport.dataset.viewerBound==='true')return false;
    viewport.dataset.viewerBound='true';
    dashboardMasterWorldViewportRuntime.initialized=true;
    dashboardMasterWorldResetZoom(map);

    root.querySelector('#dashboard-world-zoom-out')?.addEventListener('click',()=>dashboardMasterWorldApplyZoom(dashboardMasterWorldViewportRuntime.zoom-DASHBOARD_MASTERWORLD_ZOOM_STEP));
    root.querySelector('#dashboard-world-zoom-in')?.addEventListener('click',()=>dashboardMasterWorldApplyZoom(dashboardMasterWorldViewportRuntime.zoom+DASHBOARD_MASTERWORLD_ZOOM_STEP));
    root.querySelector('#dashboard-world-reset-zoom')?.addEventListener('click',()=>dashboardMasterWorldResetZoom(map));
    root.querySelector('#dashboard-world-center-selection')?.addEventListener('click',dashboardMasterWorldCenterSelection);
    root.querySelector('#dashboard-world-clear-selection')?.addEventListener('click',dashboardMasterWorldClearSelection);

    svg.addEventListener('click',event=>{
      const routeTarget=event.target.closest?.('[data-local-corridor-id]');
      if(routeTarget){dashboardMasterWorldSelectRoute(routeTarget.dataset.localCorridorId,map);return;}
      const nodeTarget=event.target.closest?.('[data-node-id]');
      if(!nodeTarget)return;
      const node=map?.nodes?.find(item=>item.id===nodeTarget.dataset.nodeId);
      if(node)dashboardMasterWorldSelectNode(node,map);
    });

    svg.addEventListener('keydown',event=>{
      const interactive=event.target.closest?.('[data-node-id],[data-local-corridor-id]');
      if(!interactive||!['Enter',' '].includes(event.key))return;
      event.preventDefault();
      interactive.dispatchEvent(new MouseEvent('click',{bubbles:true,clientX:0,clientY:0}));
    });

    viewport.addEventListener('wheel',event=>{
      if(!(event.ctrlKey||event.metaKey))return;
      event.preventDefault();
      const direction=event.deltaY<0?1:-1;
      dashboardMasterWorldApplyZoom(dashboardMasterWorldViewportRuntime.zoom+direction*DASHBOARD_MASTERWORLD_ZOOM_STEP,event.clientX,event.clientY);
    },{passive:false});

    let drag=null;
    viewport.addEventListener('pointerdown',event=>{
      if(event.button!==0||event.target.closest?.('[data-node-id],[data-local-corridor-id]'))return;
      drag={pointerId:event.pointerId,x:event.clientX,y:event.clientY,left:viewport.scrollLeft,top:viewport.scrollTop};
      viewport.classList.add('viewer-dragging');
      viewport.setPointerCapture?.(event.pointerId);
      event.preventDefault();
    });
    viewport.addEventListener('pointermove',event=>{
      if(!drag||drag.pointerId!==event.pointerId)return;
      viewport.scrollLeft=drag.left-(event.clientX-drag.x);
      viewport.scrollTop=drag.top-(event.clientY-drag.y);
    });
    const endDrag=event=>{
      if(!drag||drag.pointerId!==event.pointerId)return;
      viewport.releasePointerCapture?.(event.pointerId);
      drag=null;
      viewport.classList.remove('viewer-dragging');
    };
    viewport.addEventListener('pointerup',endDrag);
    viewport.addEventListener('pointercancel',endDrag);

    viewport.addEventListener('keydown',event=>{
      const scrollStep=Math.max(80,Math.round(Math.min(viewport.clientWidth,viewport.clientHeight)*.16));
      if(event.key==='+'||event.key==='='){event.preventDefault();dashboardMasterWorldApplyZoom(dashboardMasterWorldViewportRuntime.zoom+DASHBOARD_MASTERWORLD_ZOOM_STEP);}
      else if(event.key==='-'){event.preventDefault();dashboardMasterWorldApplyZoom(dashboardMasterWorldViewportRuntime.zoom-DASHBOARD_MASTERWORLD_ZOOM_STEP);}
      else if(event.key==='0'){event.preventDefault();dashboardMasterWorldResetZoom(map);}
      else if(event.key==='Home'){event.preventDefault();dashboardMasterWorldCenterSelection();}
      else if(event.key==='ArrowLeft'){event.preventDefault();viewport.scrollLeft-=scrollStep;}
      else if(event.key==='ArrowRight'){event.preventDefault();viewport.scrollLeft+=scrollStep;}
      else if(event.key==='ArrowUp'){event.preventDefault();viewport.scrollTop-=scrollStep;}
      else if(event.key==='ArrowDown'){event.preventDefault();viewport.scrollTop+=scrollStep;}
    });

    dashboardMasterWorldApplySelectionClasses();
    return true;
  }
