  function eligibleEvents(category = 'All', sanityEnabled = true) {
    return EVENTS.filter(event => (category === 'All' || event.category === category) && (sanityEnabled || event.category !== 'Sanity'));
  }

  function drawRandomEvent(category = 'All', sanityEnabled = true) {
    const pool = eligibleEvents(category, sanityEnabled);
    return pool[Math.floor(Math.random() * pool.length)] || EVENTS[0];
  }

  function weightedCategory(table, sanityEnabled, rng = Math.random) {
    const entries = Object.entries(table).filter(([category, weight]) => number(weight) > 0 && (sanityEnabled || category !== 'Sanity'));
    const total = entries.reduce((sum, [, weight]) => sum + number(weight), 0);
    if (!total) return 'Hazard';
    let roll = rng() * total;
    for (const [category, weight] of entries) {
      roll -= number(weight);
      if (roll <= 0) return category;
    }
    return entries[entries.length - 1][0];
  }

  function routeTotals() {
    const route = state.route;
    const selectedJobs = JOBS.filter(job => route.jobs.includes(job.id));
    const jobEvents = selectedJobs.reduce((sum, job) => sum + job.events, 0);
    const reduction = route.navigationExpert ? Math.max(0, number(route.navigationReduction, 1)) : 0;
    const routeEvents = Math.max(0, number(route.distance) + number(route.routePressure) - reduction);
