    return ['arrived','active','success','Onboard'].includes(value) ? 'good' : ['underway','prepared','Injured','pending'].includes(value) ? 'warn' : ['lost','aborted','failure','Dead'].includes(value) ? 'bad' : '';

  function dashboardMapSelection(context, full = false) {
    const map = state.world?.map;
    if (!map?.nodes?.length) return { nodes: [], edges: [], idSet: new Set() };
    if (full) return { nodes: map.nodes, edges: map.edges, idSet: new Set(map.nodes.map(node => node.id)) };
    const wanted = new Set();
    const queue = [];
    const add = id => { if (id && !wanted.has(id)) { wanted.add(id); queue.push(id); } };
    add(context.group.currentNodeId || map.startId);
    add(state.operationsDashboard.focusedNodeId);
    for (const id of state.world.selected || []) add(id);
    for (const id of normalizeRouteCrossing().pathNodeIds || []) add(id);
    let depth = 0;
    while (queue.length && wanted.size < 72 && depth < 3) {
      const batch = queue.splice(0, queue.length);
      for (const id of batch) {
        for (const edge of map.edges) {
          if (edge.a === id) add(edge.b);
          else if (edge.b === id) add(edge.a);
          if (wanted.size >= 72) break;
        }
        if (wanted.size >= 72) break;
      }
      depth += 1;
    }
    const nodes = map.nodes.filter(node => wanted.has(node.id));
    const idSet = new Set(nodes.map(node => node.id));
    const edges = map.edges.filter(edge => idSet.has(edge.a) && idSet.has(edge.b));
    return { nodes, edges, idSet };
  }

  function dashboardOperationalMapSvg(context) {
    const map = state.world?.map;
    if (!map?.nodes?.length) return '';
    const full = state.operationsDashboard.mapMode === 'full';
    const selection = dashboardMapSelection(context, full);
    const nodes = selection.nodes;
    const edges = selection.edges;
    const nodeById = Object.fromEntries(nodes.map(node => [node.id, node]));
    const xs = nodes.map(node => number(node.x));
    const ys = nodes.map(node => number(node.y));
    const margin = full ? 160 : 90;
    const minX = Math.min(...xs) - margin;
    const maxX = Math.max(...xs) + margin;
    const minY = Math.min(...ys) - margin;
    const maxY = Math.max(...ys) + margin;
    const width = Math.max(240, maxX - minX);
    const height = Math.max(180, maxY - minY);
    const routeIds = new Set(normalizeRouteCrossing().pathNodeIds || []);
    const selectedIds = new Set(state.world.selected || []);
    const focused = state.operationsDashboard.focusedNodeId;
    const current = context.group.currentNodeId;
    let stationLabels = 0;
    return `<svg id="dashboard-operational-map" viewBox="${minX} ${minY} ${width} ${height}" role="img" aria-label="Active submarine operational world map" style="width:100%;height:${full ? 760 : 520}px;display:block;background:radial-gradient(circle at center,rgba(200,138,53,.07),rgba(0,0,0,.35));border-radius:14px">
      ${edges.map(edge => { const a = nodeById[edge.a]; const b = nodeById[edge.b]; if (!a || !b) return ''; const active = routeIds.has(edge.a) && routeIds.has(edge.b); return `<line x1="${number(a.x)}" y1="${number(a.y)}" x2="${number(b.x)}" y2="${number(b.y)}" stroke="${active ? '#ffffff' : '#b8c2cc'}" stroke-width="${isCurrent ? 7 : selected || route ? 4 : 2}"/>${showLabel ? `<text x="${radius + 7}" y="4" fill="#e8edf2" font-size="${isCurrent ? 18 : 13}" font-weight="${isCurrent ? 900 : 700}" pointer-events="none">${escapeHtml(node.name)}</text>` : ''}</g>`; }).join('')}

  function dashboardGenerateWorld(randomize = false) {
    try {

      generateWorldMap();

        if (typeof reconcileWorldStationLevels === 'function') reconcileWorldStationLevels(state.world.map);

      dashboardSetMessage(`World generation failed: ${error.message}`, true);
    }
    renderSubmarine();
  }

    root.querySelector('#dashboard-generate-crossing')?.addEventListener('click', () => { try { generateValidCrossing(); dashboardSetMessage('Valid crossing generated for the active submarine.'); } catch (error) { dashboardSetMessage(error.message, true); } renderSubmarine(); });
    root.querySelector('#dashboard-start-crossing')?.addEventListener('click', () => { startCrossing(); renderSubmarine(); });
    root.querySelector('#dashboard-abort-crossing')?.addEventListener('click', () => { abortCrossing(); normalizeActiveSubmarineDashboard(true); renderSubmarine(); });
    root.querySelector('#dashboard-clear-crossing')?.addEventListener('click', () => { clearRouteCrossing(); normalizeActiveSubmarineDashboard(true); renderSubmarine(); });
    root.querySelector('#dashboard-complete-leg')?.addEventListener('click', () => { completeCrossingLeg(); renderSubmarine(); });
    root.querySelector('#dashboard-roll-next')?.addEventListener('click', () => { const slot = nextPendingCrossingSlot(); if (slot) rollCrossingEvent(slot.id); renderSubmarine(); });
    root.querySelector('#dashboard-success-next')?.addEventListener('click', () => { const slot = nextPendingCrossingSlot(); if (slot) resolveCrossingEvent(slot.id, 'success'); renderSubmarine(); });
    root.querySelector('#dashboard-failure-next')?.addEventListener('click', () => { const slot = nextPendingCrossingSlot(); if (slot) resolveCrossingEvent(slot.id, 'failure'); renderSubmarine(); });
    root.querySelector('#dashboard-redraw-next')?.addEventListener('click', () => { const slot = nextPendingCrossingSlot(); if (slot) redrawCrossingEvent(slot.id); renderSubmarine(); });

    if (focus === 'route' || focus === 'world') requestAnimationFrame(() => root.querySelector(`#active-dashboard-${focus}`)?.scrollIntoView({ block: 'start' }));
  }

  const renderSubmarineBeforeActiveDashboard = renderSubmarine;

  function renderActiveSubmarineDashboard(requestedFocus = activeDashboardRawFocus()) {
    let context;
    let baseError = '';
    try {
      context = normalizeActiveSubmarineDashboard(false);
    } catch (error) {
      root.innerHTML = `<div class="tool-shell"><section class="tool-hero"><div><p class="eyebrow">Active submarine dashboard</p><h2>Identity normalization failed</h2><p class="rules-note bad">${escapeHtml(error.message)}</p></div></section></div>`;
      return;
    }
    try {
      renderSubmarineBeforeActiveDashboard();
    } catch (error) {
      baseError = error.message || String(error);
      root.innerHTML = `<div class="tool-shell"><section class="tool-hero"><div><p class="eyebrow">Unified operations fallback</p><h2>${escapeHtml(state.submarine.name || state.submarine.model)}</h2><p class="rules-note">The older submarine panel encountered an error, but the active route and world dashboard has remained available.</p></div></section></div>`;
    }
    const shell = root.querySelector('.tool-shell');
    if (!shell) return;
    const hero = shell.querySelector(':scope > .tool-hero');
    if (hero) {
      const eyebrow = hero.querySelector('.eyebrow');
      const title = hero.querySelector('h2');
      const note = hero.querySelector('.rules-note');
      if (eyebrow) eyebrow.textContent = 'Single-vessel campaign command';
      if (title) title.textContent = 'Active Submarine Management Dashboard';
      if (note) note.textContent = 'Manage the vessel, its ID-linked crew, filed route, current world position, systems, stores, cargo, casualties, and expedition state from one operational page.';
      hero.insertAdjacentHTML('afterend', activeDashboardHeaderHtml(context, requestedFocus, baseError));
    } else shell.insertAdjacentHTML('afterbegin', activeDashboardHeaderHtml(context, requestedFocus, baseError));
    shell.insertAdjacentHTML('beforeend', dashboardRoutePanelHtml(context, requestedFocus));
    shell.insertAdjacentHTML('beforeend', dashboardWorldPanelHtml(context, requestedFocus));
    bindActiveDashboardControls(context, requestedFocus);
  }

  renderSubmarine = function renderUnifiedActiveSubmarineDashboard() {
    renderActiveSubmarineDashboard(activeDashboardRawFocus());
  };
  renderRoute = function renderRouteInsideActiveSubmarineDashboard() {
    renderActiveSubmarineDashboard('route');
  };
  renderWorld = function renderWorldInsideActiveSubmarineDashboard() {
    renderActiveSubmarineDashboard('world');
  };
