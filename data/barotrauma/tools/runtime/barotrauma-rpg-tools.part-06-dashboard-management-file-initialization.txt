    Engineer:['Intelligence','Agility'],

  function dashboardSetupRandom(seed=''){
    return seededRandom(`${state.world?.seed||'EUROPA'}|management-file|${seed}|${Date.now()}|${Math.random()}`);
  }

  function dashboardSetupChoice(values,rng){
    return values[Math.floor(rng()*Math.max(1,values.length))]||values[0];
  }

  function dashboardRandomPersonName(rng){
    return `${dashboardSetupChoice(DASHBOARD_RANDOM_GIVEN_NAMES,rng)} ${dashboardSetupChoice(DASHBOARD_RANDOM_SURNAMES,rng)}`;
  }

  function dashboardVariedStatBlock(className='Engineer',seed=''){
    const rng=dashboardSetupRandom(`${seed}|${className}`);
    const stats={
      Strength:2+Math.floor(rng()*4),
      Agility:2+Math.floor(rng()*4),
      Intelligence:2+Math.floor(rng()*4),
      Charisma:2+Math.floor(rng()*4)
    };
    for(const stat of DASHBOARD_CLASS_STAT_BIASES[className]||['Intelligence'])stats[stat]=clamp(stats[stat]+1,1,6);
    const values=Object.values(stats);
    if(new Set(values).size===1){
      const primary=(DASHBOARD_CLASS_STAT_BIASES[className]||['Intelligence'])[0];
      stats[primary]=clamp(stats[primary]+1,1,6);
      const lowest=Object.keys(stats).find(stat=>stat!==primary);
      stats[lowest]=clamp(stats[lowest]-1,1,6);
    }
    const highest=Math.max(...Object.values(stats));
    const lowest=Math.min(...Object.values(stats));
    if(highest-lowest<2){
      const primary=(DASHBOARD_CLASS_STAT_BIASES[className]||['Intelligence'])[0];
      stats[primary]=clamp(stats[primary]+1,1,6);
      const secondary=Object.keys(stats).sort((left,right)=>stats[left]-stats[right])[0];
      if(secondary!==primary)stats[secondary]=clamp(stats[secondary]-1,1,6);
    }
    return stats;
  }

  function dashboardRandomCharacter(force=false){
    const setup=dashboardManagementRuntime.newFileSetup;
    if(setup.randomCharacter&&!force)return cloneWorldData(setup.randomCharacter);
    const rng=dashboardSetupRandom(`player-character|${force?'reroll':'initial'}`);
    const className=dashboardSetupChoice(Object.keys(CLASSES),rng);
    const background=dashboardSetupChoice(BACKGROUNDS,rng);
    const talent=dashboardSetupChoice(CLASSES[className]?.talents||[['Quick Fix','']],rng)?.[0]||'';
    const stats=dashboardVariedStatBlock(className,`player|${rng()}`);
    const character=deepMerge(defaultState().character,{
      name:dashboardRandomPersonName(rng),
      player:state.character?.player||'',
      className,
      level:1+Math.floor(rng()*3),
      background:background?.name||'',
      talent,
      stats,
      current:{...stats},
      fateUsed:false,
      lostSanity:0,
      morale:0,
      credits:0,
      inventory:'',
      conditions:'',
      notes:'Randomized during management-file initialization.',
      rollLog:[],
      catalogInventory:[],
      consumptionLog:[],
      playerSubmission:defaultPlayerSubmissionSettings()
    });
    setup.randomCharacter=cloneWorldData(character);
    return character;
  }

  function dashboardCharacterFromImportText(text){
    const clean=String(text||'').trim();
    if(!clean)throw new Error('Paste or select a character JSON file before creating the management file.');
    let parsed;
    try{parsed=JSON.parse(clean);}catch(error){throw new Error(`Character JSON could not be parsed: ${error.message}`);}
    let source=null;
    let playerName='';
    if(parsed?.schema==='barotrauma-crew-submission'&&parsed.character){source=parsed.character;playerName=parsed.player?.name||'';}
    else if(parsed?.character&&typeof parsed.character==='object'){source=parsed.character;playerName=parsed.player?.name||parsed.character.player||'';}
    else if(parsed?.state?.character&&typeof parsed.state.character==='object'){source=parsed.state.character;playerName=source.player||'';}
    else if(parsed?.payload?.character&&typeof parsed.payload.character==='object'){source=parsed.payload.character;playerName=source.player||'';}
    else if(parsed?.stats&&typeof parsed.stats==='object'){source=parsed;playerName=parsed.player||'';}
    if(!source)throw new Error('The imported JSON does not contain a recognizable Barotrauma character record.');
    const stats={
      Strength:clamp(source.stats?.Strength,1,12),
      Agility:clamp(source.stats?.Agility,1,12),
      Intelligence:clamp(source.stats?.Intelligence,1,12),
      Charisma:clamp(source.stats?.Charisma,1,12)
    };
    const current={
      Strength:clamp(source.current?.Strength??stats.Strength,0,12),
      Agility:clamp(source.current?.Agility??stats.Agility,0,12),
      Intelligence:clamp(source.current?.Intelligence??stats.Intelligence,0,12),
      Charisma:clamp(source.current?.Charisma??stats.Charisma,0,12)
    };
    const className=CLASSES[source.className]?source.className:'Engineer';
    const character=deepMerge(defaultState().character,cloneWorldData(source));
    Object.assign(character,{
      name:String(source.name||source.characterName||'Imported Character'),
      player:String(playerName||source.player||''),
      className,
      level:clamp(source.level,1,99),
      stats,
      current,
      rollLog:Array.isArray(source.rollLog)?source.rollLog:[],
      catalogInventory:Array.isArray(source.catalogInventory)?source.catalogInventory:Array.isArray(source.inventory)?source.inventory:[],
      consumptionLog:Array.isArray(source.consumptionLog)?source.consumptionLog:[],
      inventory:Array.isArray(source.inventory)?String(source.inventoryText||''):String(source.inventory||source.inventoryText||''),
      playerSubmission:source.playerSubmission&&typeof source.playerSubmission==='object'?cloneWorldData(source.playerSubmission):defaultPlayerSubmissionSettings()
    });
    return character;
  }

  function dashboardCharacterForNewManagementFile(setup={}){
    const mode=['import','create','random'].includes(setup.mode)?setup.mode:'random';
    if(mode==='import')return dashboardCharacterFromImportText(setup.importText);
    if(mode==='create')return dashboardCharacterFromManual(setup.manual||{});
    const random=setup.randomCharacter&&typeof setup.randomCharacter==='object'?cloneWorldData(setup.randomCharacter):dashboardRandomCharacter(false);
    if(setup.randomPlayerName)random.player=String(setup.randomPlayerName).trim();
    return random;
  }

  function dashboardNewGroupStartingStations(refresh=false,count=12){
    const setup=dashboardManagementRuntime.newFileSetup;
    const map=state.world?.map;
    if(!map?.nodes?.length)return [];
    const outerRing=number(map.rings,Math.max(...map.nodes.map(node=>number(node.ring))));
    const outerStations=map.nodes.filter(node=>node.type==='station'&&number(node.ring)===outerRing);
    const stations=outerStations.length?outerStations:map.nodes.filter(node=>node.type==='station');
    const validIds=new Set(stations.map(node=>node.id));
    setup.startOptionNodeIds=(setup.startOptionNodeIds||[]).filter(id=>validIds.has(id));
    if(refresh||!setup.startOptionNodeIds.length){
      setup.startRevision=Math.max(0,number(setup.startRevision))+1;
      const rng=dashboardSetupRandom(`new-group-start|${setup.startRevision}`);
      setup.startOptionNodeIds=[...stations].map(node=>({node,value:rng()})).sort((left,right)=>left.value-right.value).slice(0,Math.min(count,stations.length)).map(entry=>entry.node.id);
      setup.selectedStartNodeId=setup.startOptionNodeIds.includes(setup.selectedStartNodeId)?setup.selectedStartNodeId:setup.startOptionNodeIds[0]||'';
    }
    return setup.startOptionNodeIds.map(id=>map.nodes.find(node=>node.id===id)).filter(Boolean);
  }

  function dashboardApplyNewGroupStart(group,nodeId){
    const map=state.world?.map;
    const node=map?.nodes?.find(candidate=>candidate.id===nodeId&&candidate.type==='station');
    if(!node)throw new Error('Choose a valid randomized starting station for the new expedition group.');
    group.startingNodeId=node.id;
    group.currentNodeId=node.id;
    group.routeId='';
    normalizeWorldHub().publicState.locations[group.groupId]={groupId:group.groupId,nodeId:node.id,updatedAt:new Date().toISOString()};
    return node;
  }

  function dashboardGroupFleetIds(group){
    group.submarineIds=Array.isArray(group.submarineIds)?[...new Set(group.submarineIds.filter(Boolean))]:[];
    return group.submarineIds;
  }

  function dashboardGroupActiveVesselId(group){
    const ids=dashboardGroupFleetIds(group);
    const candidate=String(group.activeSubmarineId||group.primarySubmarineId||'');
    return ids.includes(candidate)?candidate:ids.find(id=>normalizeWorldHub().publicState.submarines?.[id]?.status==='active')||ids[0]||'';
  }

  function dashboardVesselStationNodeId(vesselId,group){
    const record=normalizeWorldHub().publicState.submarines?.[vesselId]||{};
    return String(record.storedAtNodeId||record.currentNodeId||group?.currentNodeId||group?.startingNodeId||'');
  }

  function dashboardVesselCrossingStatus(vesselId){
    const snapshot=normalizeDashboardManagementAuthority().vesselSnapshots?.[vesselId];
    return String(snapshot?.route?.crossing?.status||'');
  }

  function dashboardMarkVesselInactive(vesselId,group,nodeId,reason='Another submarine became active'){
    if(!vesselId)return;
    const hub=normalizeWorldHub();
    const now=new Date().toISOString();
    const locationNodeId=String(nodeId||group?.currentNodeId||group?.startingNodeId||'');
    const record=hub.publicState.submarines[vesselId]||{submarineId:vesselId,groupId:group?.groupId||''};
    record.groupId=group?.groupId||record.groupId||'';
    record.status='inactive';
    record.activeForGroup=false;
    record.currentNodeId=locationNodeId;
    record.storedAtNodeId=locationNodeId;
    record.storedAtAt=now;
    record.storageReason=reason;
    record.updatedAt=now;
    if(record.submarine?.managedExpedition){record.submarine.managedExpedition.status='inactive';record.submarine.managedExpedition.currentNodeId=locationNodeId;}
    hub.publicState.submarines[vesselId]=record;
    const snapshot=normalizeDashboardManagementAuthority().vesselSnapshots?.[vesselId];
    if(snapshot){
      snapshot.submarine=snapshot.submarine&&typeof snapshot.submarine==='object'?snapshot.submarine:{};
      snapshot.submarine.managedExpedition=snapshot.submarine.managedExpedition&&typeof snapshot.submarine.managedExpedition==='object'?snapshot.submarine.managedExpedition:{};
      snapshot.submarine.managedExpedition.status='inactive';
      snapshot.submarine.managedExpedition.currentNodeId=locationNodeId;
      snapshot.operationsDashboard=snapshot.operationsDashboard&&typeof snapshot.operationsDashboard==='object'?snapshot.operationsDashboard:{};
      snapshot.operationsDashboard.activeNodeId=locationNodeId;
      snapshot.storedAtNodeId=locationNodeId;
      snapshot.storedAtAt=now;
    }
    if(typeof worldVesselRegistry==='function'){
      const vessel=worldVesselRegistry()?.[vesselId];
      if(vessel){
        vessel.active=true;
        vessel.status='inactive';
        vessel.ownerGroupId=group?.groupId||vessel.ownerGroupId||'';
        vessel.location={...(vessel.location||{}),mode:'station',stationNodeId:locationNodeId,currentNodeId:locationNodeId,routeId:'',fromNodeId:'',toNodeId:'',progress:0,decisionState:'stored-inactive'};
        vessel.updatedAt=now;
        vessel.notes=`Inactive group submarine stored at ${locationNodeId}.`;
      }
    }
  }

  function dashboardMarkVesselActive(vesselId,group,nodeId,reason='Group active submarine selected'){
    const hub=normalizeWorldHub();
    const now=new Date().toISOString();
    const locationNodeId=String(nodeId||group.currentNodeId||group.startingNodeId||'');
    const record=hub.publicState.submarines[vesselId]||{submarineId:vesselId};
    record.groupId=group.groupId;
    record.status='active';
    record.activeForGroup=true;
    record.currentNodeId=locationNodeId;
    record.storedAtNodeId='';
    record.activatedAt=now;
    record.updatedAt=now;
    if(record.submarine?.managedExpedition){record.submarine.managedExpedition.status='active';record.submarine.managedExpedition.currentNodeId=locationNodeId;}
    hub.publicState.submarines[vesselId]=record;
    const snapshot=normalizeDashboardManagementAuthority().vesselSnapshots?.[vesselId];
    if(snapshot?.submarine){
      snapshot.submarine.managedExpedition=snapshot.submarine.managedExpedition&&typeof snapshot.submarine.managedExpedition==='object'?snapshot.submarine.managedExpedition:{};
      snapshot.submarine.managedExpedition.status='active';
      snapshot.submarine.managedExpedition.currentNodeId=locationNodeId;
    }
    group.primarySubmarineId=vesselId;
    group.activeSubmarineId=vesselId;
    group.submarineIds=[...new Set([...dashboardGroupFleetIds(group),vesselId])];
    hub.activeGroupId=group.groupId;
    hub.activeSubmarineId=vesselId;
    if(typeof worldVesselRegistry==='function'){
      const vessel=worldVesselRegistry()?.[vesselId];
      if(vessel){
        vessel.active=true;
        vessel.status='docked';
        vessel.ownerGroupId=group.groupId;
        vessel.location={...(vessel.location||{}),mode:'station',stationNodeId:locationNodeId,currentNodeId:locationNodeId,routeId:'',fromNodeId:'',toNodeId:'',progress:0,decisionState:'active-at-station'};
        vessel.updatedAt=now;
        vessel.notes=reason;
      }
    }
  }

  function dashboardPrepareGroupVesselActivation(vesselId){
    const group=dashboardManagementGroupForVessel(vesselId);
    if(!group)throw new Error('This submarine is not assigned to an expedition group.');
    const activeId=dashboardGroupActiveVesselId(group);
    if(activeId&&activeId!==vesselId){
      if(dashboardVesselCrossingStatus(activeId)==='underway')throw new Error('The group cannot change active submarines while its current submarine is In Route. Complete or abort the crossing at a station first.');
      const currentNodeId=String(group.currentNodeId||group.startingNodeId||'');
      const currentNode=state.world?.map?.nodes?.find(node=>node.id===currentNodeId);
      if(currentNode?.type!=='station')throw new Error('The group can change active submarines only while physically docked at a station.');
      const selectedLocation=dashboardVesselStationNodeId(vesselId,group);
      const selectedRecord=normalizeWorldHub().publicState.submarines?.[vesselId]||{};
      if(selectedRecord.status==='inactive'&&selectedLocation&&selectedLocation!==currentNodeId)throw new Error(`This inactive submarine is stored at ${state.world?.map?.nodes?.find(node=>node.id===selectedLocation)?.name||selectedLocation}, not at the group’s current station.`);
    }
    return {group,activeId};
  }

  function dashboardFinalizeGroupVesselActivation(vesselId,group,previousActiveId){
    const nodeId=String(group.currentNodeId||group.startingNodeId||'');
    if(previousActiveId&&previousActiveId!==vesselId)dashboardMarkVesselInactive(previousActiveId,group,nodeId,`Stored when ${vesselId} became active`);
    for(const otherId of dashboardGroupFleetIds(group)){
      if(otherId===vesselId||otherId===previousActiveId)continue;
      const record=normalizeWorldHub().publicState.submarines?.[otherId];
      if(record?.status==='active')dashboardMarkVesselInactive(otherId,group,record.currentNodeId||nodeId,'Normalized to enforce one active submarine per group');
    }
    dashboardMarkVesselActive(vesselId,group,nodeId,'Selected as the group’s sole active submarine');
  }

  function dashboardNormalizeAllGroupFleets(){
    const hub=normalizeWorldHub();
    for(const group of Object.values(hub.publicState.groups||{})){
      const ids=dashboardGroupFleetIds(group);
      if(!ids.length){group.primarySubmarineId='';group.activeSubmarineId='';continue;}
      const activeId=dashboardGroupActiveVesselId(group);
      group.primarySubmarineId=activeId;
      group.activeSubmarineId=activeId;
      for(const vesselId of ids){
        if(vesselId===activeId)dashboardMarkVesselActive(vesselId,group,dashboardVesselStationNodeId(vesselId,group)||group.currentNodeId,'Normalized group active vessel');
        else if(hub.publicState.submarines?.[vesselId]?.status==='active')dashboardMarkVesselInactive(vesselId,group,dashboardVesselStationNodeId(vesselId,group)||group.currentNodeId,'Normalized duplicate active vessel');
      }
    }
  }

  const activateDashboardManagedVesselBeforeFleetRules=activateDashboardManagedVessel;
  activateDashboardManagedVessel=function activateDashboardManagedVesselWithFleetRules(vesselId){
    const preparation=dashboardPrepareGroupVesselActivation(vesselId);
    const result=activateDashboardManagedVesselBeforeFleetRules(vesselId);
    dashboardFinalizeGroupVesselActivation(vesselId,preparation.group,preparation.activeId);
    return result;
  };

  const syncAllWorldPlayerVesselsBeforeFleetRules=typeof syncAllWorldPlayerVessels==='function'?syncAllWorldPlayerVessels:null;
  if(syncAllWorldPlayerVesselsBeforeFleetRules){
    syncAllWorldPlayerVessels=function syncAllWorldPlayerVesselsWithFleetRules(){
      const registry=syncAllWorldPlayerVesselsBeforeFleetRules();
      const hub=normalizeWorldHub();
      for(const group of Object.values(hub.publicState.groups||{})){
        const activeId=dashboardGroupActiveVesselId(group);
        for(const vesselId of dashboardGroupFleetIds(group)){
          if(vesselId===activeId)continue;
          const record=hub.publicState.submarines?.[vesselId]||{};
          dashboardMarkVesselInactive(vesselId,group,record.storedAtNodeId||record.currentNodeId||group.currentNodeId,'Preserved inactive vessel during world-registry synchronization');
        }
      }
      return registry;
    };
  }

  const captureDashboardManagedVesselStateBeforeFleetRules=captureDashboardManagedVesselState;

    if(group&&dashboardGroupActiveVesselId(group)===vesselId)dashboardMarkVesselActive(vesselId,group,group.currentNodeId,'Captured active managed vessel state');
    return snapshot;
  };

    if(typeof normalizeCharacterInventory==='function')normalizeCharacterInventory();

        delete authority.submarineCredentials[vesselId];
        delete hub.publicState.submarines[vesselId];
        if(group)group.submarineIds=(group.submarineIds||[]).filter(id=>id!==vesselId);
      }
      if(createdGroup&&group){delete hub.publicState.groups[group.groupId];delete hub.publicState.locations[group.groupId];}
      save();
      throw error;
    }
  };
