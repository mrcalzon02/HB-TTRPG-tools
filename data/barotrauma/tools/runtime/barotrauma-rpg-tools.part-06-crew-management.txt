
  const CREW_DUTIES = ['Unassigned','Command','Navigation','Engineering','Reactor','Mechanical','Medical','Security','Weapons','Sonar','Logistics','Diving / Salvage','Research','Passenger'];
  const CREW_SHIFTS = ['Unassigned','Alpha','Beta','Gamma','Day','Night','Relief'];
  const CREW_STATUSES = ['Onboard','Reserve','Away','Injured','Missing','Dead'];
  const CREW_APPROVALS = ['Pending','Accepted','Rejected'];
  const PARTICIPANT_ROLES = ['Player','Co-GM','Observer','NPC'];

  function defaultPlayerSubmissionSettings() {
    return {
      playerId: uid('player'),
      characterId: uid('character'),
      contact: '',
      availability: '',
      preferredDuty: 'Unassigned',
      preferredShift: 'Unassigned',
      participantRole: 'Player',
      includeInventory: true,
      notes: '',
      lastExportedAt: ''
    };
  }

  function normalizePlayerSubmissionSettings() {
    const existing = state.character.playerSubmission && typeof state.character.playerSubmission === 'object'
      ? state.character.playerSubmission
      : {};
    state.character.playerSubmission = deepMerge(defaultPlayerSubmissionSettings(), existing);
    if (!state.character.playerSubmission.playerId) state.character.playerSubmission.playerId = uid('player');
    if (!state.character.playerSubmission.characterId) state.character.playerSubmission.characterId = uid('character');
    return state.character.playerSubmission;
  }

  function defaultCrewManagement() {
    return {
      roster: [],
      activeCrewId: 'local',
      autoSyncCrewResource: false,
      vesselNotes: '',
      pasteBuffer: '',
      draft: {
        playerName: '',
        characterName: '',
        participantRole: 'NPC',
        className: 'Engineer',
        level: 1,
        duty: 'Unassigned',
        shift: 'Unassigned'
      }
    };
  }

  function normalizeCrewRecord(raw = {}) {
    const stats = raw.stats && typeof raw.stats === 'object' ? raw.stats : { Strength: 3, Agility: 3, Intelligence: 3, Charisma: 3 };
    const current = raw.current && typeof raw.current === 'object' ? raw.current : { ...stats };
    return {
      rosterId: raw.rosterId || uid('crew'),
      sourcePlayerId: raw.sourcePlayerId || '',
      sourceCharacterId: raw.sourceCharacterId || '',
      source: raw.source || 'manual',
      playerName: raw.playerName || '',
      contact: raw.contact || '',
      availability: raw.availability || '',
      participantRole: PARTICIPANT_ROLES.includes(raw.participantRole) ? raw.participantRole : (raw.playerName ? 'Player' : 'NPC'),
      characterName: raw.characterName || 'Unnamed Crew Member',
      className: raw.className || 'Engineer',
      level: clamp(raw.level, 1, 99),
      background: raw.background || '',
      talent: raw.talent || '',
      duty: CREW_DUTIES.includes(raw.duty) ? raw.duty : 'Unassigned',
      preferredDuty: CREW_DUTIES.includes(raw.preferredDuty) ? raw.preferredDuty : 'Unassigned',
      shift: CREW_SHIFTS.includes(raw.shift) ? raw.shift : 'Unassigned',
      preferredShift: CREW_SHIFTS.includes(raw.preferredShift) ? raw.preferredShift : 'Unassigned',
      status: CREW_STATUSES.includes(raw.status) ? raw.status : 'Onboard',
      approval: CREW_APPROVALS.includes(raw.approval) ? raw.approval : (raw.source === 'submission' ? 'Pending' : 'Accepted'),
      active: raw.active !== false,
      stats: {
        Strength: number(stats.Strength, 3),
        Agility: number(stats.Agility, 3),
        Intelligence: number(stats.Intelligence, 3),
        Charisma: number(stats.Charisma, 3)
      },
      current: {
        Strength: number(current.Strength, number(stats.Strength, 3)),
        Agility: number(current.Agility, number(stats.Agility, 3)),
        Intelligence: number(current.Intelligence, number(stats.Intelligence, 3)),
        Charisma: number(current.Charisma, number(stats.Charisma, 3))
      },
      lostSanity: Math.max(0, number(raw.lostSanity)),
      morale: number(raw.morale),
      conditions: raw.conditions || '',
      inventory: Array.isArray(raw.inventory) ? raw.inventory : [],
      inventoryText: raw.inventoryText || '',
      characterNotes: raw.characterNotes || '',
      playerNotes: raw.playerNotes || '',
      gmNotes: raw.gmNotes || '',
      submittedAt: raw.submittedAt || '',
      updatedAt: raw.updatedAt || raw.submittedAt || new Date().toISOString()
    };
  }

  function normalizeCrewManagement() {
    const existing = state.submarine.crewManagement && typeof state.submarine.crewManagement === 'object'
      ? state.submarine.crewManagement
      : {};
    state.submarine.crewManagement = deepMerge(defaultCrewManagement(), existing);
    state.submarine.crewManagement.roster = Array.isArray(state.submarine.crewManagement.roster)
      ? state.submarine.crewManagement.roster.map(normalizeCrewRecord)
      : [];
    if (state.submarine.crewManagement.activeCrewId !== 'local' && !state.submarine.crewManagement.roster.some(item => item.rosterId === state.submarine.crewManagement.activeCrewId)) {
      state.submarine.crewManagement.activeCrewId = 'local';
    }
    return state.submarine.crewManagement;
  }

  normalizePlayerSubmissionSettings();

  function cloneJson(value) {
    return JSON.parse(JSON.stringify(value));
  }

  function inventorySubmissionSummary(entries) {
    return (entries || []).map(entry => {
      const item = inventoryDefinition(entry);
      const counter = entry.tracker === 'ammo'
        ? `${entry.currentAmmo}/${entry.magazineCapacity} ${entry.ammoName || 'ammo'}`
        : entry.tracker === 'uses'
          ? `${entry.usesRemaining}/${entry.maxUses} uses`
          : entry.tracker === 'condition'
            ? `${entry.condition}% condition`
            : '';
      return `${item.name} x${Math.max(1, number(entry.quantity, 1))}${counter ? ` (${counter})` : ''}`;
    }).join('; ');
  }

  function buildCrewSubmission() {
    normalizeCharacterInventory();
    const settings = normalizePlayerSubmissionSettings();
    const submittedAt = new Date().toISOString();
    const inventory = settings.includeInventory ? cloneJson(state.character.catalogInventory) : [];
    return {
      schema: 'barotrauma-crew-submission',
      version: 1,
      submittedAt,
      player: {
        id: settings.playerId,
        name: state.character.player || '',
        contact: settings.contact || '',
        availability: settings.availability || '',
        participantRole: settings.participantRole || 'Player',
        notes: settings.notes || ''
      },
      character: {
        id: settings.characterId,
        name: state.character.name || '',
        className: state.character.className,
        level: state.character.level,
        background: state.character.background,
        talent: state.character.talent,
        fateUsed: state.character.fateUsed,
        lostSanity: state.character.lostSanity,
        morale: state.character.morale,
        stats: cloneJson(state.character.stats),
        current: cloneJson(state.character.current),
        conditions: state.character.conditions || '',
        inventory,
        inventoryText: state.character.inventory || '',
        inventorySummary: inventorySubmissionSummary(inventory),
        notes: state.character.notes || ''
      },
      preferences: {
        duty: settings.preferredDuty || 'Unassigned',
        shift: settings.preferredShift || 'Unassigned'
      }
    };
  }

  function submissionFileName(submission) {
    const base = `${submission.player?.name || 'player'}-${submission.character?.name || 'character'}`.toLowerCase().replace(/[^a-z0-9]+/g, '-').replace(/^-|-$/g, '') || 'crew-submission';
    return `${base}-crew-submission.json`;
  }

  function copyText(text) {
    if (navigator.clipboard?.writeText) return navigator.clipboard.writeText(text);
    const area = document.createElement('textarea');
    area.value = text;
    area.style.position = 'fixed';
    area.style.opacity = '0';
    document.body.appendChild(area);
    area.select();
    document.execCommand('copy');
    area.remove();
    return Promise.resolve();
  }

  function crewRecordFromSubmission(submission) {
    if (!submission || submission.schema !== 'barotrauma-crew-submission' || !submission.character) throw new Error('This is not a Barotrauma crew submission file.');
    const player = submission.player || {};
    const character = submission.character || {};
    const preferences = submission.preferences || {};
    return normalizeCrewRecord({
      sourcePlayerId: player.id || '',
      sourceCharacterId: character.id || '',
      source: 'submission',
      playerName: player.name || '',
      contact: player.contact || '',
      availability: player.availability || '',
      participantRole: player.participantRole || 'Player',
      characterName: character.name || 'Unnamed Crew Member',
      className: character.className || 'Engineer',
      level: character.level || 1,
      background: character.background || '',
      talent: character.talent || '',
      duty: preferences.duty || 'Unassigned',
      preferredDuty: preferences.duty || 'Unassigned',
      shift: preferences.shift || 'Unassigned',
      preferredShift: preferences.shift || 'Unassigned',
      status: 'Onboard',
      approval: 'Pending',
      active: true,
      stats: character.stats,
      current: character.current,
      lostSanity: character.lostSanity,
      morale: character.morale,
      conditions: character.conditions || '',
      inventory: character.inventory || [],
      inventoryText: character.inventoryText || '',
      characterNotes: character.notes || '',
      playerNotes: player.notes || '',
      submittedAt: submission.submittedAt || new Date().toISOString(),
      updatedAt: submission.submittedAt || new Date().toISOString()
    });
  }

  function crewIdentityMatch(record, incoming) {
    if (incoming.sourceCharacterId && record.sourceCharacterId === incoming.sourceCharacterId) return true;
    if (incoming.sourcePlayerId && record.sourcePlayerId === incoming.sourcePlayerId && record.characterName.toLowerCase() === incoming.characterName.toLowerCase()) return true;
    return false;
  }

  function mergeCrewRecord(incomingRaw) {
    const management = normalizeCrewManagement();
    const incoming = normalizeCrewRecord(incomingRaw);
    const existing = management.roster.find(record => crewIdentityMatch(record, incoming) || (incoming.rosterId && record.rosterId === incoming.rosterId));
    if (!existing) {
      management.roster.push(incoming);
      syncCrewResourceIfEnabled();
      return incoming;
    }
    const preserved = {
      rosterId: existing.rosterId,
      duty: existing.duty,
      shift: existing.shift,
      status: existing.status,
      approval: existing.approval,
      active: existing.active,
      gmNotes: existing.gmNotes
    };
    Object.assign(existing, incoming, preserved, { updatedAt: incoming.updatedAt || new Date().toISOString() });
    syncCrewResourceIfEnabled();
    return existing;
  }

  function mergeCrewSubmission(submission) {
    return mergeCrewRecord(crewRecordFromSubmission(submission));
  }

  function importedCrewPayloads(parsed) {
    if (parsed?.schema === 'barotrauma-crew-submission') return [{ kind: 'submission', value: parsed }];
    if (parsed?.schema === 'barotrauma-crew-roster' && Array.isArray(parsed.roster)) return parsed.roster.map(value => ({ kind: 'record', value }));
    if (Array.isArray(parsed?.submissions)) return parsed.submissions.map(value => ({ kind: 'submission', value }));
    if (Array.isArray(parsed?.roster)) return parsed.roster.map(value => ({ kind: 'record', value }));
    if (Array.isArray(parsed)) return parsed.map(value => value?.schema === 'barotrauma-crew-submission' ? ({ kind: 'submission', value }) : ({ kind: 'record', value }));
    return [];
  }

  function importCrewPayload(parsed) {
    const payloads = importedCrewPayloads(parsed);
    if (!payloads.length) throw new Error('No crew submissions or roster records were found.');
    for (const payload of payloads) {
      if (payload.kind === 'submission') mergeCrewSubmission(payload.value);
      else mergeCrewRecord(payload.value);
    }
    save();
    return payloads.length;
  }

  function acceptedOnboardCrew() {
    return normalizeCrewManagement().roster.filter(record => record.active && record.approval === 'Accepted' && record.status === 'Onboard');
  }

  function syncCrewResourceIfEnabled() {
    const management = normalizeCrewManagement();
    if (management.autoSyncCrewResource) state.submarine.resources.crew = acceptedOnboardCrew().length;
  }

  function crewRollOperator() {
    const management = normalizeCrewManagement();
    if (management.activeCrewId === 'local') return {
      rosterId: 'local',
      characterName: state.character.name || 'Current Character',
      playerName: state.character.player || '',
      current: state.character.current,
      status: 'Local record'
    };
    return management.roster.find(record => record.rosterId === management.activeCrewId) || {
      rosterId: 'local', characterName: state.character.name || 'Current Character', playerName: state.character.player || '', current: state.character.current, status: 'Local record'
    };
  }

  }

  function crewInventorySummary(record) {

  }

  function crewRecommendedRange() {
    const model = SUBMARINES[state.submarine.model] || SUBMARINES.Barsuk;
    const profile = model.profile || {};
    return {
      min: Math.max(0, number(profile.crewMin, 0)),
      max: Math.max(0, number(profile.crewMax, 0))
    };
  }

  function patchUpkeepRollOperators() {
    const operator = crewRollOperator();
    const resultBox = root.querySelector('#upkeep-result');
    if (resultBox && !resultBox.dataset.crewPatched) {
      resultBox.dataset.crewPatched = 'true';
      resultBox.innerHTML = `Upkeep rolls use <strong>${escapeHtml(operator.characterName)}</strong>${operator.playerName ? ` (${escapeHtml(operator.playerName)})` : ''}. Select another roll operator in Crew Management.`;
    }
    root.querySelectorAll('.upkeep-roll').forEach(original => {
      const replacement = original.cloneNode(true);
      original.replaceWith(replacement);
      replacement.addEventListener('click', event => {
        const selected = crewRollOperator();
        const stat = event.target.dataset.stat;
        const dice = [d6(), d6()];
        const modifier = number(selected.current?.[stat]);
        const total = dice[0] + dice[1] + modifier;
        const success = total >= 8;
        const task = event.target.dataset.task;
        state.submarine.upkeep[task] = { complete: success, roll: total, operator: selected.characterName };
        save();
        const output = root.querySelector('#upkeep-result');
        if (output) output.innerHTML = `<strong class="${success ? 'good' : 'bad'}">${success ? 'UPKEEP COMPLETE' : 'UPKEEP FAILED'}</strong><br>${escapeHtml(selected.characterName)}: ${dice[0]} + ${dice[1]} + ${modifier} ${escapeHtml(stat)} = ${total} vs 8.`;
        const checkbox = root.querySelector(`.upkeep-check[data-task="${task}"]`);
        if (checkbox) checkbox.checked = success;
      });
    });
  }

          <td><div class="button-row"><button class="compact-button crew-set-operator" data-id="${escapeHtml(record.rosterId)}" type="button">Use for Rolls</button><button class="compact-button crew-export-one" data-id="${escapeHtml(record.rosterId)}" type="button">Export</button><button class="compact-button crew-remove" data-id="${escapeHtml(record.rosterId)}" type="button">Remove</button></div></td>

    const updateRecord = (id, callback, rerender = false) => {
      const record = management.roster.find(item => item.rosterId === id);
      if (!record) return;
      callback(record);
      record.updatedAt = new Date().toISOString();
      syncCrewResourceIfEnabled();
      save();
      if (rerender) renderSubmarine();
    };

    panel.querySelector('#crew-auto-sync')?.addEventListener('change', event => { management.autoSyncCrewResource = event.target.checked; syncCrewResourceIfEnabled(); save(); renderSubmarine(); });
    panel.querySelector('#crew-sync-now')?.addEventListener('click', () => { state.submarine.resources.crew = onboard.length; save(); renderSubmarine(); });
    panel.querySelector('#crew-export-roster')?.addEventListener('click', () => download(`${(state.submarine.name || state.submarine.model || 'submarine').toLowerCase().replace(/[^a-z0-9]+/g, '-')}-crew-roster.json`, JSON.stringify({ schema: 'barotrauma-crew-roster', version: 1, exportedAt: new Date().toISOString(), vessel: { name: state.submarine.name, model: state.submarine.model }, roster: management.roster }, null, 2)));
    panel.querySelector('#crew-add-current')?.addEventListener('click', () => { const record = mergeCrewSubmission(buildCrewSubmission()); record.approval = 'Accepted'; syncCrewResourceIfEnabled(); save(); renderSubmarine(); });
    panel.querySelector('#crew-import-files')?.addEventListener('change', async event => {
      const files = [...(event.target.files || [])];
      let imported = 0;
      const errors = [];
      for (const file of files) {
        try { imported += importCrewPayload(JSON.parse(await file.text())); }
        catch (error) { errors.push(`${file.name}: ${error.message}`); }
      }
      event.target.value = '';
      save(); renderSubmarine();
      if (errors.length) alert(`Imported ${imported} record(s). Some files failed:\n${errors.join('\n')}`);
    });
    panel.querySelector('#crew-import-paste')?.addEventListener('click', () => {
      try { importCrewPayload(JSON.parse(panel.querySelector('#crew-paste-json').value)); management.pasteBuffer = ''; save(); renderSubmarine(); }
      catch (error) { alert(`Crew data could not be imported: ${error.message}`); }
    });
    panel.querySelector('#crew-paste-json')?.addEventListener('input', event => { management.pasteBuffer = event.target.value; save(); });
    panel.querySelector('#crew-vessel-notes')?.addEventListener('input', event => { management.vesselNotes = event.target.value; save(); });
    for (const [selector, key, numeric] of [['.crew-player-name','playerName'],['.crew-contact','contact'],['.crew-character-name','characterName'],['.crew-level','level',true],['.crew-gm-notes','gmNotes']]) {
      panel.querySelectorAll(selector).forEach(input => input.addEventListener(numeric ? 'change' : 'input', event => updateRecord(event.target.dataset.id, record => { record[key] = numeric ? clamp(event.target.value, 1, 99) : event.target.value; })));
    }
    for (const [selector, key] of [['.crew-participant-role','participantRole'],['.crew-class','className'],['.crew-duty','duty'],['.crew-shift','shift'],['.crew-approval','approval'],['.crew-status','status']]) {
      panel.querySelectorAll(selector).forEach(select => select.addEventListener('change', event => updateRecord(event.target.dataset.id, record => { record[key] = event.target.value; }, ['approval','status'].includes(key))));
    }
    panel.querySelectorAll('.crew-active').forEach(input => input.addEventListener('change', event => updateRecord(event.target.dataset.id, record => { record.active = event.target.checked; }, true)));
    panel.querySelectorAll('.crew-set-operator').forEach(button => button.addEventListener('click', event => { management.activeCrewId = event.target.dataset.id; save(); renderSubmarine(); }));
    panel.querySelectorAll('.crew-export-one').forEach(button => button.addEventListener('click', event => { const record = management.roster.find(item => item.rosterId === event.target.dataset.id); if (record) download(`${record.characterName.toLowerCase().replace(/[^a-z0-9]+/g, '-') || 'crew-member'}-roster-record.json`, JSON.stringify({ schema: 'barotrauma-crew-roster', version: 1, roster: [record] }, null, 2)); }));
    panel.querySelectorAll('.crew-remove').forEach(button => button.addEventListener('click', event => { management.roster = management.roster.filter(record => record.rosterId !== event.target.dataset.id); if (management.activeCrewId === event.target.dataset.id) management.activeCrewId = 'local'; syncCrewResourceIfEnabled(); save(); renderSubmarine(); }));
    const draftBindings = [['#crew-draft-player','playerName'],['#crew-draft-character','characterName'],['#crew-draft-role','participantRole'],['#crew-draft-class','className'],['#crew-draft-level','level'],['#crew-draft-duty','duty'],['#crew-draft-shift','shift']];
    for (const [selector, key] of draftBindings) panel.querySelector(selector)?.addEventListener(selector.includes('level') ? 'change' : selector.includes('player') || selector.includes('character') ? 'input' : 'change', event => { management.draft[key] = key === 'level' ? clamp(event.target.value, 1, 99) : event.target.value; save(); });
    panel.querySelector('#crew-add-manual')?.addEventListener('click', () => {
      if (!String(management.draft.characterName || '').trim()) return alert('Enter a character or NPC name.');
      management.roster.push(normalizeCrewRecord({ ...management.draft, source: 'manual', approval: 'Accepted', status: 'Onboard', active: true }));
      management.draft = defaultCrewManagement().draft;
      syncCrewResourceIfEnabled(); save(); renderSubmarine();
    });
    patchUpkeepRollOperators();
  }
