  function dashboardManagementMessage(message,isError=false){
    dashboardManagementRuntime.message=String(message||'');
    const output=root.querySelector('#dashboard-management-message');
    if(output){output.textContent=dashboardManagementRuntime.message;output.classList.toggle('bad',isError);output.classList.toggle('good',!isError);}
  }

  function bindDashboardManagementGate(){
    const selector=root.querySelector('#dashboard-management-vessel');
    selector?.addEventListener('change',event=>{dashboardManagementSetSelectedVesselId(event.target.value);dashboardManagementRuntime.message='Enter the selected submarine’s management password.';renderActiveSubmarineDashboard();});
    root.querySelector('#dashboard-management-register-current')?.addEventListener('click',()=>{try{registerCurrentSubmarineForDashboardManagement();dashboardManagementRuntime.message='Current submarine registered. Set its management password.';renderActiveSubmarineDashboard();}catch(error){dashboardManagementMessage(error.message,true);}});
    root.querySelector('#dashboard-management-login')?.addEventListener('click',async event=>{
      const button=event.currentTarget;
      const password=root.querySelector('#dashboard-management-password');
      button.disabled=true;
      dashboardManagementMessage('Verifying password…');
      try{
        const vesselId=root.querySelector('#dashboard-management-vessel')?.value||'';
        await authenticateDashboardManagementPassword(vesselId,password?.value||'',Boolean(root.querySelector('#dashboard-management-remember')?.checked));
        if(password)password.value='';
        renderActiveSubmarineDashboard();
      }catch(error){if(password)password.value='';dashboardManagementMessage(error.message,true);button.disabled=false;}
    });
    root.querySelector('#dashboard-management-claim')?.addEventListener('click',async event=>{
      const button=event.currentTarget;
      const password=root.querySelector('#dashboard-management-new-password');
      const confirmation=root.querySelector('#dashboard-management-confirm-password');
      const value=password?.value||'';
      if(value!==(confirmation?.value||'')){dashboardManagementMessage('The new password and confirmation do not match.',true);return;}
      button.disabled=true;
      dashboardManagementMessage('Creating the submarine credential…');
      try{
        const vesselId=root.querySelector('#dashboard-management-vessel')?.value||'';
        await createDashboardManagementCredential(vesselId,value,root.querySelector('#dashboard-management-manager-name')?.value||'',Boolean(root.querySelector('#dashboard-management-remember')?.checked));
        if(password)password.value='';if(confirmation)confirmation.value='';
        renderActiveSubmarineDashboard();
      }catch(error){if(password)password.value='';if(confirmation)confirmation.value='';dashboardManagementMessage(error.message,true);button.disabled=false;}
    });
  }

  function bindDashboardManagementSessionPanel(){
    root.querySelector('#dashboard-management-logout')?.addEventListener('click',()=>{logoutDashboardManagementSession('manual dashboard logout');renderActiveSubmarineDashboard();});
    root.querySelector('#dashboard-management-change')?.addEventListener('click',async event=>{
      const button=event.currentTarget;
      const current=root.querySelector('#dashboard-management-current-password');
      const next=root.querySelector('#dashboard-management-change-password');
      const confirm=root.querySelector('#dashboard-management-change-confirm');
      const output=root.querySelector('#dashboard-management-session-message');
      if((next?.value||'')!==(confirm?.value||'')){if(output){output.textContent='The new passwords do not match.';output.classList.add('bad');}return;}
      button.disabled=true;
      if(output){output.textContent='Changing password and revoking older receipts…';output.classList.remove('bad');}
      try{
        await changeDashboardManagementPassword(current?.value||'',next?.value||'');
        if(current)current.value='';if(next)next.value='';if(confirm)confirm.value='';
        renderActiveSubmarineDashboard();
      }catch(error){if(current)current.value='';if(next)next.value='';if(confirm)confirm.value='';if(output){output.textContent=error.message;output.classList.add('bad');}button.disabled=false;}
    });
  }

  function enforceDashboardManagementSessionBoundary(){
    const groupSelector=root.querySelector('#dashboard-active-group');
    if(groupSelector){groupSelector.disabled=true;groupSelector.title='The active expedition group is fixed by the authenticated submarine session.';}
    const createGroup=root.querySelector('#dashboard-create-group');
    if(createGroup){createGroup.disabled=true;createGroup.title='Create expedition groups from the standalone World tool, then assign submarines before logging in.';}
    const syncButton=root.querySelector('#dashboard-force-sync');
    if(syncButton)syncButton.title='Synchronize the currently authenticated vessel only.';
  }

  const renderActiveSubmarineDashboardBeforeManagementAuth=renderActiveSubmarineDashboard;
  renderActiveSubmarineDashboard=function renderActiveSubmarineDashboardWithManagementAuth(requestedFocus=activeDashboardRawFocus()){
    if(dashboardManagementRuntime.status!=='authenticated'){
      root.innerHTML=dashboardManagementGateHtml();
      bindDashboardManagementGate();
      return;
    }
    if(!dashboardManagementAuthorized(dashboardManagementRuntime.vesselId)){
      logoutDashboardManagementSession('expired or invalid session');
      root.innerHTML=dashboardManagementGateHtml();
      bindDashboardManagementGate();
      return;
    }
    if(state.submarine?.worldSubmarineId!==dashboardManagementRuntime.vesselId)activateDashboardManagedVessel(dashboardManagementRuntime.vesselId);
    const result=renderActiveSubmarineDashboardBeforeManagementAuth(requestedFocus);
    const command=root.querySelector('#active-dashboard-command');
    command?.insertAdjacentHTML('beforebegin',dashboardManagementSessionPanelHtml());
    bindDashboardManagementSessionPanel();
    enforceDashboardManagementSessionBoundary();
    return result;
  };
