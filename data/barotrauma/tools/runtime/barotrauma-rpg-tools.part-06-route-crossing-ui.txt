
  function crossingStatusLabel(status) {
    return ({ none:'No crossing', prepared:'Prepared', underway:'Underway', arrived:'Arrived', aborted:'Aborted' })[status] || status;
  }

  function crossingStatusTone(status) {
    if (status === 'arrived') return 'good';
    if (status === 'underway') return 'good';
    if (status === 'prepared') return 'warn';
    if (status === 'aborted') return 'bad';
    return '';
  }

  function crossingNodeName(nodeId) {
    return crossingNode(nodeId)?.name || nodeId || 'Unknown location';
  }

  function crossingCostText(costs = {}) {
    return `Fuel ${number(costs.fuel)} · Food ${number(costs.food)} · Water ${number(costs.water)} · Supplies ${number(costs.supplies)}`;
  }

  function crossingProgress(crossing = normalizeRouteCrossing()) {
    const totalLegs = crossing.legs.length;
    const completedLegs = crossing.legs.filter(leg => leg.status === 'complete').length;
    const totalEvents = state.route.plan.length;
    const resolvedEvents = state.route.plan.filter(slot => slot.status !== 'pending').length;
    return {
      totalLegs,
      completedLegs,
      totalEvents,
      resolvedEvents,
      legPercent: totalLegs ? Math.round(completedLegs / totalLegs * 100) : 0,
      eventPercent: totalEvents ? Math.round(resolvedEvents / totalEvents * 100) : 0
    };
  }

  function crossingEventDetails(event) {
    if (!event) return '';
    const location = event.locationTypeId ? `<div class="notice location-encounter-detail"><strong>Level ${number(event.locationLevel)}: ${escapeHtml(event.locationLevelName || '')} — ${escapeHtml(event.locationTypeName || '')}</strong><br>${escapeHtml(event.locationLore || '')}<br><strong>Availability:</strong> ${escapeHtml(event.locationAvailability || 'None')}<br><strong>Shops:</strong> ${escapeHtml((event.locationShops || []).join(', ') || 'None')}<br><strong>Services:</strong> ${escapeHtml((event.locationServices || []).join(', ') || 'None')}<br><strong>Hazards:</strong> ${escapeHtml((event.locationHazards || []).join(', ') || 'Unknown')}</div>` : '';
    return `${location}<p>${escapeHtml(event.description || '')}</p><p><strong>Reward:</strong> ${escapeHtml(event.reward || 'GM determination')}<br><strong>Failure:</strong> ${escapeHtml(event.failure || 'GM determination')}</p><span class="tag">${escapeHtml(event.category || 'Event')}</span><span class="tag">${escapeHtml(event.stat || 'Intelligence')} vs ${number(event.dc, 8)}</span>`;
  }

  function routeCrossingLegRows(crossing) {
    return crossing.legs.map(leg => {
      const eventSlots = state.route.plan.filter(slot => leg.eventIds.includes(slot.id));
      const resolved = eventSlots.filter(slot => slot.status !== 'pending').length;
      const check = crossingLegCanComplete(leg, crossing);
      const active = crossing.status === 'underway' && leg.index === crossing.activeLegIndex;
      return `<tr style="${active ? 'background:rgba(200,138,53,.09)' : ''}"><td>${leg.index + 1}</td><td><strong>${escapeHtml(crossingNodeName(leg.fromNodeId))}</strong><br>→ ${escapeHtml(crossingNodeName(leg.toNodeId))}</td><td>Level ${leg.fromLevel} → ${leg.toLevel}</td><td>${leg.distance}${leg.danger ? ' · <span class="bad">Dangerous</span>' : ' · Standard'}</td><td>${escapeHtml(crossingCostText(leg.costs))}</td><td>${resolved}/${eventSlots.length}</td><td><span class="status-pill status-${leg.status === 'complete' ? 'success' : leg.status === 'underway' ? 'pending' : 'pending'}">${escapeHtml(leg.status)}</span></td><td>${active ? `<button class="compact-button complete-crossing-leg no-print" type="button" ${check.ready ? '' : 'disabled'}>Complete Leg</button><div class="small-note">${escapeHtml(check.issue)}</div>` : ''}</td></tr>`;
    }).join('');
  }

      <div class="button-row no-print" style="margin-top:10px"><button id="sub-open-route" class="primary-action" type="button">Open Full Route Tracker</button><button id="sub-start-crossing" class="secondary-action" type="button" ${crossing.status === 'prepared' && readiness.ready ? '' : 'disabled'}>Start Crossing</button><button id="sub-roll-next" class="secondary-action" type="button" ${crossing.status === 'underway' && nextSlot ? '' : 'disabled'}>Roll Next Encounter</button><button id="sub-success-next" class="secondary-action" type="button" ${crossing.status === 'underway' && nextSlot ? '' : 'disabled'}>Next Success</button><button id="sub-failure-next" class="danger-button" type="button" ${crossing.status === 'underway' && nextSlot ? '' : 'disabled'}>Next Failure</button><button id="sub-complete-leg" class="secondary-action" type="button" ${legCheck.ready ? '' : 'disabled'}>Complete Active Leg</button></div>
      <div class="button-row no-print" style="margin-top:8px"><button class="compact-button sub-crossing-effect" data-effect="hull" type="button">Hull −1</button><button class="compact-button sub-crossing-effect" data-effect="repair" type="button">Hull +1</button><button class="compact-button sub-crossing-effect" data-effect="fuel" type="button">Fuel −1</button><button class="compact-button sub-crossing-effect" data-effect="supplies" type="button">Supply −1</button><button class="compact-button sub-crossing-effect" data-effect="food" type="button">Food −1</button><button class="compact-button sub-crossing-effect" data-effect="water" type="button">Water −1</button></div>
      <p class="small-note" style="margin-bottom:0">${escapeHtml(state.route.crossingMessage || '')}</p>
    ` : `<div class="tool-hero" style="padding:0;border:0;background:none"><div><p class="eyebrow">Expedition route tracking</p><h3>No station crossing prepared</h3><p class="small-note">Use the Route Planner to calculate a valid path between mapped stations and generate its encounters.</p></div><button id="sub-open-route" class="primary-action no-print" type="button">Build a Crossing</button></div>`;
    hero.insertAdjacentElement('afterend', panel);
    panel.querySelector('#sub-open-route')?.addEventListener('click', () => setTool('route'));
    panel.querySelector('#sub-start-crossing')?.addEventListener('click', () => { startCrossing(); renderSubmarine(); });
    panel.querySelector('#sub-roll-next')?.addEventListener('click', () => { const slot = nextPendingCrossingSlot(); if (slot) rollCrossingEvent(slot.id); renderSubmarine(); });
    panel.querySelector('#sub-success-next')?.addEventListener('click', () => { const slot = nextPendingCrossingSlot(); if (slot) resolveCrossingEvent(slot.id, 'success'); renderSubmarine(); });
    panel.querySelector('#sub-failure-next')?.addEventListener('click', () => { const slot = nextPendingCrossingSlot(); if (slot) resolveCrossingEvent(slot.id, 'failure'); renderSubmarine(); });
    panel.querySelector('#sub-complete-leg')?.addEventListener('click', () => { completeCrossingLeg(); renderSubmarine(); });
    panel.querySelectorAll('.sub-crossing-effect').forEach(button => button.addEventListener('click', event => { applyCrossingQuickEffect(event.target.dataset.effect); renderSubmarine(); }));
  }
