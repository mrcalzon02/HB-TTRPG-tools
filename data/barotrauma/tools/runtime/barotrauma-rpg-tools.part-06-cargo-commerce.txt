
  function defaultCargoManagement() {
    return {
      entries: [],
      autoSync: false,
      draftName: '',
      draftQuantity: 1,
      draftUnitSize: 1,
      draftOwner: 'Submarine',
      draftLocation: 'Cargo Bay',
      notes: ''
    };
  }

  function normalizeCargoEntry(raw = {}) {
    const item = raw.id ? catalogueItem(raw.id) : null;
    return {
      cargoId: raw.cargoId || uid('cargo'),
      id: raw.id || '',
      customName: raw.customName || item?.name || 'Unnamed Cargo',
      quantity: Math.max(1, number(raw.quantity, 1)),
      unitSize: Math.max(0, number(raw.unitSize, item?.category === 'Submarine Ammunition' ? 1 : item?.carryUnits || 1)),
      owner: raw.owner || 'Submarine',
      location: raw.location || 'Cargo Bay',
      condition: clamp(raw.condition ?? 100, 0, 100),
      notes: raw.notes || '',
      acquiredFrom: raw.acquiredFrom || ''
    };
  }

  function normalizeCargoManagement() {
    const existing = state.submarine.cargoManagement && typeof state.submarine.cargoManagement === 'object'
      ? state.submarine.cargoManagement
      : {};
    state.submarine.cargoManagement = deepMerge(defaultCargoManagement(), existing);
    state.submarine.cargoManagement.entries = Array.isArray(state.submarine.cargoManagement.entries)
      ? state.submarine.cargoManagement.entries.map(normalizeCargoEntry)
      : [];
    return state.submarine.cargoManagement;
  }

  function cargoItem(entry) {
    return entry.id ? catalogueItem(entry.id) : {
      id: '', name: entry.customName || 'Unnamed Cargo', category: 'Cargo', subcategory: 'Manual Entry',
      description: entry.notes || '', carryUnits: number(entry.unitSize, 1)
    };
  }

  function cargoUnits() {
    return normalizeCargoManagement().entries.reduce((sum, entry) => sum + Math.max(0, number(entry.unitSize, 1)) * Math.max(1, number(entry.quantity, 1)), 0);
  }

  function syncCargoResourceIfEnabled() {
    const cargo = normalizeCargoManagement();
    if (cargo.autoSync) state.submarine.resources.cargo = Math.ceil(cargoUnits());
  }

  function addCargoItem(id, quantity = 1, options = {}) {
    const cargo = normalizeCargoManagement();
    const item = id ? catalogueItem(id) : null;
    const name = options.customName || item?.name || 'Unnamed Cargo';
    const existing = cargo.entries.find(entry => entry.id === (id || '') && entry.customName === name && entry.owner === (options.owner || 'Submarine') && entry.location === (options.location || 'Cargo Bay'));
    if (existing) existing.quantity += Math.max(1, number(quantity, 1));
    else cargo.entries.push(normalizeCargoEntry({
      id: id || '', customName: name, quantity,
      unitSize: options.unitSize ?? (item?.category === 'Submarine Ammunition' ? 1 : item?.carryUnits || 1),
      owner: options.owner || 'Submarine', location: options.location || 'Cargo Bay', notes: options.notes || '', acquiredFrom: options.acquiredFrom || ''
    }));
    syncCargoResourceIfEnabled();
    save();
  }

  function removeCargoQuantity(cargoId, quantity = 1) {
    const cargo = normalizeCargoManagement();
    const entry = cargo.entries.find(item => item.cargoId === cargoId);
    if (!entry) return false;
    entry.quantity -= Math.max(1, number(quantity, 1));
    if (entry.quantity <= 0) cargo.entries = cargo.entries.filter(item => item.cargoId !== cargoId);
    syncCargoResourceIfEnabled();
    save();
    return true;
  }

  function moveCharacterInventoryToCargo(instanceId) {
    const entry = inventoryEntry(instanceId);
    if (!entry) return false;
    const item = inventoryDefinition(entry);
    addCargoItem(entry.id, 1, {
      customName: item.name,
      unitSize: item.carryUnits || entry.carryUnits || 1,
      owner: state.character.name || state.character.player || 'Crew',
      location: 'Cargo Bay',
      notes: entry.notes || '',
      acquiredFrom: 'Character inventory'
    });
    if (entry.quantity > 1) entry.quantity -= 1;
    else state.character.catalogInventory = state.character.catalogInventory.filter(itemEntry => itemEntry.instanceId !== instanceId);
    save();
    return true;
  }

  function moveCargoToCharacter(cargoId) {
    const entry = normalizeCargoManagement().entries.find(item => item.cargoId === cargoId);
    if (!entry) return false;
    if (entry.id) addCharacterCatalogueItem(entry.id, 1);
    else state.character.catalogInventory.push(normalizeInventoryEntry({
      customName: entry.customName, customCategory: 'Cargo', customDescription: entry.notes, carryUnits: entry.unitSize,
      quantity: 1, tracker: 'none', notes: `Transferred from ${entry.location}.`
    }));
    removeCargoQuantity(cargoId, 1);
    save();
    return true;
  }

  function stationSlug(value) {
    return String(value || '').toLowerCase().replace(/[^a-z0-9]+/g, '-').replace(/^-|-$/g, '') || 'station';
  }

  function stationIdentity(station) {
    return `${stationSlug(station.name)}--${stationSlug(station.faction || 'independent')}`;
  }

  function defaultStationCommerce() {
    return {
      activeStationId: '',
      wallet: 'submarine',
      destination: 'cargo',
      factionReputation: {},
      stations: {},
      transactionLog: []
    };
  }

  function defaultStationRecord(station) {
    const firstLocal = String(station.locals || '').split(';')[0].trim();
    return {
      id: stationIdentity(station),
      name: station.name,
      faction: station.faction || 'Independent',
      specialty: station.specialty || 'General trade',
      locals: station.locals || '',
      stationReputation: 0,
      ownerName: firstLocal || `${station.name} Quartermaster`,
      ownerReputation: 0,
      playerStation: false,
      stationOwner: '',
      treasury: 2500,
      marketRevision: 0,
      vendors: [],
      notes: ''
    };
  }

  function normalizeStationRecord(raw, station) {
    const base = defaultStationRecord(station || raw || {});
    const record = deepMerge(base, raw || {});
    record.vendors = Array.isArray(record.vendors) ? record.vendors : [];
    record.stationReputation = clamp(record.stationReputation, -100, 100);
    record.ownerReputation = clamp(record.ownerReputation, -100, 100);
    record.treasury = Math.max(0, number(record.treasury, 2500));
    return record;
  }

  function normalizeStationCommerce() {
    const existing = state.stationCommerce && typeof state.stationCommerce === 'object' ? state.stationCommerce : {};
    state.stationCommerce = deepMerge(defaultStationCommerce(), existing);
    state.stationCommerce.factionReputation = state.stationCommerce.factionReputation && typeof state.stationCommerce.factionReputation === 'object' ? state.stationCommerce.factionReputation : {};
    state.stationCommerce.stations = state.stationCommerce.stations && typeof state.stationCommerce.stations === 'object' ? state.stationCommerce.stations : {};
    state.stationCommerce.transactionLog = Array.isArray(state.stationCommerce.transactionLog) ? state.stationCommerce.transactionLog : [];
    syncCustomStations();
    for (const station of STATIONS) {
      const id = stationIdentity(station);
      state.stationCommerce.stations[id] = normalizeStationRecord(state.stationCommerce.stations[id], station);
      if (!(station.faction in state.stationCommerce.factionReputation)) state.stationCommerce.factionReputation[station.faction] = 0;
    }
    const validIds = STATIONS.map(stationIdentity);
    if (!validIds.includes(state.stationCommerce.activeStationId)) {
      const routeStation = STATIONS.find(station => station.name === state.route.end) || STATIONS[0];
      state.stationCommerce.activeStationId = routeStation ? stationIdentity(routeStation) : '';
    }
    return state.stationCommerce;
  }

  function activeStationRecord() {
    const commerce = normalizeStationCommerce();
    return commerce.stations[commerce.activeStationId] || null;
  }

  function seededHash(text) {
    let hash = 2166136261;
    for (const character of String(text)) {
      hash ^= character.charCodeAt(0);
      hash = Math.imul(hash, 16777619);
    }
    return hash >>> 0;
  }

  function seededRandom(seedText) {
    let seed = seededHash(seedText) || 1;
    return () => {
      seed += 0x6D2B79F5;
      let value = seed;
      value = Math.imul(value ^ value >>> 15, value | 1);
      value ^= value + Math.imul(value ^ value >>> 7, value | 61);
      return ((value ^ value >>> 14) >>> 0) / 4294967296;
    };
  }

  function allTradeRecords() {
    const records = [];
    for (const item of WIKI_CATALOG.items) records.push({ key: `item:${item.id}`, kind: 'item', id: item.id, name: item.name, category: item.category, item });
    for (const weapon of WIKI_CATALOG.submarineWeapons) records.push({ key: `submarineWeapon:${weapon.id}`, kind: 'submarineWeapon', id: weapon.id, name: weapon.name, category: 'Submarine Weapon', item: weapon });
    for (const upgrade of UPGRADES) records.push({ key: `upgrade:${upgrade.id}`, kind: 'upgrade', id: upgrade.id, name: upgrade.name, category: 'Upgrade', item: upgrade });
    for (const entry of state.customContent.entries) {
      const derived = derivedCustomEntry(entry);
      if (['handheld-weapon','personal-ammunition','submarine-ammunition','equipment','general-item','consumable'].includes(entry.type)) {
        const item = customEntryAsItem(entry);
        records.push({ key: `item:${entry.id}`, kind: 'item', id: entry.id, name: entry.name, category: item.category, item, customEntry: entry, derived });
      } else if (entry.type === 'submarine-weapon') {
        records.push({ key: `submarineWeapon:${entry.id}`, kind: 'submarineWeapon', id: entry.id, name: entry.name, category: 'Submarine Weapon', item: customEntryAsWeapon(entry), customEntry: entry, derived });
      } else if (['submarine-system','power-source','submarine-module'].includes(entry.type)) {
        records.push({ key: `module:${entry.id}`, kind: 'module', id: entry.id, name: entry.name, category: 'Submarine Module', item: entry, customEntry: entry, derived });
      }
    }
    return records;
  }

  function tradeRecordByKey(key) {
    return allTradeRecords().find(record => record.key === key);
  }

  function tradeBasePrice(record) {
    if (!record) return 1;
    if (record.derived?.price) return Math.max(1, Math.round(record.derived.price));
    if (record.kind === 'upgrade') return Math.max(1, number(record.item.price, 100));
    if (record.kind === 'submarineWeapon') {
      const mountFactor = record.item.mount === 'Large Turret' ? 1800 : record.item.mount === 'Depth Charge' ? 1000 : 1200;
      return Math.round(mountFactor + number(record.item.fireRate, 1) * 120 + (record.item.ammo?.[0]?.magazineSize || 0) * 4);
    }
    const item = record.item || {};
    const categoryBase = {
      Weapons: 260, Ammunition: 25, Tools: 110, Gear: 180, Medical: 70, Consumables: 45,
      'Submarine Ammunition': 180, Miscellaneous: 65
    }[record.category] || 80;
    return Math.max(5, Math.round(categoryBase + number(item.carryUnits, 1) * 35 + number(item.capacity, 0) * (record.category === 'Ammunition' ? 2 : 4) + number(item.maxAmmo, 0) * 3 + number(item.fireRate, 0) * 15));
  }

  function vendorArchetype(id) {
    return (ITEM_FUNCTIONALITY.vendorArchetypes || []).find(item => item.id === id) || ITEM_FUNCTIONALITY.vendorArchetypes?.[0] || { id: 'general', name: 'General Trader', categories: [], stock: 8, markup: 1.1 };
  }

  function stationSpecialtyVendorIds(station) {
    const specialty = String(station.specialty || '').toLowerCase();
    const ids = [];
    if (/weapon|security|military/.test(specialty)) ids.push('arms-dealer');
    if (/medical|research/.test(specialty)) ids.push('medical-supplier');
    if (/engineering|salvage|fabrication|technology/.test(specialty)) ids.push('engineering-broker');
    if (/submarine|technology|fabrication/.test(specialty)) ids.push('systems-dealer');
    if (/diving|survival|equipment|clothing/.test(specialty)) ids.push('quartermaster');
    ids.push('general-trader');
    return [...new Set(ids)];
  }

  function randomOwnerName(rng, archetypeName) {
    const first = ['Avery','Mara','Jonas','Kira','Tomas','Rhea','Niko','Sera','Viktor','Elena','Pavel','Nadia','Ilya','Mina','Corin','Ada'];
    const last = ['Voss','Rook','Calder','Morrow','Kessler','Vale','Orlov','Dane','Sato','Kade','Marin','Holt','Renn','Bauer','Quill','Stroud'];
    return `${first[Math.floor(rng() * first.length)]} ${last[Math.floor(rng() * last.length)]}, ${archetypeName}`;
  }

  function generateVendorStock(archetype, rng) {
    const pool = allTradeRecords().filter(record => !archetype.categories?.length || archetype.categories.includes(record.category));
    const shuffled = [...pool].sort(() => rng() - 0.5);
    return shuffled.slice(0, Math.min(pool.length, Math.max(1, number(archetype.stock, 8)))).map(record => ({
      stockId: uid('stock'),
      tradeKey: record.key,
      quantity: archetype.machine ? 1 + Math.floor(rng() * 4) : 1 + Math.floor(rng() * 6),
      basePrice: tradeBasePrice(record)
    }));
  }

  function generateStationMarket(stationRecord) {
    stationRecord.marketRevision = Math.max(0, number(stationRecord.marketRevision)) + 1;
    const rng = seededRandom(`${stationRecord.id}:${stationRecord.marketRevision}`);
    const preferred = stationSpecialtyVendorIds(stationRecord);
    const normalPool = (ITEM_FUNCTIONALITY.vendorArchetypes || []).filter(item => !item.machine);
    const machinePool = (ITEM_FUNCTIONALITY.vendorArchetypes || []).filter(item => item.machine);
    const selectedIds = [...preferred];
    while (selectedIds.length < 3 && normalPool.length) {
      const candidate = normalPool[Math.floor(rng() * normalPool.length)]?.id;
      if (candidate && !selectedIds.includes(candidate)) selectedIds.push(candidate);
    }
    const machine = machinePool[Math.floor(rng() * machinePool.length)];
    stationRecord.vendors = [...selectedIds.slice(0, 3).map(id => vendorArchetype(id)), ...(machine ? [machine] : [])].map(archetype => ({
      vendorId: uid('vendor'),
      archetypeId: archetype.id,
      name: archetype.machine ? archetype.name : randomOwnerName(rng, archetype.name),
      machine: Boolean(archetype.machine),
      markup: number(archetype.markup, 1.1),
      stock: generateVendorStock(archetype, rng)
    }));
    save();
  }

  function combinedStationReputation(stationRecord) {
    const faction = number(state.stationCommerce.factionReputation[stationRecord.faction]);
    return faction * 0.5 + number(stationRecord.stationReputation) * 0.3 + number(stationRecord.ownerReputation) * 0.2;
  }

  function stationBuyPrice(stationRecord, vendor, stock) {
    const reputation = combinedStationReputation(stationRecord);
    const reputationFactor = clamp(1 - reputation / 250, 0.6, 1.6);
    return Math.max(1, Math.round(number(stock.basePrice, 1) * number(vendor.markup, 1.1) * reputationFactor));
  }

  function stationSellPrice(stationRecord, record) {
    const reputation = combinedStationReputation(stationRecord);
    return Math.max(1, Math.round(tradeBasePrice(record) * clamp(0.42 + reputation / 500, 0.2, 0.7)));
  }

  function commerceWalletCredits() {
    const commerce = normalizeStationCommerce();
    return commerce.wallet === 'character' ? Math.max(0, number(state.character.credits)) : Math.max(0, number(state.submarine.resources.credits));
  }

  function setCommerceWalletCredits(value) {
    const commerce = normalizeStationCommerce();
    if (commerce.wallet === 'character') state.character.credits = Math.max(0, number(value));
    else state.submarine.resources.credits = Math.max(0, number(value));
  }

  function recordTransaction(text) {
    state.stationCommerce.transactionLog.unshift({ id: uid('transaction'), text, at: new Date().toISOString() });
    state.stationCommerce.transactionLog = state.stationCommerce.transactionLog.slice(0, 40);
  }

  function deliverTradePurchase(record, destination, stationRecord) {
    if (record.kind === 'item') {
      if (record.category === 'Submarine Ammunition') addSubmarineAmmo(record.id, 1);
      else if (destination === 'character') addCharacterCatalogueItem(record.id, 1);
      else addCargoItem(record.id, 1, { acquiredFrom: stationRecord.name });
      return true;
    }
    if (record.kind === 'submarineWeapon') { addSubmarineWeapon(record.id); return true; }
    if (record.kind === 'upgrade') {
      if (!state.submarine.upgrades.includes(record.id)) state.submarine.upgrades.push(record.id);
      return true;
    }
    if (record.kind === 'module') { installCustomModule(record.id); return true; }
    return false;
  }

  function buyStationStock(stationRecord, vendorId, stockId) {
    const vendor = stationRecord.vendors.find(item => item.vendorId === vendorId);
    const stock = vendor?.stock?.find(item => item.stockId === stockId);
    const record = stock && tradeRecordByKey(stock.tradeKey);
    if (!vendor || !stock || !record || stock.quantity <= 0) return { ok: false, message: 'Out of stock' };
    const price = stationBuyPrice(stationRecord, vendor, stock);
    if (commerceWalletCredits() < price) return { ok: false, message: 'Insufficient credits' };
    if (!deliverTradePurchase(record, state.stationCommerce.destination, stationRecord)) return { ok: false, message: 'No compatible destination' };
    setCommerceWalletCredits(commerceWalletCredits() - price);
    stock.quantity -= 1;
    if (stationRecord.playerStation) stationRecord.treasury += price;
    stationRecord.ownerReputation = clamp(number(stationRecord.ownerReputation) + 0.25, -100, 100);
    recordTransaction(`Bought ${record.name} from ${vendor.name} at ${stationRecord.name} for ${price} credits.`);
    save();
    return { ok: true, message: `Bought for ${price}` };
  }

  function sellableRecords() {
    const cargo = normalizeCargoManagement().entries.map(entry => ({
      sourceKey: `cargo:${entry.cargoId}`, source: 'cargo', sourceId: entry.cargoId,
      name: cargoItem(entry).name, record: entry.id ? tradeRecordByKey(`item:${entry.id}`) : { key: `manual:${entry.cargoId}`, kind: 'manual', id: '', name: entry.customName, category: 'Miscellaneous', item: cargoItem(entry) }, quantity: entry.quantity
    }));
    const character = state.character.catalogInventory.map(entry => ({
      sourceKey: `character:${entry.instanceId}`, source: 'character', sourceId: entry.instanceId,
      name: inventoryDefinition(entry).name, record: entry.id ? tradeRecordByKey(`item:${entry.id}`) : { key: `manual-character:${entry.instanceId}`, kind: 'manual', id: '', name: entry.customName, category: 'Miscellaneous', item: inventoryDefinition(entry) }, quantity: entry.quantity
    }));
    return [...cargo, ...character].filter(item => item.record);
  }

  function sellToStation(sourceKey, stationRecord) {
    const sellable = sellableRecords().find(item => item.sourceKey === sourceKey);
    if (!sellable) return { ok: false, message: 'Item unavailable' };
    const price = stationSellPrice(stationRecord, sellable.record);
    if (stationRecord.playerStation && stationRecord.treasury < price) return { ok: false, message: 'Station treasury cannot afford this item' };
    if (sellable.source === 'cargo') removeCargoQuantity(sellable.sourceId, 1);
    else {
      const entry = inventoryEntry(sellable.sourceId);
      if (!entry) return { ok: false, message: 'Item unavailable' };
      if (entry.quantity > 1) entry.quantity -= 1;
      else state.character.catalogInventory = state.character.catalogInventory.filter(item => item.instanceId !== sellable.sourceId);
    }
    setCommerceWalletCredits(commerceWalletCredits() + price);
    if (stationRecord.playerStation) stationRecord.treasury -= price;
    stationRecord.stationReputation = clamp(number(stationRecord.stationReputation) + 0.1, -100, 100);
    const receivingVendor = stationRecord.vendors.find(vendor => !vendor.machine && vendorArchetype(vendor.archetypeId).categories.includes(sellable.record.category));
    if (receivingVendor && sellable.record.key && !sellable.record.key.startsWith('manual')) {
      const existing = receivingVendor.stock.find(stock => stock.tradeKey === sellable.record.key);
      if (existing) existing.quantity += 1;
      else receivingVendor.stock.push({ stockId: uid('stock'), tradeKey: sellable.record.key, quantity: 1, basePrice: tradeBasePrice(sellable.record) });
    }
    recordTransaction(`Sold ${sellable.name} at ${stationRecord.name} for ${price} credits.`);
    save();
    return { ok: true, message: `Sold for ${price}` };
  }

  function enhanceCharacterCargoTransfer() {
    const panel = [...root.querySelectorAll('.tool-panel')].find(item => item.querySelector('h3')?.textContent.includes('Tracked Inventory and Loadout'));
    if (!panel) return;
    panel.querySelectorAll('[data-inventory-instance]').forEach(row => {
      const actionCell = row.querySelector('td:last-child');
      if (!actionCell || actionCell.querySelector('.inventory-to-cargo')) return;
      const button = document.createElement('button');
      button.type = 'button';
      button.className = 'compact-button inventory-to-cargo';
      button.dataset.instance = row.dataset.inventoryInstance;
      button.textContent = 'Stow in Cargo';
      actionCell.prepend(button);
      button.addEventListener('click', () => {
        if (moveCharacterInventoryToCargo(button.dataset.instance)) renderCharacter();
      });
    });
  }
