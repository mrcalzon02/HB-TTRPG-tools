
  function normalizeWorldMapView() {
    state.world.mapView = state.world.mapView && typeof state.world.mapView === 'object' ? state.world.mapView : {};
    const view = state.world.mapView;
    view.scale = clamp(view.scale ?? 1, 0.2, 4);
    view.scrollLeft = Math.max(0, number(view.scrollLeft));
    view.scrollTop = Math.max(0, number(view.scrollTop));
    view.focusedNodeId = String(view.focusedNodeId || '');
    view.focusedEdgeId = String(view.focusedEdgeId || '');
    view.initialized = Boolean(view.initialized);
    return view;
  }

  function activeCrossingLegByEdge(edgeId) {
    return normalizeRouteCrossing().legs.find(leg => leg.edgeId === edgeId) || null;
  }

  function mapEdgeTooltipHtml(edge, map = state.world?.map) {
    const from = crossingNode(edge.a, map);
    const to = crossingNode(edge.b, map);
    const leg = activeCrossingLegByEdge(edge.id);
    return `<strong>${escapeHtml(from?.name || edge.a)} → ${escapeHtml(to?.name || edge.b)}</strong><br><span>Distance ${number(edge.distance, 1)} · ${edge.danger ?   }

  function managedCrossingEdgeState(edgeId) {
    const crossing = normalizeRouteCrossing();
    const leg = crossing.legs.find(item => item.edgeId === edgeId);
    if (!leg) return '';
    if (leg.status === 'complete') return 'crossing-complete';
    if (leg.status === 'underway' || leg.status === 'lost') return 'crossing-current';
    return 'crossing-filed';
  }

  function installWorldMapZoomAndInspection() {
    const frame = root.querySelector('.expanded-map-frame, .map-frame');
    const svg = frame?.querySelector('#world-map-svg');
    if (!frame || !svg) return;
    const view = normalizeWorldMapView();
    const basePixels = 6000;
    const applyScale = () => {
      svg.style.width = `${basePixels * view.scale}px`;
      svg.style.height = `${basePixels * view.scale}px`;
      svg.style.maxWidth = 'none';
    };
    applyScale();
    requestAnimationFrame(() => {
      if (!view.initialized) {
        const node = crossingNode(managedSubmarineNodeId());
        if (node) {
          const x = node.x / state.world.map.width * svg.clientWidth;
          const y = node.y / state.world.map.height * svg.clientHeight;
          frame.scrollLeft = Math.max(0, x - frame.clientWidth / 2);
          frame.scrollTop = Math.max(0, y - frame.clientHeight / 2);
        }
        view.initialized = true;
      } else {
        frame.scrollLeft = view.scrollLeft;
        frame.scrollTop = view.scrollTop;
      }
    });
    frame.addEventListener('wheel', event => {
      event.preventDefault();
      const rect = frame.getBoundingClientRect();
      const localX = event.clientX - rect.left;
      const localY = event.clientY - rect.top;
      const contentX = frame.scrollLeft + localX;
      const contentY = frame.scrollTop + localY;
      const oldScale = view.scale;
      const factor = event.deltaY < 0 ? 1.16 : 1 / 1.16;
      view.scale = clamp(oldScale * factor, 0.2, 4);
      const ratio = view.scale / oldScale;
      applyScale();
      frame.scrollLeft = contentX * ratio - localX;
      frame.scrollTop = contentY * ratio - localY;
      view.scrollLeft = frame.scrollLeft;
      view.scrollTop = frame.scrollTop;
    }, { passive: false });
    frame.addEventListener('scroll', () => { view.scrollLeft = frame.scrollLeft; view.scrollTop = frame.scrollTop; });

    let tooltip = document.getElementById('world-map-hover-tooltip');
    if (!tooltip) {
      tooltip = document.createElement('div');
      tooltip.id = 'world-map-hover-tooltip';
      tooltip.className = 'world-map-hover-tooltip';
      tooltip.style.cssText = 'position:fixed;display:none;z-index:10000;pointer-events:none;max-width:360px;padding:10px 12px;border:1px solid var(--accent);border-radius:10px;background:#0d1016;color:var(--ink);box-shadow:var(--shadow);font-size:.82rem;line-height:1.45';
      document.body.appendChild(tooltip);
    }
    frame.addEventListener('pointermove', event => {
      const nodeElement = event.target.closest?.('[data-node-id]');
      const edgeElement = event.target.closest?.('[data-edge-id]');
      if (!nodeElement && !edgeElement) { tooltip.style.display = 'none'; return; }
      if (nodeElement) {
        const node = crossingNode(nodeElement.dataset.nodeId);
        tooltip.innerHTML = node ? mapNodeTooltipHtml(node) : '';
      } else {
        const edge = state.world.map.edges.find(item => item.id === edgeElement.dataset.edgeId);
        tooltip.innerHTML = edge ? mapEdgeTooltipHtml(edge) : '';
      }
      tooltip.style.display = 'block';
      tooltip.style.left = `${Math.min(window.innerWidth - 380, event.clientX + 16)}px`;
      tooltip.style.top = `${Math.min(window.innerHeight - 180, event.clientY + 16)}px`;
    });
    frame.addEventListener('pointerleave', () => { tooltip.style.display = 'none'; });

    root.querySelectorAll('.map-node').forEach(oldElement => {
      const element = oldElement.cloneNode(true);
      oldElement.replaceWith(element);
      element.addEventListener('click', event => {
        event.stopPropagation();
        const nodeId = event.currentTarget.dataset.nodeId;
        view.focusedNodeId = nodeId;
        view.focusedEdgeId = '';
        save();
        renderWorld();
      });
    });
    root.querySelectorAll('.map-edge-group').forEach(element => element.addEventListener('click', event => {
      event.stopPropagation();
      view.focusedEdgeId = event.currentTarget.dataset.edgeId;
      view.focusedNodeId = '';
      save();
      renderWorld();
    }));
  }

  function bindMapInspectionControls() {
    const view = normalizeWorldMapView();
    const node = focusedMapNode();
    root.querySelector('#map-clear-focus')?.addEventListener('click', () => { view.focusedNodeId = ''; view.focusedEdgeId = ''; save(); renderWorld(); });
    root.querySelector('#map-set-departure')?.addEventListener('click', () => {
      if (!node || node.type !== 'station') return;
      state.route.startNodeId = node.id;
      state.route.start = node.name;
      state.world.selected = [node.id, ...state.world.selected.filter(id => id !== node.id).slice(0, 1)];
      save(); renderWorld();
    });
    root.querySelector('#map-set-destination')?.addEventListener('click', () => {
      if (!node || node.type !== 'station') return;
      state.route.endNodeId = node.id;
      state.route.end = node.name;
      const departure = state.route.startNodeId && state.route.startNodeId !== node.id ? state.route.startNodeId : state.world.selected.find(id => id !== node.id);
      state.world.selected = [departure, node.id].filter(Boolean);
      save(); renderWorld();
    });
    root.querySelector('#map-open-station')?.addEventListener('click', () => {
      if (!node || node.type !== 'station') return;
      syncWorldMapStations();
      const station = STATIONS.find(item => item.worldNodeId === node.id || item.name === node.name);
      if (station) normalizeStationCommerce().activeStationId = stationIdentity(station);
      save(); setTool('submarine');
    });
    root.querySelector('#map-scavenge-wreck')?.addEventListener('click', () => {
      const wreck = node ? wreckByNodeId(node.id) : null;
      if (!wreck) return;
      try { scavengeManagedSubmarineWreck(wreck.wreckSiteId); renderWorld(); }
      catch (error) { alert(error.message); }
    });
  }

    root.querySelector('#expedition-sync')?.addEventListener('click', () => { normalizeManagedExpedition(); save(); renderSubmarine(); });
    root.querySelector('#expedition-open-world')?.addEventListener('click', () => { const view = normalizeWorldMapView(); view.focusedNodeId = managedSubmarineNodeId(); view.focusedEdgeId = ''; view.initialized = false; save(); setTool('world'); });
    root.querySelector('#expedition-open-route')?.addEventListener('click', () => setTool('route'));
    root.querySelector('#expedition-declare-lost')?.addEventListener('click', () => { try { declareManagedSubmarineLostWithAllHands(); renderSubmarine(); } catch (error) { alert(error.message); } });
  }

  const renderSubmarineBeforeExpeditionMapUi = renderSubmarine;
  renderSubmarine = function renderSubmarineWithManagedExpedition() {
    normalizeManagedExpedition();
    renderSubmarineBeforeExpeditionMapUi();
    const shell = root.querySelector('.tool-shell');
    const hero = shell?.querySelector(':scope > .tool-hero');
    if (hero) hero.insertAdjacentHTML('afterend', managedExpeditionPanelHtml());
    bindManagedExpeditionPanel();
  };

  const crossingStatusLabelBeforeExpeditionMapUi = crossingStatusLabel;
  crossingStatusLabel = function crossingStatusLabelWithLoss(status) {
    return status === 'lost' ? 'Lost With All Hands' : crossingStatusLabelBeforeExpeditionMapUi(status);
  };
