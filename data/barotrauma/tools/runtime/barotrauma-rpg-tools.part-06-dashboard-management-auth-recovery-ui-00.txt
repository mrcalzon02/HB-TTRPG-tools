
  function dashboardManagementGroupOptions(selectedId=''){
    const groups=Object.values(normalizeWorldHub().publicState.groups||{});
    const fallbackId=selectedId||normalizeWorldHub().activeGroupId||groups[0]?.groupId||'';
    return `<option value="" ${fallbackId?'':'selected'}>Create a new expedition group</option>${groups.map(group=>`<option value="${escapeHtml(group.groupId)}" ${group.groupId===fallbackId?'selected':''}>${escapeHtml(group.name)} · ${escapeHtml(group.groupId)}</option>`).join('')}`;
  }

  function dashboardManagementGroupFields(prefix,selectedId=''){
    return `<label class="field">Expedition group<select id="${prefix}-group">${dashboardManagementGroupOptions(selectedId)}</select></label><label class="field">New group name<input id="${prefix}-new-group" placeholder="Used only when creating a new group"></label>`;
  }

  function dashboardManagementCreationPanelHtml(){
    const models=typeof dashboardStartingSubmarineNames==='function'?dashboardStartingSubmarineNames():Object.keys(SUBMARINES);
    return `<details class="tool-card" style="margin-top:12px" open><summary><strong>Create New Manageable Submarine</strong></summary><p class="small-note">Create a new vessel identity, assign it to an expedition group, build its independent campaign snapshot, set its management password, and issue its recovery code.</p><div class="field-grid four" style="margin-top:10px"><label class="field">Vessel name<input id="dashboard-create-vessel-name" placeholder="New submarine name"></label><label class="field">Submarine model<select id="dashboard-create-vessel-model">${models.map(name=>`<option value="${escapeHtml(name)}">${SUBMARINES[name]?.custom?'Custom — ':''}${escapeHtml(name)}</option>`).join('')}</select></label>${dashboardManagementGroupFields('dashboard-create-vessel')}<label class="field">Manager name<input id="dashboard-create-vessel-manager" value="${escapeHtml(state.character.player||'')}"></label><label class="field">Management password<input id="dashboard-create-vessel-password" type="password" autocomplete="new-password"></label><label class="field">Confirm password<input id="dashboard-create-vessel-confirm" type="password" autocomplete="new-password"></label><label class="check-row"><input id="dashboard-create-vessel-remember" type="checkbox" checked><span>Remember this device</span></label></div><div class="button-row" style="margin-top:10px"><button id="dashboard-create-manageable-vessel" class="primary-action" type="button">Create, Pair, and Log In</button></div></details>`;
  }

  function dashboardManagementClaimExistingPanelHtml(selected){
    if(!selected||selected.claimed||selected.groupId)return '';
    return `<details class="tool-card" style="margin-top:12px"><summary><strong>Pair and Claim This Existing Submarine</strong></summary><p class="small-note">This public-world vessel exists but has no expedition pairing or management credential.</p><div class="field-grid four" style="margin-top:10px">${dashboardManagementGroupFields('dashboard-claim-existing')}<label class="field">Manager name<input id="dashboard-claim-existing-manager" value="${escapeHtml(state.character.player||'')}"></label><label class="field">New password<input id="dashboard-claim-existing-password" type="password" autocomplete="new-password"></label><label class="field">Confirm password<input id="dashboard-claim-existing-confirm" type="password" autocomplete="new-password"></label><label class="check-row"><input id="dashboard-claim-existing-remember" type="checkbox" checked><span>Remember this device</span></label></div><div class="button-row" style="margin-top:10px"><button id="dashboard-claim-existing-vessel" class="primary-action" type="button">Pair, Claim, and Log In</button></div></details>`;
  }

  function dashboardManagementRepairPanelHtml(selected){
    if(!selected?.claimed||selected.groupId)return '';
    return `<details class="tool-card" style="margin-top:12px" open><summary><strong>Repair Login Pairing</strong></summary><p class="small-note">The password exists, but this submarine is not attached to an expedition group. Verify the current password and pair it.</p><div class="field-grid four" style="margin-top:10px"><label class="field">Current password<input id="dashboard-repair-pairing-password" type="password" autocomplete="current-password"></label>${dashboardManagementGroupFields('dashboard-repair-pairing')}<label class="check-row"><input id="dashboard-repair-pairing-remember" type="checkbox" checked><span>Remember this device</span></label></div><div class="button-row" style="margin-top:10px"><button id="dashboard-repair-pairing" class="primary-action" type="button">Repair Pairing and Log In</button></div></details>`;
  }

  function dashboardManagementResetPanelHtml(selected){
    if(!selected?.claimed)return '';
    const credential=dashboardManagementCredential(selected.vesselId);
    const recoveryReady=Boolean(credential?.recovery?.verifier);
    const groupFields=selected.groupId?'':dashboardManagementGroupFields('dashboard-reset');
    return `<details class="tool-card" style="margin-top:12px"><summary><strong>Reset Forgotten Password</strong></summary><p class="small-note">${recoveryReady?'Enter the recovery code. A successful reset rotates the password and recovery code and revokes every old session.':'This legacy credential has no recovery code. Log in once, or repair its pairing with the current password, to issue one.'}</p><div class="field-grid four" style="margin-top:10px"><label class="field">Recovery code<input id="dashboard-reset-recovery-code" autocomplete="off" ${recoveryReady?'':'disabled'} placeholder="XXXX-XXXX-XXXX-XXXX-XXXX-XXXX"></label><label class="field">New password<input id="dashboard-reset-new-password" type="password" autocomplete="new-password" ${recoveryReady?'':'disabled'}></label><label class="field">Confirm password<input id="dashboard-reset-confirm-password" type="password" autocomplete="new-password" ${recoveryReady?'':'disabled'}></label><label class="check-row"><input id="dashboard-reset-remember" type="checkbox" checked ${recoveryReady?'':'disabled'}><span>Remember this device</span></label>${groupFields}</div><div class="button-row" style="margin-top:10px"><button id="dashboard-reset-password" class="secondary-action" type="button" ${recoveryReady?'':'disabled'}>Reset Password and Log In</button></div></details>`;
  }

  function dashboardManagementCreationAndRecoveryHtml(selected){
    return `${dashboardManagementRepairPanelHtml(selected)}${dashboardManagementClaimExistingPanelHtml(selected)}${dashboardManagementResetPanelHtml(selected)}${dashboardManagementCreationPanelHtml()}`;
  }

  const dashboardManagementGateHtmlBeforeRecovery=dashboardManagementGateHtml;
  dashboardManagementGateHtml=function dashboardManagementGateHtmlWithRecovery(){
    const html=dashboardManagementGateHtmlBeforeRecovery();
    if(dashboardManagementRuntime.status==='checking')return html;
    const selectedId=dashboardManagementSelectedVesselId();
    const selected=dashboardManagementVessels().find(vessel=>vessel.vesselId===selectedId)||null;
    return html.replace('<div id="dashboard-management-message"',`${dashboardManagementCreationAndRecoveryHtml(selected)}<div id="dashboard-management-message"`);
  };
