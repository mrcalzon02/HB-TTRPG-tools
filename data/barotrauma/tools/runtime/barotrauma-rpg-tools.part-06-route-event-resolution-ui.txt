
  function routeResolutionOutcomeLabel(outcome) {
    return ({'critical-success':'Critical Success',success:'Success',mixed:'Mixed Success',failure:'Failure','critical-failure':'Critical Failure'})[outcome] || outcome;
  }

  function routeResolutionOutcomeTone(outcome) {
    return outcome==='critical-success'||outcome==='success'?'good':outcome==='mixed'?'warn':'bad';
  }

  function routeResolutionExpandedEffectsText(effects={}) {
    const parts = [encounterEffectsText(effects)];
    if (number(effects.crewMorale)) parts.push(`all-crew Morale ${number(effects.crewMorale)>0?'+':''}${number(effects.crewMorale)}`);
    if (number(effects.injuryCount)) parts.push(`${number(effects.injuryCount)} crew injury`);
    return parts.filter(part => part&&part!=='No automatic state changes').join(' · ')||'Narrative consequence only';
  }

  function routeResolutionOutcomeDescription(session,event,outcome) {
    const solution = routeResolutionSelectedSolution(session);
    if (outcome==='critical-success') return `${solution.successText} The crew also gains exceptional control of the route and converts the incident into a lasting advantage.`;
    if (outcome==='success') return solution.successText;
    if (outcome==='mixed') return solution.mixedText;
    if (outcome==='critical-failure') return `${solution.failureText} The failure cascades into a severe vessel-wide or crew-wide consequence.`;
    return solution.failureText;
  }

  function routeResolutionStepRowsHtml(session) {
    return session.steps.map((step,index) => {
      const active = index===session.activeStepIndex&&step.status==='pending';
      const tone = step.status==='success'?'good':step.status==='failure'?'bad':active?'warn':'';
      return `<tr class="${active?'dashboard-active-turn-row':''}"><td>${index+1}</td><td><strong>${escapeHtml(step.phase)}</strong><br><span class="small-note">${escapeHtml(step.description)}</span></td><td>${escapeHtml(step.requiredRole)}</td><td>${escapeHtml(step.stat)} vs ${number(step.dc)}</td><td><strong class="${tone}">${step.status==='pending'?(active?'ACTIVE STEP':'Pending'):step.status}</strong>${step.actorName?`<br><span class="small-note">${escapeHtml(step.actorName)} · ${escapeHtml(step.actorDuty)}</span>`:''}</td><td>${step.rollText?escapeHtml(step.rollText):'—'}</td></tr>`;
    }).join('');
  }

  function routeResolutionSolutionsHtml(session) {
    const locked = session.steps.some(step => step.status!=='pending');
    return session.solutions.map(solution => `<label class="tool-card" style="display:block;border-color:${solution.id===session.selectedSolutionId?'var(--accent)':'var(--line)'}"><div class="check-row" style="grid-template-columns:auto 1fr"><input class="route-resolution-solution" type="radio" name="route-resolution-solution" value="${escapeHtml(solution.id)}" ${solution.id===session.selectedSolutionId?'checked':''} ${locked?'disabled':''}><span><strong>${escapeHtml(solution.label)}</strong><br><span class="small-note">Lead role: ${escapeHtml(solution.leadRole)} · ${escapeHtml(solution.stat)} vs ${number(solution.dc)} · Supports: ${escapeHtml(solution.supportRoles.join(', '))}</span></span></div><p>${escapeHtml(solution.description)}</p><div class="notice good"><strong>Success:</strong> ${escapeHtml(solution.successText)}</div><div class="notice warn" style="margin-top:6px"><strong>Mixed:</strong> ${escapeHtml(solution.mixedText)}</div><div class="notice bad" style="margin-top:6px"><strong>Failure:</strong> ${escapeHtml(solution.failureText)}</div></label>`).join('');
  }

  function routeResolutionSupportHtml(session) {
    const locked = session.steps.some(step => step.status!=='pending');
    return dashboardTurnActors().map(actor => {
      const duty = routeResolutionActorDuty(actor);
      return `<label class="check-row" style="grid-template-columns:auto 1fr"><input class="route-resolution-support" type="checkbox" value="${escapeHtml(actor.ref)}" ${session.supportCrewRefs.includes(actor.ref)?'checked':''} ${locked?'disabled':''}><span><strong>${escapeHtml(actor.name)}</strong><br><span class="small-note">${escapeHtml(duty)} · ${escapeHtml(actor.status)} · can contribute up to +1 when role/stat matches</span></span></label>`;
    }).join('')||'<div class="notice bad">No living commissioned crew are available.</div>';
  }

  function routeResolutionOutcomeCardsHtml(session,event) {
    return ROUTE_RESOLUTION_OUTCOMES.map(outcome => {
      const effects = routeResolutionEffectsForOutcome(session,event,outcome);
      const selected = (session.outcomeOverride||session.provisionalOutcome)===outcome;
      return `<div class="tool-card" style="border-color:${selected?'var(--accent)':'var(--line)'}"><div class="tool-hero" style="padding:0;border:0;background:none"><strong class="${routeResolutionOutcomeTone(outcome)}">${routeResolutionOutcomeLabel(outcome)}</strong>${selected?'<span class="status-pill warn">CURRENT</span>':''}</div><p class="small-note">${escapeHtml(routeResolutionOutcomeDescription(session,event,outcome))}</p><div class="notice"><strong>Applied state changes:</strong> ${escapeHtml(routeResolutionExpandedEffectsText(effects))}</div></div>`;
    }).join('');
  }

  function renderCurrentRouteEventResolution() {
    if (normalizeRouteCrossing().status!=='underway') return;
    const context = activeRouteResolutionContext();
    if (!context||root.querySelector('#explicit-route-event-resolution')) return;
    const dashboardCard = root.querySelector('#dashboard-roll-next')?.closest('.tool-card');
    const standaloneCard = root.querySelector(`.current-crossing-event[data-id="${context.slot.id}"]`);
    const target = dashboardCard||standaloneCard;
    if (!target) return;
    target.insertAdjacentHTML('beforebegin',routeEventResolutionHtml(context,activeSubmarineDashboardContext()?'dashboard':'standalone'));
    target.remove();
    bindCurrentRouteEventResolution();
  }

  function rerenderRouteResolutionView() {
    if (activeSubmarineDashboardContext()) renderSubmarine();
    else renderRoute();
  }

  function routeResolutionSetMessage(message,isError=false) {
    normalizeRouteEventResolutionState().lastMessage = String(message||'');
    const output = root.querySelector('#route-resolution-message');
    if (output) { output.textContent=message; output.classList.toggle('bad',isError); output.classList.toggle('good',!isError); }
    save();
  }

  function bindCurrentRouteEventResolution() {
    const panel = root.querySelector('#explicit-route-event-resolution');
    if (!panel) return;
    panel.querySelectorAll('.route-resolution-solution').forEach(input=>input.addEventListener('change',event=>{ selectRouteResolutionSolution(event.target.value); rerenderRouteResolutionView(); }));
    panel.querySelectorAll('.route-resolution-support').forEach(input=>input.addEventListener('change',()=>{ const refs=[...panel.querySelectorAll('.route-resolution-support:checked')].map(item=>item.value); setRouteResolutionSupportCrew(refs); rerenderRouteResolutionView(); }));
    panel.querySelector('#route-resolution-opportunity')?.addEventListener('change',event=>{ setRouteResolutionOpportunity(event.target.checked); rerenderRouteResolutionView(); });
    panel.querySelector('#route-resolution-contract')?.addEventListener('change',event=>{ setRouteResolutionContract(event.target.checked); rerenderRouteResolutionView(); });
    panel.querySelector('#route-resolution-roll-step')?.addEventListener('click',()=>{ const result=rollRouteResolutionStep(); if (result.ok) rerenderRouteResolutionView(); else routeResolutionSetMessage(result.message,true); });
    panel.querySelector('#route-resolution-outcome-override')?.addEventListener('change',event=>{ const context=activeRouteResolutionContext(); if (!context) return; context.session.outcomeOverride=event.target.value; save(); rerenderRouteResolutionView(); });
    panel.querySelector('#route-resolution-commit')?.addEventListener('click',()=>{ const result=commitRouteEventResolution(); if (result.ok) rerenderRouteResolutionView(); else routeResolutionSetMessage(result.message,true); });
    panel.querySelector('#route-resolution-reset')?.addEventListener('click',()=>{ const context=activeRouteResolutionContext(); if (!context) return; resetRouteResolutionSteps(context.session,context.event); rerenderRouteResolutionView(); });
    panel.querySelector('#route-resolution-redraw')?.addEventListener('click',()=>{ if (resetActiveRouteResolution(true)) rerenderRouteResolutionView(); });
  }

  function routeResolutionContractLedgerHtml() {
    const contracts = normalizeRouteEventResolutionState().contracts.slice(0,12);
    if (!contracts.length) return '';
    return `<section class="tool-panel" style="margin-top:12px"><h3>Route Mission and Contract Ledger</h3><div class="table-wrap"><table class="tool-table"><thead><tr><th>Contract</th><th>Issuer</th><th>Objective</th><th>Outcome</th><th>Effects</th></tr></thead><tbody>${contracts.map(contract=>`<tr><td><strong>${escapeHtml(contract.title)}</strong><br><code>${escapeHtml(contract.id)}</code></td><td>${escapeHtml(contract.issuer)}</td><td>${escapeHtml(contract.objective)}</td><td><strong class="${routeResolutionOutcomeTone(contract.outcome)}">${escapeHtml(routeResolutionOutcomeLabel(contract.outcome))}</strong><br>${escapeHtml(contract.status)}</td><td>${escapeHtml(contract.effectSummary||'Pending')}</td></tr>`).join('')}</tbody></table></div></section>`;
  }

  function injectRouteResolutionContractLedger() {
    if (root.querySelector('#explicit-route-event-resolution')||!normalizeRouteEventResolutionState().contracts.length) return;
    root.querySelector('.tool-shell')?.insertAdjacentHTML('beforeend',routeResolutionContractLedgerHtml());
  }

  const renderActiveSubmarineDashboardBeforeExplicitRouteResolution = renderActiveSubmarineDashboard;
  renderActiveSubmarineDashboard = function renderActiveSubmarineDashboardWithExplicitRouteResolution(requestedFocus=activeDashboardRawFocus()) {
    const result = renderActiveSubmarineDashboardBeforeExplicitRouteResolution(requestedFocus);
    renderCurrentRouteEventResolution();
    injectRouteResolutionContractLedger();
    return result;
  };

  const renderRouteBeforeExplicitRouteResolution = renderRoute;
  renderRoute = function renderStandaloneRouteWithExplicitResolution() {
    const result = renderRouteBeforeExplicitRouteResolution();
    if (!activeSubmarineDashboardContext()) {
      renderCurrentRouteEventResolution();
      injectRouteResolutionContractLedger();
    }
    return result;
  };
