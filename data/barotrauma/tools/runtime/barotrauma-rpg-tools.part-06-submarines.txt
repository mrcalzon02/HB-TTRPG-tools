
  const SUBMARINE_ROSTER_DATA = globalThis.BAROTRAUMA_SUBMARINE_ROSTER || { submarines: [], derivation: {}, source: {} };
  const OFFICIAL_SUBMARINE_PROFILES = SUBMARINE_ROSTER_DATA.submarines || [];
  const OFFICIAL_SUBMARINE_NAMES = new Set(OFFICIAL_SUBMARINE_PROFILES.map(item => item.name));
  const RPG_SUBMARINE_ANCHORS = {
    Barsuk: { speed: 8, maneuverability: 10, defense: 5, capacity: 5, hull: 8 },
    Dugong: { speed: 6, maneuverability: 7, defense: 10, capacity: 7, hull: 16 },
    Orca: { speed: 6, maneuverability: 7, defense: 7, capacity: 7, hull: 10 },
    Humpback: { speed: 5, maneuverability: 6, defense: 8, capacity: 10, hull: 12 }
  };
  const SUBMARINE_STAT_LIMITS = SUBMARINE_ROSTER_DATA.derivation?.ranges || {
    speed: [3, 10],
    maneuverability: [3, 10],
    defense: [5, 16],
    capacity: [4, 30],
    hull: [6, 30]
  };

  function symmetricRound(value) {
    return value >= 0 ? Math.floor(value + 0.5) : Math.ceil(value - 0.5);
  }

  function profileNumber(profile, key, fallback = 0) {
    return number(profile?.[key], fallback);
  }

  function submarineWeaponWeight(profile) {
    return profileNumber(profile, 'smallFixed')
      + (profileNumber(profile, 'smallHardpoints') * 0.5)
      + (profileNumber(profile, 'largeFixed') * 2)
      + profileNumber(profile, 'largeHardpoints')
      + (profileNumber(profile, 'utilitySystems') * 0.5);
  }

  function selectSubmarineAnchor(profile) {
    if (profile.class === 'Attack') return profileNumber(profile, 'tier', 1) === 1 ? 'Barsuk' : 'Humpback';
    if (profile.class === 'Scout') return profileNumber(profile, 'horizontal') >= 23 || profileNumber(profile, 'utilitySystems') >= 2 ? 'Orca' : 'Dugong';
    return 'Humpback';
  }

  function deriveSubmarineStats(profile) {
    if (profile.official && RPG_SUBMARINE_ANCHORS[profile.name]) {
      return { ...RPG_SUBMARINE_ANCHORS[profile.name], anchor: profile.name, exactAnchor: true };
    }
    const anchorName = selectSubmarineAnchor(profile);
    const anchorProfile = OFFICIAL_SUBMARINE_PROFILES.find(item => item.name === anchorName) || OFFICIAL_SUBMARINE_PROFILES[0] || {};
    const anchorStats = RPG_SUBMARINE_ANCHORS[anchorName] || RPG_SUBMARINE_ANCHORS.Humpback;
    let speed = anchorStats.speed + symmetricRound((profileNumber(profile, 'horizontal') - profileNumber(anchorProfile, 'horizontal')) / 6);
    if (profileNumber(profile, 'crewMax') - profileNumber(anchorProfile, 'crewMax') >= 6) speed -= 1;
    let maneuverability = anchorStats.maneuverability
      + symmetricRound((profileNumber(profile, 'descent') - profileNumber(anchorProfile, 'descent')) / 4)
      - symmetricRound(Math.max(0, profileNumber(profile, 'crewMax') - profileNumber(anchorProfile, 'crewMax')) / 4)
      - (profileNumber(profile, 'cargo') >= 60 ? 1 : 0);
    let defense = anchorStats.defense
      + symmetricRound((submarineWeaponWeight(profile) - submarineWeaponWeight(anchorProfile)) * 1.2)
      + (profileNumber(profile, 'tier', 1) - profileNumber(anchorProfile, 'tier', 1))
      + (profile.class === 'Transport' ? -1 : 0);
    let capacity = anchorStats.capacity
      + symmetricRound((profileNumber(profile, 'cargo') - profileNumber(anchorProfile, 'cargo')) / 12)
      + symmetricRound((profileNumber(profile, 'crewMax') - profileNumber(anchorProfile, 'crewMax')) / 3)
      + (profile.class === 'Transport' ? 2 : 0);
    let hull = anchorStats.hull
      + (2 * (profileNumber(profile, 'tier', 1) - profileNumber(anchorProfile, 'tier', 1)))
      + symmetricRound((profileNumber(profile, 'crewMax') - profileNumber(anchorProfile, 'crewMax')) / 2)
      + symmetricRound((profileNumber(profile, 'cargo') - profileNumber(anchorProfile, 'cargo')) / 24)
      + symmetricRound((submarineWeaponWeight(profile) - submarineWeaponWeight(anchorProfile)) / 2);

    const bounded = {};
    for (const [stat, value] of Object.entries({ speed, maneuverability, defense, capacity, hull })) {
      const [minimum, maximum] = SUBMARINE_STAT_LIMITS[stat] || [-999, 999];
      bounded[stat] = clamp(value, minimum, maximum);
    }
    return { ...bounded, anchor: anchorName, exactAnchor: false };
  }

  function submarineProfileSummary(profile, stats) {
    const strongest = Object.entries(stats).filter(([key]) => ['speed','maneuverability','defense','capacity','hull'].includes(key)).sort((a, b) => b[1] - a[1])[0];
    const weakest = Object.entries(stats).filter(([key]) => ['speed','maneuverability','defense','capacity','hull'].includes(key)).sort((a, b) => a[1] - b[1])[0];
    const classText = profile.class === 'Transport'
      ? `Transport frame with ${profileNumber(profile, 'cargo')} crate capacity.`
      : profile.class === 'Attack'
        ? `Attack frame carrying weapon weight ${submarineWeaponWeight(profile).toFixed(1)}.`
        : `Scout frame rated for ${profileNumber(profile, 'horizontal')} km/h horizontal travel.`;
    return {
      strengths: `${classText} Highest derived RPG statistic: ${strongest?.[0] || 'capacity'} ${strongest?.[1] ?? ''}.`,
      weaknesses: `Lowest derived RPG statistic: ${weakest?.[0] || 'speed'} ${weakest?.[1] ?? ''}; crew requirement ${profileNumber(profile, 'crewMin')}-${profileNumber(profile, 'crewMax')}.`
    };
  }

  function defaultCustomSubmarineDraft() {
    return {
      id: '',
      name: '',
      class: 'Scout',
      tier: 1,
      crewMin: 3,
      crewMax: 5,
      cargo: 8,
      horizontal: 22,
      descent: 14,
      smallFixed: 1,
      smallHardpoints: 1,
      largeFixed: 0,
      largeHardpoints: 0,
      utilitySystems: 1,
      notes: ''
    };
  }

  state.submarineLibrary = state.submarineLibrary && typeof state.submarineLibrary === 'object'
    ? state.submarineLibrary
    : { custom: [], draft: defaultCustomSubmarineDraft() };
  state.submarineLibrary.custom = Array.isArray(state.submarineLibrary.custom) ? state.submarineLibrary.custom : [];
  state.submarineLibrary.draft = deepMerge(defaultCustomSubmarineDraft(), state.submarineLibrary.draft || {});

  function genericSystemsForProfile(profile) {
    if (Array.isArray(profile.defaultSystems) && profile.defaultSystems.length) return [...profile.defaultSystems];
    return [
      ...Array(Math.max(0, profileNumber(profile, 'smallFixed'))).fill('coilgun'),
      ...Array(Math.max(0, profileNumber(profile, 'largeFixed'))).fill('railgun'),
      ...Array(Math.max(0, profileNumber(profile, 'utilitySystems'))).fill('electrical-discharge-coil')
    ];
  }

  function makeWikiWeaponMount(systemId) {
    const weapon = catalogueWeapon(systemId);
    return weapon ? {
      instanceId: uid('wiki-weapon'),
      id: systemId,
      ammoName: weapon.ammo?.[0]?.name || '',
      loaded: weapon.ammo?.[0]?.magazineSize || 0
    } : null;
  }

  function installDefaultWikiWeapons(profile) {
    state.submarine.wikiWeapons = genericSystemsForProfile(profile).map(makeWikiWeaponMount).filter(Boolean);
    state.submarine.wikiAmmo = [];
  }

  function registerOfficialSubmarines() {
    for (const rawProfile of OFFICIAL_SUBMARINE_PROFILES) {
      const profile = { ...rawProfile, official: true };
      const stats = deriveSubmarineStats(profile);
      const summary = submarineProfileSummary(profile, stats);
      if (SUBMARINES[profile.name]) {
        Object.assign(SUBMARINES[profile.name], {
          ...stats,
          profile,
          source: 'Official Barotrauma Wiki',
          custom: false
        });
      } else {
        SUBMARINES[profile.name] = {
          ...stats,
          weapons: [],
          strengths: summary.strengths,
          weaknesses: summary.weaknesses,
          profile,
          source: 'Official Barotrauma Wiki',
          custom: false
        };
      }
      if (!SUBMARINES[profile.name].strengths) SUBMARINES[profile.name].strengths = summary.strengths;
      if (!SUBMARINES[profile.name].weaknesses) SUBMARINES[profile.name].weaknesses = summary.weaknesses;
    }
  }

  function syncCustomSubmarines() {
    for (const [name, model] of Object.entries(SUBMARINES)) {
      if (model?.custom) delete SUBMARINES[name];
    }
    for (const profile of state.submarineLibrary.custom) {
      if (!profile?.name || OFFICIAL_SUBMARINE_NAMES.has(profile.name)) continue;
      const stats = deriveSubmarineStats({ ...profile, official: false });
      const summary = submarineProfileSummary(profile, stats);
      SUBMARINES[profile.name] = {
        ...stats,
        weapons: [],
        strengths: summary.strengths,
        weaknesses: summary.weaknesses,
        profile,
        source: 'Custom derived submarine',
        custom: true
      };
    }
  }

  registerOfficialSubmarines();
  syncCustomSubmarines();

  if (state.submarine.wikiDefaultsInitialized == null) {
    const initialProfile = SUBMARINES[state.submarine.model]?.profile;
    if (initialProfile && !state.submarine.wikiWeapons.length) installDefaultWikiWeapons(initialProfile);
    state.submarine.wikiDefaultsInitialized = true;
    save();
  }

  const baseCalculateSubmarineWithCatalogue = calculateSubmarine;
  calculateSubmarine = function calculateSubmarineWithOfficialSystems() {
    const result = baseCalculateSubmarineWithCatalogue();
    const wikiWeaponCapacity = state.submarine.wikiWeapons.reduce((sum, entry) => {
      const weapon = catalogueWeapon(entry.id);
      if (!weapon) return sum;
      return sum + (weapon.mount === 'Large Turret' ? 2 : 1);
    }, 0);
    result.wikiWeaponCapacity = wikiWeaponCapacity;
    result.weaponCapacity += wikiWeaponCapacity;
    result.usedCapacity += wikiWeaponCapacity;
    result.remainingCapacity = result.totals.capacity - result.usedCapacity;
    return result;
  };

  function useSubmarineModel(name) {
    syncCustomSubmarines();
    const model = SUBMARINES[name];
    if (!model) return;
    state.submarine.model = name;
    state.submarine.weapons = (model.weapons || []).map(item => ({ ...item }));
    state.submarine.hullCurrent = model.hull;
    if (model.profile) {
      state.submarine.resources.crew = model.profile.crewMin;
      installDefaultWikiWeapons(model.profile);
    }
    state.submarine.wikiDefaultsInitialized = true;
    save();
    renderSubmarine();
  }

  function cloneProfileIntoDraft(profile, suffix = ' Custom') {
    const draft = {
      ...defaultCustomSubmarineDraft(),
      ...profile,
      id: '',
      name: `${profile.name || 'Custom Submarine'}${suffix}`,
      notes: profile.notes || ''
    };
    delete draft.official;
    delete draft.wikiSlug;
    delete draft.price;
    delete draft.experience;
    delete draft.drone;
    delete draft.defaultSystems;
    state.submarineLibrary.draft = draft;
    save();
    renderSubmarine();
  }

  function draftInputRow(label, key, options = {}) {
    const draft = state.submarineLibrary.draft;
    if (options.type === 'select') {
      return `<label class="field">${escapeHtml(label)}<select class="custom-sub-field" data-key="${escapeHtml(key)}">${optionList(options.values, draft[key])}</select></label>`;
    }
    return `<label class="field">${escapeHtml(label)}<input class="custom-sub-field" data-key="${escapeHtml(key)}" type="${options.type || 'number'}" ${options.min != null ? `min="${options.min}"` : ''} ${options.max != null ? `max="${options.max}"` : ''} ${options.step != null ? `step="${options.step}"` : ''} value="${escapeHtml(draft[key])}"></label>`;
  }

        alert('Enter a unique submarine name before saving.');
        return;

        alert('That name belongs to an official submarine. Choose a distinct custom name.');
        return;

        renderSubmarine();
      }
    }));
    rosterPanel.querySelectorAll('.custom-sub-delete').forEach(button => button.addEventListener('click', event => {
      const item = customById(event.target.dataset.id);
      if (!item) return;
      state.submarineLibrary.custom = state.submarineLibrary.custom.filter(entry => entry.id !== item.id);
      if (state.submarine.model === item.name) useSubmarineModel('Barsuk');
      else {
        syncCustomSubmarines();
        save();
        renderSubmarine();
      }
    }));
  }

  const renderSubmarineWithCatalogue = renderSubmarine;
  renderSubmarine = function renderSubmarineWithExpandedRoster() {
    registerOfficialSubmarines();
    syncCustomSubmarines();
    if (!SUBMARINES[state.submarine.model]) state.submarine.model = 'Barsuk';
    renderSubmarineWithCatalogue();
    enhanceSubmarineRoster();
  };
