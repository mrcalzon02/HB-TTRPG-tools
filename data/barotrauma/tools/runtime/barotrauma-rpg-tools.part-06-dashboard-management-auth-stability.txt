
  function dashboardManagementDenied(action,returnValue=false){
    const message=`Log in to ${action}.`;
    dashboardManagementRuntime.message=message;
    if(typeof dashboardSetMessage==='function')dashboardSetMessage(message,true);
    return returnValue;
  }

  function dashboardManagementGuardFunction(original,action,deniedValue=false){
    return function dashboardManagementGuardedFunction(...args){
      if(activeSubmarineDashboardContext()&&!dashboardManagementAuthorized())return dashboardManagementDenied(action,typeof deniedValue==='function'?deniedValue():deniedValue);
      return original.apply(this,args);
    };
  }

    return normalizeActiveSubmarineDashboardBeforeManagementBoundary(persist);
  };

  const importWorldSaveBeforeDashboardManagementAuth=importWorldSave;

  function dashboardManagementContainsPlaintextPassword(value){
    if(!value||typeof value!=='object')return false;
    for(const [key,entry] of Object.entries(value)){
      if(/password/i.test(key))return true;
      if(entry&&typeof entry==='object'&&dashboardManagementContainsPlaintextPassword(entry))return true;
    }
    return false;
  }

  function dashboardManagementAuthDiagnostics(){
    const authority=normalizeDashboardManagementAuthority();
    const issues=[];
    for(const [vesselId,credential] of Object.entries(authority.submarineCredentials)){
      if(!credential.credentialId)issues.push(`Submarine ${vesselId} has no credential ID.`);
      if(!credential.salt)issues.push(`Submarine ${vesselId} has no password salt.`);
      if(!credential.verifier)issues.push(`Submarine ${vesselId} has no password verifier.`);
      if(number(credential.iterations)<100000)issues.push(`Submarine ${vesselId} uses too few PBKDF2 iterations.`);
      for(const receipt of Object.values(credential.receipts||{})){
        if(!receipt.tokenHash)issues.push(`Submarine ${vesselId} has a receipt without a token hash.`);
        if('token' in receipt)issues.push(`Submarine ${vesselId} stores a raw receipt token in the world save.`);
      }
    }
    if(dashboardManagementContainsPlaintextPassword(authority))issues.push('The management authority contains a plaintext password field.');
    if(dashboardManagementRuntime.status==='authenticated'&&!dashboardManagementAuthorized(dashboardManagementRuntime.vesselId))issues.push('The runtime reports an authenticated but invalid management session.');
    return {valid:!issues.length,issues,version:DASHBOARD_MANAGEMENT_AUTH_VERSION,credentialCount:Object.keys(authority.submarineCredentials).length,snapshotCount:Object.keys(authority.vesselSnapshots).length,sessionStatus:dashboardManagementRuntime.status,activeManagedVesselId:dashboardManagementRuntime.vesselId||''};
  }

  const dashboardBoundaryDiagnosticsBeforeManagementAuth=dashboardBoundaryDiagnostics;
  dashboardBoundaryDiagnostics=function dashboardBoundaryDiagnosticsWithManagementAuth(){
    const result=dashboardBoundaryDiagnosticsBeforeManagementAuth();
    const auth=dashboardManagementAuthDiagnostics();
    result.managementAuth=auth;
    if(!auth.valid)result.issues.push(...auth.issues);
    result.valid=!result.issues.length;
    return result;
  };
