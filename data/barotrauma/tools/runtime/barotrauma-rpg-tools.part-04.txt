    return { selectedJobs, jobEvents, routeEvents, total: jobEvents + routeEvents, payment: selectedJobs.reduce((sum, job) => sum + job.payment, 0) };
  }

  function drawWeightedEvent(route) {
    const category = weightedCategory(route.table, route.sanityEnabled);
    const pool = eligibleEvents(category, route.sanityEnabled);
    return { ...(pool[Math.floor(Math.random() * pool.length)] || EVENTS[0]) };
  }

  function seedHash(text) {
    let hash = 2166136261;
    for (let index = 0; index < String(text).length; index += 1) {
      hash ^= String(text).charCodeAt(index);
      hash = Math.imul(hash, 16777619);
    }
    return hash >>> 0;
  }

  function seededRandom(seedText) {
    let value = seedHash(seedText) || 1;
    return () => {
      value += 0x6D2B79F5;
      let result = value;
      result = Math.imul(result ^ (result >>> 15), result | 1);
      result ^= result + Math.imul(result ^ (result >>> 7), result | 61);
      return ((result ^ (result >>> 14)) >>> 0) / 4294967296;
    };
  }

  function mapName(rng, used) {
    let name;
    do {
      name = `${MAP_PREFIX[Math.floor(rng() * MAP_PREFIX.length)]} ${MAP_SUFFIX[Math.floor(rng() * MAP_SUFFIX.length)]}`;
    } while (used.has(name));
    used.add(name);
    return name;
  }

  function generateWorldMap() {
    const world = state.world;
    const rng = seededRandom(`${world.seed}|${world.rings}|${world.nodes}|${world.branching}`);
    const rings = clamp(world.rings, 3, 8);
    const requestedNodes = clamp(world.nodes, rings * 3 + 1, 60);
    const width = 1100;
    const height = 680;
    const center = { x: width / 2, y: height / 2 };
    const maxRadius = Math.min(width, height) * .43;
    const nodes = [{ id: 'anomaly', name: 'The Central Anomaly', type: 'anomaly', ring: 0, x: center.x, y: center.y, details: 'The campaign’s ultimate destination at Europa’s center.' }];
    const usedNames = new Set(nodes.map(node => node.name));
    const stationQueue = STATIONS.map(item => ({ ...item }));
    const ringCounts = [];
    let remaining = requestedNodes - 1;
    let circumferenceWeight = 0;
    for (let ring = 1; ring <= rings; ring += 1) circumferenceWeight += ring;
    for (let ring = 1; ring <= rings; ring += 1) {
      const minimumRemaining = (rings - ring) * 3;
      let count = ring === rings ? remaining : Math.max(3, Math.round((requestedNodes - 1) * ring / circumferenceWeight));
      count = Math.min(count, remaining - minimumRemaining);
      ringCounts.push(count);
      remaining -= count;
    }
    if (remaining > 0) ringCounts[ringCounts.length - 1] += remaining;

    for (let ring = 1; ring <= rings; ring += 1) {
      const count = ringCounts[ring - 1];
      const radius = maxRadius * ring / rings;
      const phase = rng() * Math.PI * 2;
      for (let index = 0; index < count; index += 1) {
        const angle = phase + (Math.PI * 2 * index / count) + (rng() - .5) * .18;
