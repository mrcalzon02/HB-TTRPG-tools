        const jitter = (rng() - .5) * Math.min(24, radius * .08);
        const outerBias = ring / rings;
        let type;
        const roll = rng();
        if (ring === rings && roll < .52) type = 'station';
        else if (roll < .22) type = 'outpost';
        else if (roll < .43) type = 'wreck';
        else if (roll < .62) type = 'ruins';
        else if (roll < .82) type = 'hazard';
        else type = 'station';
        let name;
        let details;
        if (type === 'station' && stationQueue.length) {
          const source = stationQueue.shift();
          name = source.name;
          details = `${source.faction}. ${source.specialty}. Important locals: ${source.locals}.`;
          usedNames.add(name);
        } else {
          name = mapName(rng, usedNames);
          details = type === 'station' ? 'Friendly or contested station generated for trade, jobs, resupply, and upkeep.'
            : type === 'outpost' ? 'A small pressure outpost with limited stores and uncertain loyalties.'
            : type === 'wreck' ? 'A wreck or salvage site whose value may exceed its safety.'
            : type === 'ruins' ? 'Ancient structures or abandoned research installations.'
            : 'A route hazard: currents, storms, cave pressure, fauna, or unstable ice.';
        }
        nodes.push({
          id: `node-${ring}-${index}`,
          name, type, ring,
          x: center.x + Math.cos(angle) * (radius + jitter),
          y: center.y + Math.sin(angle) * (radius + jitter),
          details,
          pressure: Math.max(1, rings - ring + 1),
          outerBias
        });
      }
    }

    const edges = [];
    const edgeKeys = new Set();
    const addEdge = (a, b, extraDanger = false) => {
      if (!a || !b || a.id === b.id) return;
      const key = [a.id, b.id].sort().join('|');
      if (edgeKeys.has(key)) return;
      edgeKeys.add(key);
      const distance = 1 + Math.floor(rng() * 5);
      const danger = extraDanger || a.type === 'hazard' || b.type === 'hazard' || rng() < .16;
      edges.push({ id: `edge-${edges.length + 1}`, a: a.id, b: b.id, distance, danger });
    };

    for (let ring = 1; ring <= rings; ring += 1) {
      const current = nodes.filter(node => node.ring === ring);
      const inward = nodes.filter(node => node.ring === ring - 1);
      for (const node of current) {
        const nearest = [...inward].sort((left, right) => Math.hypot(node.x - left.x, node.y - left.y) - Math.hypot(node.x - right.x, node.y - right.y));
        addEdge(node, nearest[0]);
        if (nearest[1] && rng() < number(world.branching) / 100) addEdge(node, nearest[1]);
      }
      for (let index = 0; index < current.length; index += 1) {
        if (rng() < number(world.branching) / 100) addEdge(current[index], current[(index + 1) % current.length]);
      }
    }

    const outerStations = nodes.filter(node => node.ring === rings && node.type === 'station');
    const start = outerStations[0] || nodes.find(node => node.ring === rings) || nodes[nodes.length - 1];
    start.start = true;
    world.map = { seed: world.seed, width, height, rings, nodes, edges, startId: start.id, generatedAt: new Date().toISOString() };
    world.selected = [start.id, 'anomaly'];
    save();
  }

  function shortestPath(map, startId, endId) {
    if (!map || !startId || !endId) return null;
    const distances = Object.fromEntries(map.nodes.map(node => [node.id, Infinity]));
    const previous = {};
    const unvisited = new Set(map.nodes.map(node => node.id));
    distances[startId] = 0;
    while (unvisited.size) {
      let current = null;
      let best = Infinity;
      for (const id of unvisited) if (distances[id] < best) { best = distances[id]; current = id; }
      if (current === null || best === Infinity) break;
      unvisited.delete(current);
      if (current === endId) break;
      for (const edge of map.edges.filter(item => item.a === current || item.b === current)) {
        const neighbor = edge.a === current ? edge.b : edge.a;
        if (!unvisited.has(neighbor)) continue;
        const alternative = distances[current] + number(edge.distance, 1) + (edge.danger ? 1 : 0);
        if (alternative < distances[neighbor]) { distances[neighbor] = alternative; previous[neighbor] = current; }
      }
    }
    if (!Number.isFinite(distances[endId])) return null;
    const ids = [];
    let cursor = endId;
    while (cursor) { ids.unshift(cursor); if (cursor === startId) break; cursor = previous[cursor]; }
    if (ids[0] !== startId) return null;
    return { ids, distance: distances[endId] };
  }

  function mapSvg(map, selected = []) {
    if (!map) return '';
    const nodeById = Object.fromEntries(map.nodes.map(node => [node.id, node]));
    const path = selected.length === 2 ? shortestPath(map, selected[0], selected[1]) : null;
    const pathEdges = new Set();
    if (path) for (let index = 0; index < path.ids.length - 1; index += 1) pathEdges.add([path.ids[index], path.ids[index + 1]].sort().join('|'));
    return `<svg id="world-map-svg" viewBox="0 0 ${map.width} ${map.height}" role="img" aria-label="Generated Europa campaign route map">
      <defs><radialGradient id="anomalyGlow"><stop offset="0" stop-color="#efcb7e" stop-opacity=".55"/><stop offset="1" stop-color="#efcb7e" stop-opacity="0"/></radialGradient></defs>
      <circle cx="${map.width / 2}" cy="${map.height / 2}" r="92" fill="url(#anomalyGlow)"/>
      ${map.edges.map(edge => {
        const a = nodeById[edge.a]; const b = nodeById[edge.b]; const key = [edge.a, edge.b].sort().join('|');
        return `<g><line class="map-edge ${edge.danger ? 'danger' : ''}" x1="${a.x.toFixed(1)}" y1="${a.y.toFixed(1)}" x2="${b.x.toFixed(1)}" y2="${b.y.toFixed(1)}" ${pathEdges.has(key) ? 'style="stroke:#e9c36d;stroke-width:5"' : ''}/><text x="${((a.x+b.x)/2).toFixed(1)}" y="${((a.y+b.y)/2).toFixed(1)}" fill="#929aa4" font-size="10">${edge.distance}</text></g>`;
      }).join('')}
      ${map.nodes.map(node => `<g class="map-node ${escapeHtml(node.type)} ${selected.includes(node.id) ? 'selected' : ''}" data-node-id="${escapeHtml(node.id)}" transform="translate(${node.x.toFixed(1)} ${node.y.toFixed(1)})"><circle r="${node.type === 'anomaly' ? 15 : node.start ? 12 : 9}"/><text x="${node.type === 'anomaly' ? 20 : 14}" y="4">${escapeHtml(node.start ? `START: ${node.name}` : node.name)}</text></g>`).join('')}
    </svg>`;
  }
