  function deepMerge(base, incoming) {
    if (!incoming || typeof incoming !== 'object') return base;
    for (const [key, value] of Object.entries(incoming)) {
      if (Array.isArray(value)) base[key] = value;
      else if (value && typeof value === 'object') base[key] = deepMerge(base[key] && typeof base[key] === 'object' ? base[key] : {}, value);
      else base[key] = value;
    }
    return base;
  }

  let state = (() => {
    try {
      const stored = localStorage.getItem(STORAGE_KEY);
      return stored ? deepMerge(defaultState(), JSON.parse(stored)) : defaultState();
    } catch (error) {
      console.warn('Could not load Barotrauma RPG suite state.', error);
      return defaultState();
    }
  })();

  const escapeHtml = value => String(value ?? '').replace(/[&<>"']/g, character => ({ '&': '&amp;', '<': '&lt;', '>': '&gt;', '"': '&quot;', "'": '&#39;' }[character]));
  const number = (value, fallback = 0) => Number.isFinite(Number(value)) ? Number(value) : fallback;
  const clamp = (value, min, max) => Math.min(max, Math.max(min, number(value, min)));
  const d6 = () => Math.floor(Math.random() * 6) + 1;
  const uid = prefix => `${prefix}-${Date.now().toString(36)}-${Math.random().toString(36).slice(2, 8)}`;

  function save() {
    try { localStorage.setItem(STORAGE_KEY, JSON.stringify(state)); }
    catch (error) { console.warn('Could not save Barotrauma RPG suite state.', error); }
  }

  function download(name, text, type = 'application/json') {
    const blob = new Blob([text], { type });
    const url = URL.createObjectURL(blob);
    const anchor = document.createElement('a');
    anchor.href = url;
    anchor.download = name;
    document.body.appendChild(anchor);
    anchor.click();
    anchor.remove();
    URL.revokeObjectURL(url);
  }

  function optionList(values, selected, label = value => value) {
    return values.map(value => `<option value="${escapeHtml(value)}" ${value === selected ? 'selected' : ''}>${escapeHtml(label(value))}</option>`).join('');
  }

  function activeTool() {
    const value = window.location.hash.replace(/^#/, '');
    return ['character', 'submarine', 'encounters', 'route', 'world'].includes(value) ? value : 'character';
  }
