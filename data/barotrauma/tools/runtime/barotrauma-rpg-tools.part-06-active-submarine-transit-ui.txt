  function dashboardAdjacentStationMapSvg(context, routes) {
    const current = crossingNode(context.group.currentNodeId);
    if (!current) return '';
    const nodes = [current, ...routes.map(route => route.station)];
    const xs = nodes.map(node => number(node.x));
    const ys = nodes.map(node => number(node.y));
    const minX = Math.min(...xs) - 100;
    const maxX = Math.max(...xs) + 100;
    const minY = Math.min(...ys) - 100;
    const maxY = Math.max(...ys) + 100;
    const width = Math.max(280, maxX - minX);
    const height = Math.max(220, maxY - minY);
    return `<svg viewBox="${minX} ${minY} ${width} ${height}" role="img" aria-label="Immediately adjacent station routes" style="width:100%;height:500px;display:block;background:radial-gradient(circle at center,rgba(200,138,53,.08),rgba(0,0,0,.35));border-radius:14px">${routes.map(route => `<g><line x1="${number(current.x)}" y1="${number(current.y)}" x2="${number(route.station.x)}" y2="${number(route.station.y)}" stroke="${route.danger ? '#c45e68' : '#57b9ff'}" stroke-width="${route.stationId === state.route.endNodeId ? 10 : 5}"/><text x="${(number(current.x) + number(route.station.x)) / 2}" y="${(number(current.y) + number(route.station.y)) / 2 - 8}" fill="#d5dce4" font-size="13" text-anchor="middle">${route.distance}${route.danger ?
