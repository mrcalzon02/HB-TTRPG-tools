
  const DASHBOARD_LOCAL_LATTICE_VIEWER_VERSION='1.0.0';

  dashboardIncrementalLocalCorridorsForRing=function dashboardIncrementalLocalWorldEdgesForRing(packageInfo,map=state.world?.map){
    if(typeof dashboardLocalWorldTopology!=='function')return [];
    const topology=dashboardLocalWorldTopology(map);
    return topology.edges.filter(edge=>{
      const left=topology.nodeById[edge.a];
      const right=topology.nodeById[edge.b];
      if(!left||!right)return false;
      const leftRing=Math.round(number(left.ring));
      const rightRing=Math.round(number(right.ring));
      const difference=Math.abs(leftRing-rightRing);
      if(difference>1)return false;
      const displayRing=difference===0?leftRing:Math.min(leftRing,rightRing);
      return displayRing===packageInfo.ring;
    });
  };

  dashboardIncrementalAddLocalCorridors=function dashboardIncrementalAddLocalWorldLattice(packageInfo,map=state.world?.map){
    const layer=root.querySelector('#dashboard-incremental-world-corridors');
    if(!layer||typeof dashboardLocalWorldTopology!=='function')return 0;
    const topology=dashboardLocalWorldTopology(map);
    const fragment=document.createDocumentFragment();
    let added=0;
    for(const edge of dashboardIncrementalLocalCorridorsForRing(packageInfo,map)){
      if(layer.querySelector(`[data-local-corridor-id="${CSS.escape(edge.id)}"]`))continue;
      const left=topology.nodeById[edge.a];
      const right=topology.nodeById[edge.b];
      if(!left||!right)continue;
      const ringDifference=Math.abs(number(left.ring)-number(right.ring));
      if(ringDifference>1)continue;
      const line=document.createElementNS(DASHBOARD_INCREMENTAL_SVG_NS,'line');
      line.dataset.localCorridorId=edge.id;
      line.dataset.localRelation=edge.relation;
      line.setAttribute('x1',String(number(left.x)));
      line.setAttribute('y1',String(number(left.y)));
      line.setAttribute('x2',String(number(right.x)));
      line.setAttribute('y2',String(number(right.y)));
      const stationEndpoint=left.type==='station'||right.type==='station';
      line.setAttribute('stroke',edge.danger?'#9d515a':edge.relation==='one-ring-inward'?'#397ea8':'#536f5c');
      line.setAttribute('stroke-width',stationEndpoint?'2.2':'1.15');
      line.setAttribute('opacity',stationEndpoint?'.72':'.42');
      line.setAttribute('vector-effect','non-scaling-stroke');
      const title=document.createElementNS(DASHBOARD_INCREMENTAL_SVG_NS,'title');
      title.textContent=`${left.name||left.id} ↔ ${right.name||right.id} · ${edge.relation.replaceAll('-',' ')} · one local segment`;
      line.appendChild(title);
      fragment.appendChild(line);
      added+=1;
    }
    layer.appendChild(fragment);
    return added;
  };

  const dashboardIncrementalWorldViewerHtmlBeforeLocalLattice=dashboardIncrementalWorldViewerHtml;
  dashboardIncrementalWorldViewerHtml=function dashboardIncrementalWorldViewerHtmlWithCompleteLocalLattice(context){
    return dashboardIncrementalWorldViewerHtmlBeforeLocalLattice(context)
      .replace('Only same-ring neighboring stations and one-ring-down station corridors are drawn.','The complete local location lattice is drawn: immediate neighbors around each ring and one short step inward, continuing to the central anomaly.')
      .replace('Incremental saved Masterworld viewer','Incremental saved Masterworld local-lattice viewer');
  };

  const latticeTopology=typeof dashboardLocalWorldTopology==='function'&&dashboardMasterWorldMapValid()?dashboardLocalWorldTopology():{edges:[],nodeById:{}};
  const maximumRingDifference=Math.max(0,...latticeTopology.edges.map(edge=>Math.abs(number(latticeTopology.nodeById[edge.a]?.ring)-number(latticeTopology.nodeById[edge.b]?.ring))));
  const maximumGeometricSegment=Math.max(0,...latticeTopology.edges.map(edge=>number(edge.geometric,0)));
  globalThis.BAROTRAUMA_DASHBOARD_LOCAL_LATTICE_VIEWER_DIAGNOSTICS={
    version:DASHBOARD_LOCAL_LATTICE_VIEWER_VERSION,
    nodeCount:Object.keys(latticeTopology.nodeById).length,
    edgeCount:latticeTopology.edges.length,
    maximumRingDifference,
    maximumGeometricSegment,
    sameRingNeighborsOnly:latticeTopology.edges.filter(edge=>edge.relation==='same-ring-adjacent').every(edge=>number(latticeTopology.nodeById[edge.a]?.ring)===number(latticeTopology.nodeById[edge.b]?.ring)),
    oneRingInwardOnly:latticeTopology.edges.filter(edge=>edge.relation==='one-ring-inward').every(edge=>Math.abs(number(latticeTopology.nodeById[edge.a]?.ring)-number(latticeTopology.nodeById[edge.b]?.ring))===1),
    reachesCentralAnomaly:latticeTopology.edges.some(edge=>edge.a==='anomaly'||edge.b==='anomaly'),
    directStationChordsRendered:false,
    completeLocalLatticeRendered:true
  };
