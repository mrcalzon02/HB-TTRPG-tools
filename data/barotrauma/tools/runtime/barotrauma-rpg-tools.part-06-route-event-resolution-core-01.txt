
  function routeResolutionStepBlueprints(event, solution, pursueOpportunity = false) {
    const profile = routeResolutionProfile(event);
    const descriptions = profile.descriptions || ROUTE_RESOLUTION_FAMILY_PROFILES.Custom.descriptions;
    const clock = clamp(number(event?.encounterClock,3),2,4);
    const phases = clock === 2 ? ['Assessment','Execution'] : clock === 3 ? ['Assessment','Execution','Stabilization'] : ['Assessment','Execution','Containment','Stabilization'];
    const offsets = clock === 2 ? [-1,0] : clock === 3 ? [-1,0,0] : [-1,0,1,0];
    const steps = phases.map((phase,index) => {
      const requiredRole = index === 1 ? solution.leadRole : (index === 0 ? profile.supportRoles?.[0] : profile.supportRoles?.[index % Math.max(1,profile.supportRoles.length)]) || solution.leadRole;
      const stat = index === 0 && solution.stat !== 'Intelligence' ? 'Intelligence' : index === phases.length - 1 && solution.stat === 'Strength' ? 'Intelligence' : solution.stat;
      return { id:`${phase.toLowerCase()}-${index+1}`, phase, stat, dc:clamp(solution.dc+offsets[index],5,20), requiredRole,
        description:`${descriptions[Math.min(index,descriptions.length-1)]} ${index===1?solution.description:''}`.trim(),
        status:'pending', actorRef:'', actorName:'', actorDuty:'', dice:[], modifier:0, roleBonus:0, supportBonus:0, total:0, margin:0, rollText:'' };
    });
    if (pursueOpportunity) steps.push({ id:'opportunity-contract', phase:'Opportunity / Contract', stat:event?.category==='Faction'||event?.category==='Distress'?'Charisma':'Intelligence', dc:clamp(solution.dc+1,6,20), requiredRole:profile.leadRoles?.[1]||'Command',
      description:`Go beyond immediate survival to pursue the mission, contract, recoverable opportunity, or strategic lead: ${event?.encounterOpportunity||event?.reward||'secure additional value from the event'}.`,
      status:'pending', actorRef:'', actorName:'', actorDuty:'', dice:[], modifier:0, roleBonus:0, supportBonus:0, total:0, margin:0, rollText:'' });
    return steps;
  }

  function defaultRouteResolutionSession(slot,event) {
    const solutions = routeResolutionSolutions(event);
    const selected = solutions[0];
    const turnOrder = normalizeDashboardTurnOrder();
    const actor = activeDashboardTurnActor();
    return { version:ROUTE_EVENT_RESOLUTION_VERSION, id:`resolution-${slot.id}`, slotId:slot.id, eventId:event?.id||'', encounterTemplateId:event?.encounterTemplateId||'', crossingId:normalizeRouteCrossing().crossingId||'', masterWorldId:typeof synchronizeMasterWorldId==='function'?synchronizeMasterWorldId(false):normalizeWorldHub().worldId, submarineId:state.submarine.worldSubmarineId||'', groupId:normalizeWorldHub().activeGroupId||'', state:'planning', selectedSolutionId:selected?.id||'', solutions, supportCrewRefs:[], pursueOpportunity:false, acceptContract:false, missionOffer:routeResolutionMissionOffer(event,slot), steps:routeResolutionStepBlueprints(event,selected,false), activeStepIndex:0, provisionalOutcome:'', outcomeOverride:'', outcome:'', effectSummary:'', appliedEffects:{}, leadCrewRef:actor?.ref||'', leadCrewName:actor?.name||'', createdAt:new Date().toISOString(), resolvedAt:'', roundStarted:turnOrder.round, turnStarted:turnOrder.turn, log:[] };
  }

  function normalizeRouteResolutionSession(slot,event) {
    const resolution = normalizeRouteEventResolutionState();
    let session = resolution.sessions[slot.id];
    if (!session || session.eventId !== (event?.id||'') || session.encounterTemplateId !== (event?.encounterTemplateId||'')) {
      session = defaultRouteResolutionSession(slot,event);
      resolution.sessions[slot.id] = session;
    }
    session.solutions = Array.isArray(session.solutions)&&session.solutions.length?session.solutions:routeResolutionSolutions(event);
    if (!session.solutions.some(solution => solution.id===session.selectedSolutionId)) session.selectedSolutionId = session.solutions[0]?.id||'';
    session.supportCrewRefs = Array.isArray(session.supportCrewRefs)?session.supportCrewRefs:[];
    session.steps = Array.isArray(session.steps)&&session.steps.length?session.steps:routeResolutionStepBlueprints(event,routeResolutionSelectedSolution(session),session.pursueOpportunity);
    session.activeStepIndex = clamp(session.activeStepIndex,0,Math.max(0,session.steps.length-1));
    session.log = Array.isArray(session.log)?session.log.slice(0,100):[];
    return session;
  }

  function activeRouteResolutionContext() {
    const slot = nextPendingCrossingSlot();
    const event = crossingSelectedEvent(slot);
    if (!slot||!event) return null;
    return {slot,event,session:normalizeRouteResolutionSession(slot,event)};
  }

  function routeResolutionSelectedSolution(session) { return session.solutions.find(solution => solution.id===session.selectedSolutionId)||session.solutions[0]||null; }

  function resetRouteResolutionSteps(session,event) {
    session.steps = routeResolutionStepBlueprints(event,routeResolutionSelectedSolution(session),session.pursueOpportunity);
    session.activeStepIndex = 0;
    session.state = 'planning';
    session.provisionalOutcome = '';
    session.outcomeOverride = '';
    session.outcome = '';
    session.effectSummary = '';
    session.appliedEffects = {};
    const actor = activeDashboardTurnActor();
    session.leadCrewRef = actor?.ref||'';
    session.leadCrewName = actor?.name||'';
    session.log.unshift({at:new Date().toISOString(),text:'Resolution plan reset.'});
    save();
  }

  function selectRouteResolutionSolution(solutionId) {
    const context = activeRouteResolutionContext();
    if (!context||context.session.steps.some(step => step.status!=='pending')) return false;
    if (!context.session.solutions.some(solution => solution.id===solutionId)) return false;
    context.session.selectedSolutionId = solutionId;
    resetRouteResolutionSteps(context.session,context.event);
    return true;
  }

  function setRouteResolutionOpportunity(enabled) {
    const context = activeRouteResolutionContext();
    if (!context||context.session.steps.some(step => step.status!=='pending')) return false;
    context.session.pursueOpportunity = Boolean(enabled);
    resetRouteResolutionSteps(context.session,context.event);
    return true;
  }

  function setRouteResolutionContract(enabled) {
    const context = activeRouteResolutionContext();
    if (!context||context.session.steps.some(step => step.status!=='pending')) return false;
    context.session.acceptContract = Boolean(enabled);
    save();
    return true;
  }

  function setRouteResolutionSupportCrew(refs) {
    const context = activeRouteResolutionContext();
    if (!context||context.session.steps.some(step => step.status!=='pending')) return false;
    const valid = new Set(dashboardTurnActors().map(actor => actor.ref));
    context.session.supportCrewRefs = [...new Set(refs||[])].filter(ref => valid.has(ref)).slice(0,3);
    save();
    return true;
  }

  function routeResolutionActorDuty(actor) {
    if (!actor) return 'Unassigned';
    if (actor.ref==='local') return normalizePlayerSubmissionSettings().preferredDuty||'Unassigned';
    return normalizeCrewManagement().roster.find(record => record.rosterId===actor.ref)?.duty||'Unassigned';
  }

  function routeResolutionRoleBonus(actor,step,solution) {
    const duty = routeResolutionActorDuty(actor);
    const allowed = new Set([step.requiredRole,solution.leadRole,...(solution.supportRoles||[]),...(ROUTE_RESOLUTION_STAT_ROLES[step.stat]||[])]);
    if (allowed.has(duty)) return duty===step.requiredRole||duty===solution.leadRole?2:1;
    return 0;
  }

  function routeResolutionSupportBonus(session,step,activeActor) {
    const actors = dashboardTurnActors();
    let bonus = 0;
    for (const ref of session.supportCrewRefs) {
      const actor = actors.find(candidate => candidate.ref===ref);
      if (!actor||actor.ref===activeActor?.ref) continue;
      const duty = routeResolutionActorDuty(actor);
      if (duty===step.requiredRole||(ROUTE_RESOLUTION_STAT_ROLES[step.stat]||[]).includes(duty)) bonus += 1;
    }
    return Math.min(2,bonus);
  }

  function calculateRouteResolutionOutcome(session) {
    const completed = session.steps.filter(step => ['success','failure'].includes(step.status));
    if (completed.length!==session.steps.length||!completed.length) return '';
    const successes = completed.filter(step => step.status==='success').length;
    const failures = completed.length-successes;
    const margin = completed.reduce((sum,step) => sum+number(step.margin),0);
    if (successes===completed.length&&margin>=4) return 'critical-success';
    if (successes>failures) return 'success';
    if (successes===failures||successes>0) return 'mixed';
    if (failures===completed.length&&margin<=-4) return 'critical-failure';
    return 'failure';
  }

  function rollRouteResolutionStep() {
    const context = activeRouteResolutionContext();
    if (!context) return {ok:false,message:'No unresolved route event is available.'};
    if (normalizeRouteCrossing().status!=='underway') return {ok:false,message:'Turn-based event resolution is available only while the crossing is underway.'};
    const session = context.session;
    const step = session.steps[session.activeStepIndex];
    if (!step||step.status!=='pending') return {ok:false,message:'The active resolution step is not available.'};
    const actor = activeDashboardTurnActor();
    if (!actor) return {ok:false,message:'No living onboard crew member is available to take this turn.'};
    const solution = routeResolutionSelectedSolution(session);
    const stat = actor.current&&step.stat in actor.current?step.stat:'Intelligence';
    const dice = [d6(),d6()];
    const modifier = number(actor.current?.[stat]);
    const roleBonus = routeResolutionRoleBonus(actor,step,solution);
    const supportBonus = routeResolutionSupportBonus(session,step,actor);
    const total = dice[0]+dice[1]+modifier+roleBonus+supportBonus;
    step.status = total>=number(step.dc,8)?'success':'failure';
    Object.assign(step,{actorRef:actor.ref,actorName:actor.name,actorDuty:routeResolutionActorDuty(actor),stat,dice,modifier,roleBonus,supportBonus,total,margin:total-number(step.dc,8)});
    step.rollText = `${actor.name}: ${dice[0]} + ${dice[1]} + ${modifier} ${stat} + ${roleBonus} role + ${supportBonus} support = ${total} vs ${step.dc}`;
    if (!session.leadCrewRef) { session.leadCrewRef=actor.ref; session.leadCrewName=actor.name; }
    session.log.unshift({at:new Date().toISOString(),text:`${step.phase}: ${step.rollText} — ${step.status}.`});
    session.state = 'resolving';
    if (session.activeStepIndex<session.steps.length-1) {
      session.activeStepIndex += 1;
      advanceDashboardTurn(`${actor.name} completed ${step.phase} for ${context.event.name}.`);
    } else {
      session.provisionalOutcome = calculateRouteResolutionOutcome(session);
      session.state = 'awaiting-commit';
    }
    save();
    return {ok:true,message:step.rollText,status:step.status};
  }
